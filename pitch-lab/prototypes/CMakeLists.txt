# ---------------------------------------------------------------------------
# Pitch Lab — Task 28 candidate prototypes (research-only).
#
# BOUNDARY (task-28 rules): these targets are OUTSIDE the production
# engine registry, the VST3 product layer and the v0.1 result product.
# They are clean-room own implementations from published algorithm
# descriptions (OLA / WSOLA / TD-PSOLA / FD-PSOLA / transient-aware PV).
# The frozen v0.1 sources are reused READ-ONLY (resampler, FFT, tracker,
# windows, WAV, RNG, hash, JSON writer, the five engines for the
# differentiation baseline).
#
# WARNING: pitchlab_proto28 contains GLOBAL operator new/delete
# replacements (the allocation audit). It must never be linked into a
# product target — task28 executables only.
#
# ctest entries run the FAST selftests (--selftest); the full evaluation
# artifacts under results/research/task28-candidates/ are generated by
# running the executables with --out (see that directory's README).
# ---------------------------------------------------------------------------

add_library(pitchlab_proto28
  proto_corpus.cpp
  proto_measure.cpp
  proto_engine.cpp
  proto_alloc.cpp
  proto_baseline.cpp
  proto_run.cpp
  candidate_ola.cpp
  candidate_wsola.cpp
  candidate_tdpsola.cpp
)
target_include_directories(pitchlab_proto28 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_compile_features(pitchlab_proto28 PUBLIC cxx_std_20)
target_link_libraries(pitchlab_proto28 PUBLIC pitchlab_core)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  target_compile_options(pitchlab_proto28 PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  if(PITCHLAB_WERROR)
    target_compile_options(pitchlab_proto28 PRIVATE -Werror)
  endif()
endif()

function(task28_add_executable name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE pitchlab_proto28)
  target_compile_features(${name} PRIVATE cxx_std_20)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    if(PITCHLAB_WERROR)
      target_compile_options(${name} PRIVATE -Werror)
    endif()
  endif()
  add_test(NAME ${name}_selftest COMMAND ${name} --selftest)
endfunction()

task28_add_executable(task28_baseline task28_baseline_main.cpp)
task28_add_executable(task28_ola task28_ola_main.cpp)

task28_add_executable(task28_wsola task28_wsola_main.cpp)

task28_add_executable(task28_tdpsola task28_tdpsola_main.cpp)
