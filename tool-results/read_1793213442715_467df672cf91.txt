    42→    42→/// 64-bit LCG, identical on the Python side. u in [0,1); sample = (2u-1)*amp.
    43→    43→struct Lcg {
    44→    44→    uint64_t s;
    45→    45→    explicit Lcg(uint64_t seed) : s(seed * 6364136223846793005ULL +
    46→    46→                                     1442695040888963407ULL) {}
    47→    47→    double next() {
    48→    48→        s = s * 6364136223846793005ULL + 1442695040888963407ULL;
    49→    49→        const uint64_t m = (s >> 11) & ((1ULL << 53) - 1);
    50→    50→        return static_cast<double>(m) / 9007199254740992.0;
    51→    51→    }
    52→    52→    double noise(double amp) { return (2.0 * next() - 1.0) * amp; }
    53→    53→};

   117→   117→/// Radix-2 complex FFT (in-place, n a power of two).
   118→   118→void fft(double* re, double* im, int n) {
   119→   119→    for (int i = 1, j = 0; i < n; ++i) {
   120→   120→        int bit = n >> 1;
   121→   121→        for (; j & bit; bit >>= 1) j ^= bit;
   122→   122→        j ^= bit;
   123→   123→        if (i < j) {
   124→   124→            std::swap(re[i], re[j]);
   125→   125→            std::swap(im[i], im[j]);
   126→   126→        }
   127→   127→    }
   128→   128→    for (int len = 2; len <= n; len <<= 1) {
   129→   129→        const double ang = -2.0 * M_PI / len;
   130→   130→        const double wr = std::cos(ang), wi = std::sin(ang);
   131→   131→        for (int i = 0; i < n; i += len) {
   132→   132→            double cr = 1.0, ci = 0.0;
   133→   133→            for (int k = 0; k < len / 2; ++k) {
   134→   134→                const double ur = re[i + k], ui = im[i + k];
   135→   135→                const double vr = re[i + k + len / 2] * cr -
   136→   136→                                  im[i + k + len / 2] * ci;
   137→   137→                const double vi = re[i + k + len / 2] * ci +
   138→   138→                                  im[i + k + len / 2] * cr;
   139→   139→                re[i + k] = ur + vr;
   140→   140→                im[i + k] = ui + vi;
   141→   141→                re[i + k + len / 2] = ur - vr;
   142→   142→                im[i + k + len / 2] = ui - vi;
   143→   143→                const double nr = cr * wr - ci * wi;
   144→   144→                ci = cr * wi + ci * wr;
   145→   145→                cr = nr;
   146→   146→            }
   147→   147→        }
   148→   148→    }
   149→   149→}
   150→   150→
   151→   151→/// FFT peak pitch with the CORRECTED Hann-exact 3-point estimator.
   152→   152→/// Returns 0.0 when no peak above the threshold is found.
   153→   153→///
   154→   154→/// R15-REB1 fix (recovered from the record): the interpolation now runs on
   155→   155→/// the three bins' MAGNITUDES with the Hann-exact formula
   156→   156→///     delta = 2*(c - a) / (a + 2*b + c)
   157→   157→/// (a, b, c = magnitudes of bins kpeak-1, kpeak, kpeak+1) instead of the
   158→   158→/// old power-domain parabola 0.5*(a-c)/(a-2b+c) on squared magnitudes,
   159→   159→/// which is biased for off-bin tones under a Hann window (the historical
   160→   160→/// round-15 record: the correction made battery cases PASS, none failed).
   161→   161→///
   162→   162→/// FFT size N = the largest power of two <= min(n, 65536): the analysed
   163→   163→/// NEWEST block of N samples always carries a complete Hann window, and
   164→   164→/// the window offset off = n - N is >= 0 BY CONSTRUCTION (the estimator
   165→   165→/// never reads before the supplied buffer start).
   166→   166→double pitchFFT(const float* x, int n, int sr, double fmin, double fmax) {
   167→   167→    int N = 1;
   168→   168→    while (N < n && N < 65536) N <<= 1;  // largest pow2 <= min(n, 65536)
   169→   169→    if (N < 16) return 0.0;
   170→   170→    std::vector<double> re(N, 0.0), im(N, 0.0);
   171→   171→    const int off = n - N;  // use the newest N samples (off >= 0 by construction)
   172→   172→    for (int i = 0; i < N; ++i) {
   173→   173→        const double w = 0.5 - 0.5 * std::cos(2.0 * M_PI * i / N);
   174→   174→        re[i] = static_cast<double>(x[off + i]) * w;
   175→   175→    }
   176→   176→    fft(re.data(), im.data(), N);
   177→   177→    const int kmin = std::max(1, static_cast<int>(fmin * N / sr));
   178→   178→    const int kmax = std::min(N / 2 - 2, static_cast<int>(fmax * N / sr));
   179→   179→    if (kmax <= kmin) return 0.0;
   180→   180→    int kpeak = kmin;
   181→   181→    double best = 0.0;
   182→   182→    for (int k = kmin; k <= kmax; ++k) {
   183→   183→        const double m = re[k] * re[k] + im[k] * im[k];  // power (same argmax)
   184→   184→        if (m > best) {
   185→   185→            best = m;
   186→   186→            kpeak = k;
   187→   187→        }
   188→   188→    }
   189→   189→    if (best <= 0.0) return 0.0;
   190→   190→    // magnitudes of the three bins around the peak (Hann-exact estimator)
   191→   191→    const double a = std::sqrt(re[kpeak - 1] * re[kpeak - 1] +
   192→   192→                               im[kpeak - 1] * im[kpeak - 1]);
   193→   193→    const double b = std::sqrt(best);
   194→   194→    const double c = std::sqrt(re[kpeak + 1] * re[kpeak + 1] +
   195→   195→                               im[kpeak + 1] * im[kpeak + 1]);
   196→   196→    double d = 0.0;
   197→   197→    const double den = a + 2.0 * b + c;
   198→   198→    if (den > 1e-300) d = 2.0 * (c - a) / den;
   199→   199→    if (d > 0.5) d = 0.5;
   200→   200→    if (d < -0.5) d = -0.5;
   201→   201→    return static_cast<double>(sr) * (kpeak + d) / N;
   202→   202→}
   203→   203→
   204→   204→double rms(const float* x, int n) {
   205→   205→    double s = 0.0;
   206→   206→    for (int i = 0; i < n; ++i) s += static_cast<double>(x[i]) * x[i];
   207→   207→    return n ? std::sqrt(s / n) : 0.0;
   208→   208→}
   209→   209→
   210→   210→double peak(const float* x, int n) {
   211→   211→    double p = 0.0;
   212→   212→    for (int i = 0; i < n; ++i) p = std::max(p, std::abs(static_cast<double>(x[i])));
   213→   213→    return p;
   214→   214→}
   215→   215→
   216→   216→bool allFinite(const float* x, int n) {
   217→   217→    for (int i = 0; i < n; ++i)
   218→   218→        if (!std::isfinite(x[i])) return false;
   219→   219→    return true;
   220→   220→}

   222→   222→/// Envelope modulation in dB: max/min of RMS windows of `w` samples over the
   223→   223→/// steady region (skip the first skip samples). w = sr/1000 gives the 1 ms
   224→   224→/// metric; w = period gives the period-synchronous metric (the r11 lesson:
   225→   225→/// a 1 ms window phase-breathes several dB on a plain tone).
   226→   226→double envModulationDb(const float* x, int n, int sr, int skip, int w) {
   227→   227→    std::vector<double> env;
   228→   228→    for (int i0 = skip; i0 + w <= n; i0 += w) {
   229→   229→        env.push_back(rms(x + i0, w));
   230→   230→    }
   231→   231→    if (env.empty()) return 0.0;
   232→   232→    double lo = 1e300, hi = 0.0;
   233→   233→    for (double v : env) {
   234→   234→        if (v < lo) lo = v;
   235→   235→        if (v > hi) hi = v;
   236→   236→    }
   237→   237→    if (lo <= 0.0) return hi > 0.0 ? 999.0 : 0.0;
   238→   238→    return 20.0 * std::log10(hi / lo);
   239→   239→}

   241→   241→double centsBetween(double fMeasured, double fNominal) {
   242→   242→    if (fMeasured <= 0.0 || fNominal <= 0.0) return 999.0;
   243→   243→    return 1200.0 * std::log2(fMeasured / fNominal);
   244→   244→}

   246→   246→/// Max absolute sample-to-sample difference (transition click meter).
   247→   247→double maxStep(const float* x, int n, int i0 = 0) {
   248→   248→    double m = 0.0;
   249→   249→    for (int i = std::max(1, i0 + 1); i < n; ++i)
   250→   250→        m = std::max(m, std::abs(static_cast<double>(x[i]) - x[i - 1]));
   251→   251→    return m;
   252→   252→}

   279→   279→// Runs the engine over a full signal in one go (64-aligned internally is
   280→   280→// irrelevant -- process takes the whole block).
   281→   281→std::vector<float> run(const Config& cfg, const std::vector<float>& in,
   282→   282→                       int block = 1 << 20) {
   283→   283→    AlignedDelayLine e;
   284→   284→    if (!e.prepare(cfg)) {
   285→   285→        return {};
   286→   286→    }
   287→   287→    std::vector<float> out(in.size(), 0.0f);
   288→   288→    int done = 0;
   289→   289→    while (done < static_cast<int>(in.size())) {
   290→   290→        const int n = std::min(block, static_cast<int>(in.size()) - done);
   291→   291→        e.process(in.data() + done, out.data() + done, n);
   292→   292→        done += n;
   293→   293→    }
   294→   294→    return out;
   295→   295→}

   300→   300→void casesPrepareReset(std::vector<CaseResult>& R) {
   301→   301→    {  // 1. default prepare

   321→   321→        bad = Config();

   323→   323→        tryCfg(bad);
   324→   324→        bad = Config();
   325→   325→        bad.voices = 9;
   326→   326→        tryCfg(bad);
   327→   327→        bad = Config();
   328→   328→        bad.grain = 0.0;
   329→   329→        tryCfg(bad);
   330→   330→        bad = Config();
   331→   331→        bad.feedback = 1.0;
   332→   332→        tryCfg(bad);
   333→   333→        bad = Config();
   334→   334→        bad.base_delay = 0.0;
   335→   335→        tryCfg(bad);
   336→   336→        addValue(c, "rejected", rejected);
   337→   337→        addValue(c, "total", total);
   338→   338→        check(c, rejected == total, "some invalid configs accepted");
   339→   339→        c.pass = c.note.empty();
   340→   340→        R.push_back(std::move(c));
   341→   341→    }
   342→   342→    {  // 3. reset determinism
   343→   343→        CaseResult c{"reset_determinism"};
   344→   344→        Config cfg;
   345→   345→        cfg.ratio = semitoneRatio(400.0);
   346→   346→        const auto in = gen::tone(220.0, 48000, 48000, 0.5);
   347→   347→        AlignedDelayLine a, b;
   348→   348→        a.prepare(cfg);
   349→   349→        b.prepare(cfg);
   350→   350→        std::vector<float> oa(in.size(), 0.f), ob(in.size(), 0.f),
   351→   351→            ob2(in.size(), 0.f);
   352→   352→        a.process(in.data(), oa.data(), in.size());
   353→   353→        b.process(in.data(), ob.data(), in.size());
   354→   354→        b.reset();
   355→   355→        b.process(in.data(), ob2.data(), in.size());
   356→   356→        double d1 = 0.0, d2 = 0.0;
   357→   357→        for (size_t i = 0; i < in.size(); ++i) {
   358→   358→            d1 = std::max(d1, std::abs((double)oa[i] - ob[i]));
   359→   359→            d2 = std::max(d2, std::abs((double)ob[i] - ob2[i]));
   360→   360→        }
   361→   361→        addValue(c, "fresh_vs_fresh", d1);
   362→   362→        addValue(c, "reset_vs_fresh", d2);
   363→   363→        check(c, d1 == 0.0, "two fresh engines differ");
   364→   364→        check(c, d2 == 0.0, "reset engine differs from fresh");
   365→   365→        c.pass = c.note.empty();
   366→   366→        R.push_back(std::move(c));
   367→   367→    }
   368→   368→    {  // 4. repeated prepare
   369→   369→        CaseResult c{"repeated_prepare"};
   370→   370→        Config cfg;
   371→   371→        cfg.ratio = semitoneRatio(500.0);
   372→   372→        const auto in = gen::tone(330.0, 48000, 24000, 0.4);
   373→   373→        AlignedDelayLine a, b;
   374→   374→        a.prepare(cfg);
   375→   375→        std::vector<float> oa(in.size(), 0.f), ob(in.size(), 0.f);
   376→   376→        a.process(in.data(), oa.data(), in.size());
   377→   377→        a.prepare(cfg);  // re-prepare
   378→   378→        a.process(in.data(), ob.data(), in.size());
   379→   379→        b.prepare(cfg);
   380→   380→        std::vector<float> oc(in.size(), 0.f);
   381→   381→        b.process(in.data(), oc.data(), in.size());
   382→   382→        double d = 0.0;
   383→   383→        for (size_t i = 0; i < in.size(); ++i)
   384→   384→            d = std::max(d, std::abs((double)ob[i] - oc[i]));
   385→   385→        addValue(c, "reprepared_vs_fresh", d);
   386→   386→        check(c, d == 0.0, "re-prepared engine differs from fresh");
   387→   387→        c.pass = c.note.empty();
   388→   388→        R.push_back(std::move(c));
   389→   389→    }
   390→   390→    {  // 5. block-size invariance
   391→   391→        CaseResult c{"block_size_invariance"};
   392→   392→        Config cfg;
   393→   393→        cfg.ratio = semitoneRatio(700.0);
   394→   394→        const auto in = gen::tone(196.0, 48000, 48000, 0.4);
   395→   395→        const auto ref = run(cfg, in, 1 << 20);
   396→   396→        const int splits[] = {1, 7, 16, 64, 128, 512, 1024, 4096};
   397→   397→        double worst = 0.0;
   398→   398→        for (int s : splits) {
   399→   399→            const auto got = run(cfg, in, s);
   400→   400→            double d = 0.0;
   401→   401→            for (size_t i = 0; i < in.size(); ++i)
   402→   402→                d = std::max(d, std::abs((double)ref[i] - got[i]));
   403→   403→            worst = std::max(worst, d);
   404→   404→        }
   405→   405→        // mixed split pattern
   406→   406→        {
   407→   407→            AlignedDelayLine e;
   408→   408→            e.prepare(cfg);
   409→   409→            std::vector<float> got(in.size(), 0.f);
   410→   410→            size_t i = 0;
   411→   411→            int pat[] = {1, 64, 7, 512, 16, 4096, 3};
   412→   412→            int pi = 0;
   413→   413→            while (i < in.size()) {
   414→   414→                int n = std::min(pat[pi % 7], (int)(in.size() - i));
   415→   415→                e.process(in.data() + i, got.data() + i, n);
   416→   416→                i += n;
   417→   417→                ++pi;
   418→   418→            }
   419→   419→            double d = 0.0;
   420→   420→            for (size_t k = 0; k < in.size(); ++k)
   421→   421→                d = std::max(d, std::abs((double)ref[k] - got[k]));
   422→   422→            worst = std::max(worst, d);
   423→   423→        }
   424→   424→        addValue(c, "worst_split_diff", worst);

   496→   496→            }
