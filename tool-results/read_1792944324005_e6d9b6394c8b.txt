    41→/// 64-bit LCG, identical on the Python side. u in [0,1); sample = (2u-1)*amp.
    42→struct Lcg {
    43→    uint64_t s;
    44→    explicit Lcg(uint64_t seed) : s(seed * 6364136223846793005ULL +
    45→                                     1442695040888963407ULL) {}
    46→    double next() {
    47→        s = s * 6364136223846793005ULL + 1442695040888963407ULL;
    48→        const uint64_t m = (s >> 11) & ((1ULL << 53) - 1);
    49→        return static_cast<double>(m) / 9007199254740992.0;
    50→    }
    51→    double noise(double amp) { return (2.0 * next() - 1.0) * amp; }
    52→};

    54→std::vector<float> tone(double f, int sr, int n, double amp) {
    55→    std::vector<float> x(n);
    56→    for (int i = 0; i < n; ++i)
    57→        x[i] = static_cast<float>(amp * std::sin(2.0 * M_PI * f * i / sr));
    58→    return x;
    59→}
    60→
    61→std::vector<float> chord(const std::vector<double>& fs, int sr, int n,
    62→                         double amp) {
    63→    std::vector<float> x(n, 0.0f);
    64→    for (double f : fs)
    65→        for (int i = 0; i < n; ++i)
    66→            x[i] += static_cast<float>(amp / fs.size() *
    67→                                       std::sin(2.0 * M_PI * f * i / sr));
    68→    return x;
    69→}

    71→std::vector<float> noise(uint64_t seed, int n, double amp) {
    72→    Lcg rng(seed);
    73→    std::vector<float> x(n);
    74→    for (int i = 0; i < n; ++i) x[i] = static_cast<float>(rng.noise(amp));
    75→    return x;
    76→}

    78→std::vector<float> impulse(int n) {
    79→    std::vector<float> x(n, 0.0f);
    80→    x[0] = 1.0f;
    81→    return x;
    82→}

    84→/// Percussive burst train: exponential decays.
    85→std::vector<float> bursts(int sr, int n, double period_s, double amp) {
    86→    std::vector<float> x(n, 0.0f);
    87→    Lcg rng(99);
    88→    const int per = static_cast<int>(period_s * sr);
    89→    for (int i = 0; i < n; ++i) {
    90→        const int t = i % per;
    91→        if (t < 200)
    92→            x[i] = static_cast<float>(amp * std::exp(-t / 40.0) *
    93→                                      (t == 0 ? 1.0 : rng.noise(1.0)));
    94→    }
    95→    return x;
    96→}
    97→
    98→std::vector<float> chirp(double f0, double f1, int sr, int n, double amp) {
    99→    std::vector<float> x(n);
   100→    const double T = static_cast<double>(n) / sr;
   101→    for (int i = 0; i < n; ++i) {
   102→        const double t = static_cast<double>(i) / sr;
   103→        const double f = f0 * std::pow(f1 / f0, t / T);
   104→        x[i] = static_cast<float>(amp * std::sin(2.0 * M_PI * f * t));
   105→    }
   106→    return x;
   107→}

   116→/// Radix-2 complex FFT (in-place, n a power of two).
   117→void fft(double* re, double* im, int n) {
   118→    for (int i = 1, j = 0; i < n; ++i) {
   119→        int bit = n >> 1;
   120→        for (; j & bit; bit >>= 1) j ^= bit;
   121→        j ^= bit;
   122→        if (i < j) {
   123→            std::swap(re[i], re[j]);
   124→            std::swap(im[i], im[j]);
   125→        }
   126→    }
   127→    for (int len = 2; len <= n; len <<= 1) {
   128→        const double ang = -2.0 * M_PI / len;
   129→        const double wr = std::cos(ang), wi = std::sin(ang);
   130→        for (int i = 0; i < n; i += len) {
   131→            double cr = 1.0, ci = 0.0;
   132→            for (int k = 0; k < len / 2; ++k) {
   133→                const double ur = re[i + k], ui = im[i + k];
   134→                const double vr = re[i + k + len / 2] * cr -
   135→                                  im[i + k + len / 2] * ci;
   136→                const double vi = re[i + k + len / 2] * ci +
   137→                                  im[i + k + len / 2] * cr;
   138→                re[i + k] = ur + vr;
   139→                im[i + k] = ui + vi;
   140→                re[i + k + len / 2] = ur - vr;
   141→                im[i + k + len / 2] = ui - vi;
   142→                const double nr = cr * wr - ci * wi;
   143→                ci = cr * wi + ci * wr;
   144→                cr = nr;
   145→            }
   146→        }
   147→    }
   148→}

   150→/// FFT peak pitch with parabolic interpolation (Hann window).
   151→/// Returns 0.0 when no peak above the threshold is found.
   152→double pitchFFT(const float* x, int n, int sr, double fmin, double fmax) {
   153→    int N = 1;
   154→    while (N < n && N < 65536) N <<= 1;
   155→    if (N < 16) return 0.0;
   156→    std::vector<double> re(N, 0.0), im(N, 0.0);
   157→    const int off = n - N;  // use the newest N samples
   158→    for (int i = 0; i < N; ++i) {
   159→        const double w = 0.5 - 0.5 * std::cos(2.0 * M_PI * i / N);
   160→        re[i] = static_cast<double>(x[off + i]) * w;
   161→    }
   162→    fft(re.data(), im.data(), N);
   163→    const int kmin = std::max(1, static_cast<int>(fmin * N / sr));
   164→    const int kmax = std::min(N / 2 - 2, static_cast<int>(fmax * N / sr));
   165→    if (kmax <= kmin) return 0.0;
   166→    int kpeak = kmin;
   167→    double best = 0.0;
   168→    for (int k = kmin; k <= kmax; ++k) {
   169→        const double m = re[k] * re[k] + im[k] * im[k];
   170→        if (m > best) {
   171→            best = m;
   172→            kpeak = k;
   173→        }
   174→    }
   175→    if (best <= 0.0) return 0.0;
   176→    const double a = re[kpeak - 1] * re[kpeak - 1] + im[kpeak - 1] * im[kpeak - 1];
   177→    const double b = best;
   178→    const double c = re[kpeak + 1] * re[kpeak + 1] + im[kpeak + 1] * im[kpeak + 1];
   179→    double d = 0.0;
   180→    const double den = a - 2.0 * b + c;
   181→    if (std::abs(den) > 1e-300) d = 0.5 * (a - c) / den;
   182→    if (d > 0.5) d = 0.5;
   183→    if (d < -0.5) d = -0.5;
   184→    return static_cast<double>(sr) * (kpeak + d) / N;
   185→}
   186→
   187→double rms(const float* x, int n) {
   188→    double s = 0.0;
   189→    for (int i = 0; i < n; ++i) s += static_cast<double>(x[i]) * x[i];
   190→    return n ? std::sqrt(s / n) : 0.0;
   191→}
   192→
   193→double peak(const float* x, int n) {
   194→    double p = 0.0;
   195→    for (int i = 0; i < n; ++i) p = std::max(p, std::abs(static_cast<double>(x[i])));
   196→    return p;
   197→}
   198→
   199→bool allFinite(const float* x, int n) {
   200→    for (int i = 0; i < n; ++i)
   201→        if (!std::isfinite(x[i])) return false;
   202→    return true;
   203→}

   205→/// Envelope modulation in dB: max/min of RMS windows of `w` samples over the
   206→/// steady region (skip the first skip samples). w = sr/1000 gives the 1 ms
   207→/// metric; w = period gives the period-synchronous metric (the r11 lesson:
   208→/// a 1 ms window phase-breathes several dB on a plain tone).
   209→double envModulationDb(const float* x, int n, int sr, int skip, int w) {
   210→    std::vector<double> env;
   211→    for (int i0 = skip; i0 + w <= n; i0 += w) {
   212→        env.push_back(rms(x + i0, w));
   213→    }
   214→    if (env.empty()) return 0.0;
   215→    double lo = 1e300, hi = 0.0;
   216→    for (double v : env) {
   217→        if (v < lo) lo = v;
   218→        if (v > hi) hi = v;
   219→    }
   220→    if (lo <= 0.0) return hi > 0.0 ? 999.0 : 0.0;
   221→    return 20.0 * std::log10(hi / lo);
   222→}
   223→
   224→double centsBetween(double fMeasured, double fNominal) {
   225→    if (fMeasured <= 0.0 || fNominal <= 0.0) return 999.0;
   226→    return 1200.0 * std::log2(fMeasured / fNominal);
   227→}
   228→
   229→/// Max absolute sample-to-sample difference (transition click meter).
   230→double maxStep(const float* x, int n, int i0 = 0) {
   231→    double m = 0.0;
   232→    for (int i = std::max(1, i0 + 1); i < n; ++i)
   233→        m = std::max(m, std::abs(static_cast<double>(x[i]) - x[i - 1]));
   234→    return m;
   235→}

   260→double semitoneRatio(double cents) { return std::pow(2.0, cents / 1200.0); }

   262→// Runs the engine over a full signal in one go (64-aligned internally is
   263→// irrelevant -- process takes the whole block).
   264→std::vector<float> run(const Config& cfg, const std::vector<float>& in,
   265→                       int block = 1 << 20) {
   266→    AlignedDelayLine e;
   267→    if (!e.prepare(cfg)) {
   268→        return {};
   269→    }
   270→    std::vector<float> out(in.size(), 0.0f);
   271→    int done = 0;
   272→    while (done < static_cast<int>(in.size())) {
   273→        const int n = std::min(block, static_cast<int>(in.size()) - done);
   274→        e.process(in.data() + done, out.data() + done, n);
   275→        done += n;
   276→    }
   277→    return out;
   278→}

   283→void casesPrepareReset(std::vector<CaseResult>& R) {
   284→    {  // 1. default prepare
   285→        CaseResult c{"prepare_default"};
   286→        Config cfg;
   287→        AlignedDelayLine e;
   288→        const bool ok = e.prepare(cfg);
   289→        check(c, ok, "prepare failed");
   290→        check(c, e.latencySamples() == 2304, "latency != 2304");
   291→        check(c, e.ready(), "not ready");
   292→        c.pass = c.note.empty();
   293→        R.push_back(std::move(c));
   294→    }
   295→    {  // 2. invalid configs rejected
   296→        CaseResult c{"prepare_invalid_rejected"};
   297→        int rejected = 0, total = 0;
   298→        Config bad;
   299→        auto tryCfg = [&](const Config& cfg) {
   300→            ++total;
   301→            AlignedDelayLine e;
   302→            if (!e.prepare(cfg)) ++rejected;
   303→        };
   304→        bad = Config();
   305→        bad.sample_rate = 0.0;
   306→        tryCfg(bad);
   307→        bad = Config();
   308→        bad.voices = 9;
   309→        tryCfg(bad);
   310→        bad = Config();
   311→        bad.grain = 0.0;
   312→        tryCfg(bad);
   313→        bad = Config();
   314→        bad.feedback = 1.0;
   315→        tryCfg(bad);
   316→        bad = Config();
   317→        bad.base_delay = 0.0;
   318→        tryCfg(bad);
   319→        addValue(c, "rejected", rejected);
   320→        addValue(c, "total", total);
   321→        check(c, rejected == total, "some invalid configs accepted");
   322→        c.pass = c.note.empty();
   323→        R.push_back(std::move(c));
   324→    }
   325→    {  // 3. reset determinism
   326→        CaseResult c{"reset_determinism"};
   327→        Config cfg;
   328→        cfg.ratio = semitoneRatio(400.0);
   329→        const auto in = gen::tone(220.0, 48000, 48000, 0.5);
   330→        AlignedDelayLine a, b;
   331→        a.prepare(cfg);
   332→        b.prepare(cfg);
   333→        std::vector<float> oa(in.size(), 0.f), ob(in.size(), 0.f),
   334→            ob2(in.size(), 0.f);
   335→        a.process(in.data(), oa.data(), in.size());
   336→        b.process(in.data(), ob.data(), in.size());
   337→        b.reset();
   338→        b.process(in.data(), ob2.data(), in.size());
   339→        double d1 = 0.0, d2 = 0.0;
   340→        for (size_t i = 0; i < in.size(); ++i) {
   341→            d1 = std::max(d1, std::abs((double)oa[i] - ob[i]));
   342→            d2 = std::max(d2, std::abs((double)ob[i] - ob2[i]));
   343→        }
   344→        addValue(c, "fresh_vs_fresh", d1);
   345→        addValue(c, "reset_vs_fresh", d2);
   346→        check(c, d1 == 0.0, "two fresh engines differ");
   347→        check(c, d2 == 0.0, "reset engine differs from fresh");
   348→        c.pass = c.note.empty();
   349→        R.push_back(std::move(c));
   350→    }
   351→    {  // 4. repeated prepare
   352→        CaseResult c{"repeated_prepare"};
   353→        Config cfg;
   354→        cfg.ratio = semitoneRatio(500.0);
   355→        const auto in = gen::tone(330.0, 48000, 24000, 0.4);
   356→        AlignedDelayLine a, b;
   357→        a.prepare(cfg);
   358→        std::vector<float> oa(in.size(), 0.f), ob(in.size(), 0.f);
   359→        a.process(in.data(), oa.data(), in.size());
   360→        a.prepare(cfg);  // re-prepare
   361→        a.process(in.data(), ob.data(), in.size());
   362→        b.prepare(cfg);
   363→        std::vector<float> oc(in.size(), 0.f);
   364→        b.process(in.data(), oc.data(), in.size());
   365→        double d = 0.0;
   366→        for (size_t i = 0; i < in.size(); ++i)
   367→            d = std::max(d, std::abs((double)ob[i] - oc[i]));
   368→        addValue(c, "reprepared_vs_fresh", d);
   369→        check(c, d == 0.0, "re-prepared engine differs from fresh");
   370→        c.pass = c.note.empty();
   371→        R.push_back(std::move(c));
   372→    }
   373→    {  // 5. block-size invariance
   374→        CaseResult c{"block_size_invariance"};
   375→        Config cfg;
   376→        cfg.ratio = semitoneRatio(700.0);
   377→        const auto in = gen::tone(196.0, 48000, 48000, 0.4);
   378→        const auto ref = run(cfg, in, 1 << 20);
   379→        const int splits[] = {1, 7, 16, 64, 128, 512, 1024, 4096};
   380→        double worst = 0.0;
   381→        for (int s : splits) {
   382→            const auto got = run(cfg, in, s);
   383→            double d = 0.0;
   384→            for (size_t i = 0; i < in.size(); ++i)
   385→                d = std::max(d, std::abs((double)ref[i] - got[i]));
   386→            worst = std::max(worst, d);
   387→        }
   388→        // mixed split pattern
   389→        {
   390→            AlignedDelayLine e;
   391→            e.prepare(cfg);
   392→            std::vector<float> got(in.size(), 0.f);
   393→            size_t i = 0;
   394→            int pat[] = {1, 64, 7, 512, 16, 4096, 3};
   395→            int pi = 0;
   396→            while (i < in.size()) {
   397→                int n = std::min(pat[pi % 7], (int)(in.size() - i));
   398→                e.process(in.data() + i, got.data() + i, n);
   399→                i += n;
   400→                ++pi;
   401→            }
   402→            double d = 0.0;
   403→            for (size_t k = 0; k < in.size(); ++k)
   404→                d = std::max(d, std::abs((double)ref[k] - got[k]));
   405→            worst = std::max(worst, d);
   406→        }
   407→        addValue(c, "worst_split_diff", worst);
   408→        check(c, worst == 0.0, "block split changed the output");
   409→        c.pass = c.note.empty();
   410→        R.push_back(std::move(c));
   411→    }
   412→    {  // 6. steady-state allocation-free
   413→        CaseResult c{"steady_state_no_allocation"};
   414→        Config cfg;
   415→        cfg.ratio = semitoneRatio(400.0);
   416→        cfg.voices = 8;
   417→        cfg.feedback = 0.35;
   418→        const auto in = gen::noise(1234, 48000, 0.5);
   419→        AlignedDelayLine e;
   420→        e.prepare(cfg);
   421→        std::vector<float> out(in.size(), 0.f);
   422→        e.process(in.data(), out.data(), 24000);  // warm-up (tables, paths)
   423→        allocs::reset();
   424→        allocs::arm(true);
   425→        e.process(in.data() + 24000, out.data() + 24000,
   426→                  static_cast<int>(in.size()) - 24000);
   427→        allocs::arm(false);
   428→        const long n = allocs::count();
   429→        addValue(c, "allocations_during_process", static_cast<double>(n));
   430→        check(c, n == 0, "allocations during steady-state process()");
   431→        c.pass = c.note.empty();
   432→        R.push_back(std::move(c));
   433→    }

   436→void casesUnity(std::vector<CaseResult>& R) {
   437→    const int sr = 48000;
   438→
   439→    // bit-exact delayed-copy gate used by the unity battery
   440→    auto unityExact = [&](const std::string& name, const std::vector<float>& in,
   441→                          bool arrive_from_shift = false) {
   442→        CaseResult c{name};
   443→        Config cfg;  // ratio exactly 1.0, 1 voice, neutral
   444→        AlignedDelayLine e;
   445→        e.prepare(cfg);
   446→        if (arrive_from_shift) {
   447→            // arrive from +4 st first: 1 s shifted, then unity, then settle
   448→            e.setRatio(semitoneRatio(400.0));
   449→            std::vector<float> pre(sr, 0.f);
   450→            e.process(in.data(), pre.data(), sr);
   451→            e.setRatio(1.0);
   452→            std::vector<float> mid(static_cast<int>(0.3 * sr), 0.f);
   453→            if ((int)in.size() > sr)
   454→                e.process(in.data() + sr, mid.data(),
   455→                           std::min((int)in.size() - sr, (int)mid.size()));
   456→        }
   457→        const int d_park = e.voiceStats(0).parked_delay;
   458→        addValue(c, "d_park", d_park);
   459→        check(c, d_park >= 256 && d_park < 256 + 4096, "d_park out of range");
   460→        check(c, e.voiceStats(0).parked == 1, "voice not parked");
   461→        const int skip = arrive_from_shift ? static_cast<int>(0.05 * sr) : 0;
   462→        std::vector<float> out(in.size(), 0.f);
   463→        e.process(in.data(), out.data(), in.size());
   464→        // Reference ring semantics: a read at delay d returns the sample
   465→        // pushed d-1 samples earlier, so the parked copy is in[k-d_park+1].
   466→        double worst = 0.0;
   467→        int worst_k = -1;
   468→        // an engine that arrived from a shift has legitimate ring history:
   469→        // bit-exactness is only decidable once the read reaches the current
   470→        // run's own input (k >= d_park-1)
   471→        const int k0 = std::max(skip, d_park - 1);
   472→        for (int k = k0; k < (int)in.size(); ++k) {
   473→            const int src = k - d_park + 1;
   474→            const double expect = (src >= 0) ? static_cast<double>(in[src]) : 0.0;
   475→            const double d = std::abs(static_cast<double>(out[k]) - expect);
   476→            if (d > worst) {
   477→                worst = d;
   478→                worst_k = k;
   479→            }
   480→        }
   481→        addValue(c, "max_abs_diff", worst);
   482→        addValue(c, "worst_k", worst_k);
   483→        check(c, worst == 0.0, "unity output is not the bit-exact delayed copy");
   484→        // the pre-delay region is exactly zero (fresh engines only -- an
   485→        // engine that arrived from a shift legitimately has ring history)
   486→        if (!arrive_from_shift) {
   487→            double pre_max = 0.0;
   488→            for (int k = 0; k < d_park - 1 && k < (int)in.size(); ++k)
   489→                pre_max = std::max(pre_max, std::abs((double)out[k]));
   490→            addValue(c, "pre_delay_region_max", pre_max);
   491→            check(c, pre_max == 0.0, "output before d_park not silent");
   492→        }
   493→        c.pass = c.note.empty();
   494→        R.push_back(std::move(c));
   495→    };
   496→
   497→    unityExact("unity_tone_220_bitexact", gen::tone(220.0, sr, 48000, 0.5));
   498→    unityExact("unity_tone_1100_bitexact", gen::tone(1100.0, sr, 48000, 0.5));
   499→    unityExact("unity_noise_bitexact", gen::noise(42, 48000, 0.8));
   500→    unityExact("unity_chord_bitexact",
   501→               gen::chord({196.0, 294.0, 392.0, 494.0}, sr, 48000, 0.8));

   508→               gen::tone(220.0, sr, 96000, 0.5), true);
   509→
   510→    {  // unity gain exactly 1 (follows from bit-exactness, measured anyway)
   511→        CaseResult c{"unity_gain_exact"};
   512→        const auto in = gen::tone(220.0, sr, 48000, 0.5);
   513→        Config cfg;
   514→        AlignedDelayLine e;
   515→        e.prepare(cfg);
   516→        std::vector<float> out(in.size(), 0.f);
   517→        e.process(in.data(), out.data(), in.size());
   518→        const int d = e.voiceStats(0).parked_delay - 1;
   519→        // out[d..] is an exact copy of in[0..n-d): compare matching segments
   520→        const double ri = meas::rms(in.data(), in.size() - d);
   521→        const double ro = meas::rms(out.data() + d, out.size() - d);
   522→        addValue(c, "gain_db", 20.0 * std::log10(ro / ri));

   525→        R.push_back(std::move(c));
   526→    }
   527→
   528→    {  // entry continuity (from +400 cents into unity): no click
   529→        CaseResult c{"unity_entry_continuity"};
   530→        Config cfg;
   531→        AlignedDelayLine e;
   532→        e.prepare(cfg);
   533→        e.setRatio(semitoneRatio(400.0));
   534→        const int preN = 2 * sr;
   535→        const int postN = 1 * sr;
   536→        const auto in = gen::tone(220.0, sr, preN + postN, 0.5);
   537→        std::vector<float> out(in.size(), 0.f);
   538→        e.process(in.data(), out.data(), preN);
   539→        e.setRatio(1.0);
   540→        e.process(in.data() + preN, out.data() + preN, postN);
   541→        // during the transition (first 60 ms after the change)
   542→        const int trans = static_cast<int>(0.06 * sr);
   543→        const double step = meas::maxStep(out.data() + preN, trans);
   544→        const double stepBefore = meas::maxStep(out.data() + preN - trans, trans);
   545→        addValue(c, "transition_max_step", step);
   546→        addValue(c, "pre_transition_max_step", stepBefore);
   547→        check(c, step <= std::max(0.35, 2.0 * stepBefore + 1e-9),
   548→              "transition click");
   549→        check(c, meas::allFinite(out.data(), out.size()), "non-finite");
   550→        c.pass = c.note.empty();
   551→        R.push_back(std::move(c));
   552→    }
   553→
   554→    // near-unity battery: fresh-start vs via-unity-exit
   555→    auto nearUnity = [&](const std::string& tag, double cents, bool viaUnity) {
   556→        CaseResult c{"near_unity_" + tag + "_" +
   557→                     (viaUnity ? "via_unity" : "fresh")};
   558→        const double ratio = semitoneRatio(cents);
   559→        Config cfg;
   560→        cfg.ratio = viaUnity ? 1.0 : ratio;
   561→        AlignedDelayLine e;
   562→        e.prepare(cfg);
   563→        const auto in = gen::tone(220.0, sr, 3 * sr, 0.5);
   564→        std::vector<float> out(in.size(), 0.f);
   565→        if (viaUnity) {
   566→            // 1 s at unity (parked), then depart
   567→            e.process(in.data(), out.data(), sr);
   568→            e.setRatio(ratio);
   569→            e.process(in.data() + sr, out.data() + sr, 2 * sr);
   570→        } else {
   571→            e.process(in.data(), out.data(), in.size());
   572→        }
   573→        const double f0nom = 220.0;
   574→        const double fm = meas::pitchFFT(out.data() + static_cast<int>(1.5 * sr),
   575→                                         static_cast<int>(1.5 * sr), sr, 50.0,
   576→                                         4000.0);
   577→        const double err = meas::centsBetween(fm, f0nom) - cents;
   578→        const double gainDb = 20.0 * std::log10(
   579→            meas::rms(out.data() + sr, 2 * sr) /
   580→            meas::rms(in.data() + sr, 2 * sr));
   581→        const int perW = std::max(1, static_cast<int>(std::lround(sr / 220.0)));
   582→        const double mod = meas::envModulationDb(out.data(), out.size(), sr,
   583→                                                 static_cast<int>(1.5 * sr), perW);
   584→        const double commits = static_cast<double>(e.voiceStats(0).commits);
   585→        addValue(c, "measured_f0", fm);
   586→        addValue(c, "pitch_error_cents", err);
   587→        addValue(c, "gain_db", gainDb);
   588→        addValue(c, "env_modulation_db", mod);
   589→        addValue(c, "spacing", e.voiceStats(0).spacing);
   590→        addValue(c, "commits", commits);
   591→        check(c, meas::allFinite(out.data(), out.size()), "non-finite");
   592→        check(c, std::abs(err) <= 2.0, "pitch error > 2 cents");
   593→        if (viaUnity) {
   594→            check(c, gainDb >= -0.75 && gainDb <= 0.75,
   595→                  "via-unity gain outside +-0.75 dB (comb should be aligned)");
   596→            check(c, mod <= 0.75, "via-unity modulation > 0.75 dB");
   597→        } else {
   598→            check(c, gainDb >= -4.0 && gainDb <= 2.0,
   599→                  "fresh gain outside [-4, +2] dB (static comb expected)");
   600→        }
   601→        c.pass = c.note.empty();
   602→        R.push_back(std::move(c));
   603→    };
   604→
   605→    const double centsList[] = {1.0,  -1.0, 10.0, -10.0, 50.0,  -50.0,
   606→                                100.0, -100.0, 500.0, -500.0, 1200.0, -1200.0};
   607→    for (double cents : centsList) {
   608→        char tag[32];
   609→        std::snprintf(tag, sizeof(tag), "%+gc", cents);
   610→        nearUnity(tag, cents, false);
   611→    }
   612→    // via-unity for the small shifts where the exit-commit matters most
   613→    nearUnity("+1c", 1.0, true);
   614→    nearUnity("-1c", -1.0, true);
   615→    nearUnity("+10c", 10.0, true);
   616→    nearUnity("-10c", -10.0, true);
   617→    nearUnity("+50c", 50.0, true);
   618→    nearUnity("-50c", -50.0, true);
   619→
   620→    {  // no-bypass: +-1 cent really shifts (measured change in pitch)
   621→        CaseResult c{"no_bypass_region"};
   622→        const auto in = gen::tone(220.0, sr, 3 * sr, 0.5);
   623→        Config cfg;
   624→        cfg.ratio = semitoneRatio(1.0);
   625→        const auto outP = run(cfg, in);
   626→        cfg.ratio = semitoneRatio(-1.0);
   627→        const auto outN = run(cfg, in);
   628→        const double fp = meas::pitchFFT(outP.data() + sr, 2 * sr, sr, 50.0, 4000.0);
   629→        const double fn = meas::pitchFFT(outN.data() + sr, 2 * sr, sr, 50.0, 4000.0);
   630→        const double cp = meas::centsBetween(fp, 220.0);
   631→        const double cn = meas::centsBetween(fn, 220.0);
   632→        addValue(c, "plus_1c_measured_cents", cp);
   633→        addValue(c, "minus_1c_measured_cents", cn);
   634→        check(c, cp > 0.5, "+1 cent did not shift (bypass?)");
   635→        check(c, cn < -0.5, "-1 cent did not shift (bypass?)");
   636→        check(c, cp - cn > 1.0, "+/-1 cent outputs indistinguishable");
   637→        c.pass = c.note.empty();
   638→        R.push_back(std::move(c));
   639→    }
   640→
   641→    {  // unity with feedback: bounded + finite (comb is expected behaviour)
   642→        CaseResult c{"unity_feedback_stability"};
   643→        Config cfg;
   644→        cfg.feedback = 0.35;
   645→        AlignedDelayLine e;
   646→        e.prepare(cfg);
   647→        const auto in = gen::noise(7, 5 * sr, 0.5);
   648→        std::vector<float> out(in.size(), 0.f);
   649→        e.process(in.data(), out.data(), in.size());
   650→        const double pk = meas::peak(out.data(), out.size());
   651→        addValue(c, "peak", pk);
   652→        check(c, meas::allFinite(out.data(), out.size()), "non-finite");
   653→        check(c, pk < 4.0, "unstable growth with feedback 0.35 at unity");
   654→        c.pass = c.note.empty();
   655→        R.push_back(std::move(c));
   656→    }
   657→
   658→    {  // 8 voices, spread off, all at unity: exact mean of the parked delays
   659→        CaseResult c{"unity_8voice_unison_exact_mix"};
   660→        Config cfg;
   661→        cfg.voices = 8;
   662→        cfg.spread = false;
   663→        AlignedDelayLine e;
   664→        e.prepare(cfg);
   665→        const auto in = gen::tone(220.0, sr, 48000, 0.5);
   666→        std::vector<float> out(in.size(), 0.f);
   667→        e.process(in.data(), out.data(), in.size());
   668→        int d[8];
   669→        for (int v = 0; v < 8; ++v) {
   670→            d[v] = e.voiceStats(v).parked_delay;
   671→            check(c, e.voiceStats(v).parked == 1, "voice not parked");
   672→        }
   673→        double worst = 0.0;
   674→        for (int k = 4800; k < (int)in.size(); ++k) {
   675→            double expect = 0.0;
   676→            for (int v = 0; v < 8; ++v)
   677→                expect += (k - d[v] + 1 >= 0) ? in[k - d[v] + 1] : 0.0f;
   678→            expect /= 8.0;
   679→            // the engine output is the float32 rounding of the double mean
   680→            const float expected_f = static_cast<float>(expect);
   681→            worst = std::max(worst, std::abs((double)out[k] - (double)expected_f));
   682→        }
   683→        addValue(c, "max_abs_diff_vs_exact_mean", worst);
   684→        check(c, worst == 0.0, "8-voice unison mix is not the exact mean");
   685→        c.pass = c.note.empty();
   686→        R.push_back(std::move(c));
   687→    }
   688→}

   802→void casesRatesFormats(std::vector<CaseResult>& R) {
   803→    {  // 44.1 kHz
   804→        CaseResult c{"rate_44100_+400c"};
   805→        Config cfg;
   806→        cfg.sample_rate = 44100.0;
   807→        cfg.ratio = semitoneRatio(400.0);
   808→        const int sr = 44100;
   809→        const auto in = gen::tone(220.0, sr, 3 * sr, 0.5);
   810→        const auto out = run(cfg, in);
   811→        const double f = meas::pitchFFT(out.data() + sr, 2 * sr, sr, 50.0, 4000.0);

   819→    {  // 96 kHz (detector window too short for 60 Hz: spacing falls back to
   820→       // G -- the Python reference has the same behaviour; gate is looser)
   821→        CaseResult c{"rate_96000_+400c"};
   822→        Config cfg;
   823→        cfg.sample_rate = 96000.0;
   824→        cfg.ratio = semitoneRatio(400.0);
   825→        const int sr = 96000;
   826→        const auto in = gen::tone(220.0, sr, 3 * sr, 0.5);
   827→        const auto out = run(cfg, in);
   828→        const double f = meas::pitchFFT(out.data() + sr, 2 * sr, sr, 50.0, 4000.0);
   829→        const double err = meas::centsBetween(f, 220.0) - 400.0;
   830→        addValue(c, "pitch_error_cents", err);
   831→        check(c, meas::allFinite(out.data(), out.size()), "non-finite");
   832→        check(c, std::abs(err) <= 25.0, "pitch error > 25 cents at 96 kHz");
   833→        c.pass = c.note.empty();
   834→        R.push_back(std::move(c));
   835→    }
   836→    {  // WAV round trip on the shipped example input (if present)
   837→        CaseResult c{"wav_roundtrip_example_input"};
   838→        const char* candidates[] = {"../poc/example_input.wav",
   839→                                    "../../poc/example_input.wav",
   840→                                    "../../../poc/example_input.wav"};
   841→        int sr = 0, ch = 0;
   842→        std::vector<float> x;
   843→        bool loaded = false;
   844→        for (const char* pocPath : candidates) {
   845→            try {
   846→                x = wav::readWav(pocPath, sr, ch);
   847→                loaded = true;
   848→                break;
   849→            } catch (const std::exception&) {
   850→            }
   851→        }
   852→        if (!loaded || x.empty()) {
   853→            c.note = "example_input.wav not found (running outside tree)";
   854→            c.pass = true;  // environment-dependent skip, reported honestly
   855→            addValue(c, "skipped", 1);
   856→        } else {
   857→            Config cfg;
   858→            cfg.ratio = semitoneRatio(400.0);
   859→            const auto out = run(cfg, x);
   860→            check(c, meas::allFinite(out.data(), out.size()), "non-finite");
   861→            wav::writeWav("/tmp/octavox_native_example_out.wav", out.data(),
   862→                          out.size(), ch, sr);
   863→            int sr2 = 0, ch2 = 0;
   864→            const auto back = wav::readWav("/tmp/octavox_native_example_out.wav",
   865→                                           sr2, ch2);
   866→            check(c, sr2 == sr && ch2 == ch, "round-trip format changed");
   867→            double worst = 0.0;
   868→            for (size_t i = 0; i < out.size(); ++i)
   869→                worst = std::max(worst, std::abs((double)out[i] - back[i]));
   870→            addValue(c, "quant_error_max", worst);
   871→            check(c, worst <= 2.0 / 32767.0,
   872→                  "16-bit round-trip error too large (write/read asymmetry is "
   873→                  "reference parity: round(y*32767) then /32768)");
   874→            c.pass = c.note.empty();
   875→        }
   876→        R.push_back(std::move(c));
   877→    }
   878→    {  // 24-bit WAV read
   879→        CaseResult c{"wav_24bit_read"};
   880→        // write a minimal 24-bit WAV by hand
   881→        const int sr = 48000, n = 256;
   882→        std::vector<int32_t> vals(n);
   883→        for (int i = 0; i < n; ++i) vals[i] = (i * 7919) % 4000000 - 2000000;
   884→        std::vector<unsigned char> bytes(44 + 3 * n);
   885→        std::memcpy(bytes.data(), "RIFF", 4);
   886→        const uint32_t riffSz = 36 + 3 * n;
   887→        std::memcpy(bytes.data() + 4, &riffSz, 4);
   888→        std::memcpy(bytes.data() + 8, "WAVE", 4);
   889→        std::memcpy(bytes.data() + 12, "fmt ", 4);
   890→        const uint32_t fsz = 16;
   891→        std::memcpy(bytes.data() + 16, &fsz, 4);
   892→        const uint16_t af = 1, ch = 1, bits = 24;
   893→        const uint32_t rate = sr, br = sr * 3;
   894→        const uint16_t block = 3;
   895→        std::memcpy(bytes.data() + 20, &af, 2);
   896→        std::memcpy(bytes.data() + 22, &ch, 2);
   897→        std::memcpy(bytes.data() + 24, &rate, 4);
   898→        std::memcpy(bytes.data() + 28, &br, 4);
   899→        std::memcpy(bytes.data() + 32, &block, 2);
   900→        std::memcpy(bytes.data() + 34, &bits, 2);
   901→        std::memcpy(bytes.data() + 36, "data", 4);
   902→        const uint32_t dsz = 3 * n;
   903→        std::memcpy(bytes.data() + 40, &dsz, 4);
   904→        for (int i = 0; i < n; ++i) {
   905→            bytes[44 + 3 * i] = vals[i] & 0xFF;
   906→            bytes[45 + 3 * i] = (vals[i] >> 8) & 0xFF;
   907→            bytes[46 + 3 * i] = (vals[i] >> 16) & 0xFF;
   908→        }
   909→        FILE* f = std::fopen("/tmp/octavox_24bit.wav", "wb");
   910→        std::fwrite(bytes.data(), 1, bytes.size(), f);
   911→        std::fclose(f);
   912→        int sr2 = 0, ch2 = 0;
   913→        auto x = wav::readWav("/tmp/octavox_24bit.wav", sr2, ch2);
   914→        check(c, sr2 == sr && ch2 == 1 && (int)x.size() == n, "bad header");
   915→        double worst = 0.0;
   916→        for (int i = 0; i < n && i < (int)x.size(); ++i)
   917→            worst = std::max(worst,
   918→                             std::abs((double)x[i] - vals[i] / 8388608.0));
   919→        addValue(c, "max_read_error", worst);
   920→        check(c, worst == 0.0, "24-bit values wrong");
   921→        c.pass = c.note.empty();
   922→        R.push_back(std::move(c));
   923→    }
   924→}

   926→void casesStability(std::vector<CaseResult>& R) {
   927→    const int sr = 48000;
   928→    {  // long run: 60 s noise, 8 voices, feedback
   929→        CaseResult c{"longrun_60s_8voice_feedback"};
   930→        Config cfg;
   931→        cfg.ratio = semitoneRatio(1200.0);
   932→        cfg.voices = 8;
   933→        cfg.feedback = 0.35;
   934→        const auto in = gen::noise(2025, 60 * sr, 0.5);
   935→        const auto out = run(cfg, in);
   936→        check(c, meas::allFinite(out.data(), out.size()), "non-finite");
   937→        const double pk = meas::peak(out.data(), out.size());
   938→        addValue(c, "peak", pk);
   939→        check(c, pk < 4.0, "unstable growth");
   940→        // tail RMS must not exceed head RMS by much (no divergence)
   941→        const double rh = meas::rms(out.data(), 5 * sr);
   942→        const double rt = meas::rms(out.data() + 55 * sr, 5 * sr);
   943→        addValue(c, "head_rms", rh);
   944→        addValue(c, "tail_rms", rt);
   945→        check(c, rt < 4.0 * rh + 1e-9, "diverging tail");
   946→        c.pass = c.note.empty();
   947→        R.push_back(std::move(c));
   948→    }
   949→    {  // 30 s chirp +700c
   950→        CaseResult c{"chirp_30s_+700c"};
   951→        Config cfg;
   952→        cfg.ratio = semitoneRatio(700.0);
   953→        const auto in = gen::chirp(100.0, 1000.0, sr, 30 * sr, 0.5);
   954→        const auto out = run(cfg, in);
   955→        check(c, meas::allFinite(out.data(), out.size()), "non-finite");
   956→        const double pk = meas::peak(out.data(), out.size());
   957→        addValue(c, "peak", pk);
   958→        check(c, pk < 3.0, "peak too high on chirp");
   959→        c.pass = c.note.empty();
   960→        R.push_back(std::move(c));
   961→    }
   962→    {  // impulse latency
   963→        CaseResult c{"impulse_latency"};
   964→        Config cfg;
   965→        cfg.ratio = semitoneRatio(400.0);
   966→        const auto in = gen::impulse(sr);
   967→        const auto out = run(cfg, in);
   968→        int first = -1;
   969→        for (int k = 0; k < (int)out.size(); ++k)
   970→            if (out[k] != 0.0f) {
   971→                first = k;
   972→                break;
   973→            }
   974→        addValue(c, "first_nonzero_sample", first);
   975→        addValue(c, "nominal_latency", cfg.base_delay + cfg.grain * 0.5);
   976→        check(c, first >= 250, "responds before base_delay");
   977→        check(c, first <= 256 + 32 + 4400, "responds too late");
   978→        c.pass = c.note.empty();
   979→        R.push_back(std::move(c));
   980→    }
   981→    {  // unity impulse: first nonzero exactly at d_park
   982→        CaseResult c{"impulse_latency_unity"};
   983→        Config cfg;
   984→        const auto in = gen::impulse(sr);
   985→        const auto out = run(cfg, in);
   986→        AlignedDelayLine e;
   987→        e.prepare(cfg);
   988→        const int d = e.voiceStats(0).parked_delay;
   989→        int first = -1;
   990→        for (int k = 0; k < (int)out.size(); ++k)
   991→            if (out[k] != 0.0f) {
   992→                first = k;
   993→                break;
   994→            }
   995→        addValue(c, "first_nonzero_sample", first);
   996→        addValue(c, "d_park", d);
   997→        check(c, first == d - 1, "unity impulse not at d_park-1");
   998→        c.pass = c.note.empty();
   999→        R.push_back(std::move(c));
  1000→    }
  1001→    {  // DC input: finite (guards div-by-zero in spacing/median paths)
  1002→        CaseResult c{"dc_input_finite"};
  1003→        Config cfg;
  1004→        cfg.ratio = semitoneRatio(400.0);
  1005→        std::vector<float> in(10 * sr, 1.0f);
  1006→        const auto out = run(cfg, in);
  1007→        check(c, meas::allFinite(out.data(), out.size()), "non-finite on DC");
  1008→        c.pass = c.note.empty();
  1009→        R.push_back(std::move(c));
  1010→    }
  1011→    {  // clipping behaviour: hot input stays finite; WAV writer clips
  1012→        CaseResult c{"clipping_behaviour"};
  1013→        Config cfg;
  1014→        cfg.ratio = semitoneRatio(1200.0);
  1015→        cfg.voices = 8;
  1016→        std::vector<float> in(sr, 0.0f);
  1017→        for (int i = 0; i < sr; ++i)
  1018→            in[i] = (i % 480 < 240) ? 1.0f : -1.0f;  // full-scale square
  1019→        const auto out = run(cfg, in);
  1020→        const double pk = meas::peak(out.data(), out.size());
  1021→        addValue(c, "float_domain_peak", pk);
  1022→        check(c, meas::allFinite(out.data(), out.size()), "non-finite");
  1023→        check(c, pk < 8.0, "float peak beyond sane bound");
  1024→        wav::writeWav("/tmp/octavox_clip.wav", out.data(), out.size(), 1, sr);
  1025→        int sr2 = 0, ch2 = 0;
  1026→        auto back = wav::readWav("/tmp/octavox_clip.wav", sr2, ch2);
  1027→        const double pkq = meas::peak(back.data(), back.size());
  1028→        addValue(c, "wav_peak_after_clip", pkq);
  1029→        check(c, pkq <= 1.0, "WAV writer did not clip-guard");
  1030→        c.pass = c.note.empty();
  1031→        R.push_back(std::move(c));
  1032→    }
  1033→}

  1038→std::string jsonEscape(const std::string& s) {
  1039→    std::string o;
  1040→    for (char ch : s) {
  1041→        if (ch == '"' || ch == '\\') {
  1042→            o += '\\';
  1043→            o += ch;
  1044→        } else if (ch == '\n') {
  1045→            o += "\\n";
  1046→        } else {
  1047→            o += ch;
  1048→        }
  1049→    }
  1050→    return o;
  1051→}

  1053→void writeJson(const std::string& path) {
  1054→    std::string j = "{\n  \"suite\": \"octavox_native_tests\",\n  \"version\": \"";
  1055→    j += versionString();
  1056→    j += "\",\n  \"cases\": [\n";
  1057→    int idx = 0;
  1058→    for (const auto& c : g_results) {
  1059→        j += "    {\"name\": \"" + jsonEscape(c.name) + "\", \"pass\": " +
  1060→             (c.pass ? "true" : "false") + ", \"note\": \"" +
  1061→             jsonEscape(c.note) + "\", \"values\": {";
  1062→        for (size_t i = 0; i < c.values.size(); ++i) {
  1063→            char buf[64];
  1064→            std::snprintf(buf, sizeof(buf), "%.10g", c.values[i].second);
  1065→            j += "\"" + jsonEscape(c.values[i].first) + "\": " + buf;
  1066→            if (i + 1 < c.values.size()) j += ", ";
  1067→        }
  1068→        j += "}}";
  1069→        if (++idx < (int)g_results.size()) j += ",";
  1070→        j += "\n";
  1071→    }
  1072→    int passed = 0;
  1073→    for (const auto& c : g_results) passed += c.pass ? 1 : 0;
  1074→    j += "  ],\n  \"summary\": {\"total\": " + std::to_string(g_results.size()) +
  1075→         ", \"passed\": " + std::to_string(passed) + ", \"failed\": " +
  1076→         std::to_string(g_results.size() - passed) + "}\n}\n";
  1077→    FILE* f = std::fopen(path.c_str(), "wb");
  1078→    if (f) {
  1079→        std::fwrite(j.data(), 1, j.size(), f);
  1080→        std::fclose(f);
  1081→    }
  1082→}
