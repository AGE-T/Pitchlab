    10→// BUNDLE through the pinned Steinberg SDK's module loader

    21→// R18-WDB (Windows dual build): the harness is VARIANT-AWARE.  The bundle

    24→// (octavox_build_config.h): an OBSKERNEL6 module is compared against an
    25→// OBSKERNEL6 reference -- NEVER against a LAGRANGE6 reference.  The module
    26→// path can be given explicitly (--module <path>); the default candidates
    27→// use this build's bundle name (OctavoxRE_<Variant>.vst3).

    31→// NOT a DAW compatibility claim: no DAW was involved, and passing here does
    32→// not certify behaviour in any specific DAW.  The SDK validator run remains

    73→std::string jsonEscape(const std::string& s) {
    74→    std::string o;
    75→    for (char ch : s) {
    76→        if (ch == '"' || ch == '\\') { o += '\\'; o += ch; }
    77→        else if ((unsigned char)ch < 0x20) {
    78→            char b[8];
    79→            std::snprintf(b, sizeof b, "\\u%04x", (unsigned)ch);
    80→            o += b;
    81→        } else o += ch;
    82→    }
    83→    return o;
    84→}

   100→void pushCase(std::vector<CaseResult>& R, CaseResult&& c) {
   101→    if (!c.pass) ++g_failures;
   102→    std::printf("[%-34s] %-4s %s\n", c.name.c_str(), c.pass ? "ok" : "FAIL",
   103→                c.note.c_str());
   104→    for (const auto& v : c.values)
   105→        std::printf("%-38s  %s = %.10g\n", "", v.first.c_str(), v.second);
   106→    std::fflush(stdout);
   107→    R.push_back(std::move(c));
   108→}

   110→// deterministic LCG (same spec as the core's test harness)
   111→struct Lcg {
   112→    uint64_t s;
   113→    explicit Lcg(uint64_t seed) : s(seed ? seed : 1) {}
   114→    double next() {
   115→        s = s * 6364136223846793005ULL + 1442695040888963407ULL;
   116→        return (static_cast<double>(s >> 11) / 9007199254740992.0) * 0.5 - 0.25;
   117→    }
   118→    void fill(std::vector<float>& v) {
   119→        for (auto& x : v) x = static_cast<float>(next());
   120→    }
   121→};
   122→
   123→double maxAbsDiff(const std::vector<float>& a, const std::vector<float>& b) {
   124→    if (a.size() != b.size()) return 1e30;
   125→    double d = 0.0;
   126→    for (size_t i = 0; i < a.size(); ++i) {
   127→        const double x = std::fabs(static_cast<double>(a[i]) - static_cast<double>(b[i]));
   128→        if (x > d) d = x;
   129→    }
   130→    return d;
   131→}
   132→
   133→bool bitIdentical(const float* a, const float* b, size_t n) {
   134→    return std::memcmp(a, b, n * sizeof(float)) == 0;
   135→}

   151→struct Setup {
   152→    double sampleRate = 48000.0;
   153→    int32 maxBlock = 4096;
   154→};
   155→
   156→// full standard lifecycle on the loaded instance; returns false on failure
   157→bool setupLoadedInstance(const Setup& st, int channels) {
   158→    if (!g_audio) return false;
   159→    SpeakerArrangement in = channels == 1 ? SpeakerArr::kMono : SpeakerArr::kStereo;
   160→    SpeakerArrangement out = in;
   161→    if (g_audio->setBusArrangements(&in, 1, &out, 1) != kResultOk) return false;
   162→    ProcessSetup ps;
   163→    std::memset(&ps, 0, sizeof(ps));
   164→    ps.processMode = kRealtime;
   165→    ps.symbolicSampleSize = kSample32;
   166→    ps.sampleRate = st.sampleRate;
   167→    ps.maxSamplesPerBlock = st.maxBlock;
   168→    if (g_audio->setupProcessing(ps) != kResultOk) return false;
   169→    if (g_component->setActive(true) != kResultOk) return false;
   170→    return g_audio->setProcessing(true) == kResultOk;
   171→}
   172→
   173→void teardownLoadedInstance() {
   174→    if (g_audio) g_audio->setProcessing(false);
   175→    if (g_component) g_component->setActive(false);
   176→}

   178→// one host process() call through the loaded module
   179→void driveLoaded(float* const* in, float* const* out, int channels, int n,
   180→                 IParameterChanges* changes) {
   181→    AudioBusBuffers ib{}, ob{};
   182→    ib.numChannels = channels;
   183→    ib.channelBuffers32 = const_cast<float**>(in);
   184→    ob.numChannels = channels;
   185→    ob.channelBuffers32 = const_cast<float**>(out);
   186→    AudioBusBuffers inB[1] = {ib}, outB[1] = {ob};
   187→    ProcessData pd{};
   188→    pd.processMode = kRealtime;
   189→    pd.symbolicSampleSize = kSample32;
   190→    pd.numSamples = n;
   191→    pd.inputs = inB;
   192→    pd.numInputs = 1;
   193→    pd.outputs = outB;
   194→    pd.numOutputs = 1;
   195→    pd.inputParameterChanges = changes;
   196→    g_audio->process(pd);
   197→}

   199→void addEvent(ParameterChanges& changes, uint32_t id,
   200→              int32 offset, double norm) {
   201→    int32 idx = 0;
   202→    IParamValueQueue* q = changes.addParameterData(id, idx);
   203→    if (q) {
   204→        int32 pidx = 0;
   205→        q->addPoint(offset, norm, pidx);
   206→    }
   207→}

   216→void caseModuleLoad(CaseResult& c, const std::string& modulePath) {
   217→    std::string err;
   218→    g_module = VST3::Hosting::Module::create(modulePath, err);
   219→    check(c, g_module != nullptr, "Module::create failed: " + err);
   220→    if (!g_module) return;
   221→    const auto& factory = g_module->getFactory();
   222→    check(c, factory.get() != nullptr, "no IPluginFactory exported");
   223→    const auto infos = factory.classInfos();
   224→    addValue(c, "factory_class_count", (double)infos.size());
   225→    bool found = false;
   226→    for (const auto& ci : infos)
   227→        if (std::memcmp(ci.ID().data(), octavox::vst3wrap::OctavoxNativeProcessorUID,
   228→                         sizeof(TUID)) == 0)
   229→            found = true;
   230→    check(c, found, "the processor class ID is not among the factory classes");
   231→    check(c, !infos.empty() && infos[0].category().find("Audio Module Class") !=
   232→                                    std::string::npos,
   233→          "unexpected class category");
   234→    c.pass = c.note.empty();
   235→}

   237→void caseProcessorCreation(CaseResult& c) {
   238→    if (!g_module) { c.note = "no module"; return; }
   239→    const auto& factory = g_module->getFactory();
   240→    g_component = factory.createInstance<IComponent>(
   241→        VST3::UID(octavox::vst3wrap::OctavoxNativeProcessorUID));
   242→    check(c, g_component != nullptr, "createInstance(IComponent) failed");
   243→    if (!g_component) return;
   244→    {
   245→        IAudioProcessor* ap = nullptr;
   246→        if (g_component->queryInterface(IAudioProcessor::iid,
   247→                                        reinterpret_cast<void**>(&ap)) == kResultTrue &&
   248→            ap)
   249→            g_audio = IPtr<IAudioProcessor>(ap);  // adopts the add-ref'd ptr
   250→    }
   251→    check(c, g_audio != nullptr, "component does not export IAudioProcessor");
   252→    {
   253→        IEditController* ec = nullptr;
   254→        if (g_component->queryInterface(IEditController::iid,
   255→                                        reinterpret_cast<void**>(&ec)) == kResultTrue &&
   256→            ec)
   257→            g_edit = IPtr<IEditController>(ec);
   258→    }
   259→    check(c, g_edit != nullptr,
   260→          "component does not export IEditController (single-component)");
   261→    c.pass = c.note.empty();
   262→}

   264→void caseComponentInit(CaseResult& c) {
   265→    if (!g_component) { c.note = "no component"; return; }
   266→    static HostApplication hostCtx;  // the SDK hosting utility
   267→    g_hostContext = &hostCtx;
   268→    const tresult r = g_component->initialize(&hostCtx);
   269→    check(c, r == kResultOk, "initialize(host context) failed");
   270→    const int32 inBuses = g_component->getBusCount(MediaTypes::kAudio, BusDirections::kInput);
   271→    const int32 outBuses = g_component->getBusCount(MediaTypes::kAudio, BusDirections::kOutput);
   272→    addValue(c, "audio_input_buses", (double)inBuses);
   273→    addValue(c, "audio_output_buses", (double)outBuses);
   274→    check(c, inBuses == 1 && outBuses == 1,
   275→          "expected symmetric single Main audio buses");
   276→    // the parameter surface is published at initialization (host contract)
   277→    check(c, g_edit->getParameterCount() == 38,
   278→          "getParameterCount() != 38 on the loaded instance after initialize");
   279→    addValue(c, "parameter_count", (double)g_edit->getParameterCount());
   280→    c.pass = c.note.empty();
   281→}

   283→void caseBusMono(CaseResult& c) {
   284→    if (!g_audio) { c.note = "no processor"; return; }
   285→    SpeakerArrangement in = SpeakerArr::kMono, out = SpeakerArr::kMono;
   286→    const tresult r = g_audio->setBusArrangements(&in, 1, &out, 1);
   287→    check(c, r == kResultOk, "mono negotiation rejected");
   288→    SpeakerArrangement gin = SpeakerArr::kEmpty, gout = SpeakerArr::kEmpty;
   289→    g_audio->getBusArrangement(BusDirections::kInput, 0, gin);
   290→    g_audio->getBusArrangement(BusDirections::kOutput, 0, gout);
   291→    check(c, SpeakerArr::getChannelCount(gin) == 1 &&
   292→                 SpeakerArr::getChannelCount(gout) == 1,
   293→          "negotiated mono not reflected by getBusArrangement");
   294→    c.pass = c.note.empty();
   295→}

   313→void caseActivationLifecycle(CaseResult& c) {
   314→    if (!g_component || !g_audio) { c.note = "no processor"; return; }
   315→    check(c, setupLoadedInstance(Setup{}, 1), "setupProcessing/setActive/setProcessing chain failed");
   316→    teardownLoadedInstance();
   317→    // re-activation must work (hosts activate/deactivate around latency changes)
   318→    ProcessSetup ps;
   319→    std::memset(&ps, 0, sizeof(ps));
   320→    ps.processMode = kRealtime;
   321→    ps.symbolicSampleSize = kSample32;
   322→    ps.sampleRate = 48000.0;
   323→    ps.maxSamplesPerBlock = 2048;
   324→    check(c, g_audio->setupProcessing(ps) == kResultOk, "re-setupProcessing failed");
   325→    check(c, g_component->setActive(true) == kResultOk, "re-activate failed");
   326→    check(c, g_audio->setProcessing(true) == kResultOk, "re-setProcessing failed");
   327→    teardownLoadedInstance();
   328→    c.pass = c.note.empty();
   329→}

   331→// mono: the module's output == the hand-driven core, bit-exact
   332→void caseProcessMono(CaseResult& c) {
   333→    if (!g_audio) { c.note = "no processor"; return; }
   334→    const int N = 65536;
   335→    Lcg rng(987654321);
   336→    std::vector<float> in(N), modOut(N), refOut(N);
   337→    rng.fill(in);
   338→    // ---- the hand-driven reference (the core library linked directly) ----
   339→    // R18-WDB: the reference uses THIS build variant's interpolation
   340→    // (octavox_build_config.h via defaultConfig()); an OBSKERNEL6 module is
   341→    // never compared against a LAGRANGE6 reference.
   342→    {
   343→        octavox::MultichannelShifter ref;
   344→        check(c, ref.prepare(octavox::vst3wrap::defaultConfig(), 1), "reference prepare failed");
   345→        float* inP[1] = {in.data()};
   346→        float* outP[1] = {refOut.data()};
   347→        ref.process(inP, outP, 1, N);
   348→    }
   349→    // ---- the loaded module ----
   350→    check(c, setupLoadedInstance(Setup{}, 1), "activation failed");
   351→    for (int off = 0; off < N; off += 4096) {
   352→        const int n = std::min(4096, N - off);
   353→        float* inP[1] = {in.data() + off};
   354→        float* outP[1] = {modOut.data() + off};
   355→        driveLoaded(inP, outP, 1, n, nullptr);
   356→    }
   357→    teardownLoadedInstance();
   358→    const bool bitex = bitIdentical(modOut.data(), refOut.data(), N);
   359→    const double d = maxAbsDiff(modOut, refOut);
   360→    addValue(c, "mono_max_abs_diff", d);
   361→    addValue(c, "mono_bit_exact", bitex ? 1.0 : 0.0);
   362→    check(c, bitex, "mono stream is NOT bit-identical to the hand-driven core");
   363→    c.pass = c.note.empty();
   364→}

   366→// stereo: per-channel distinct inputs, both channels bit-exact
   367→void caseProcessStereo(CaseResult& c) {
   368→    if (!g_audio) { c.note = "no processor"; return; }
   369→    const int N = 49152;  // not a multiple of the block size on purpose
   370→    Lcg rng(24680);
   371→    std::vector<float> inL(N), inR(N), outL(N), outR(N), refL(N), refR(N);
   372→    rng.fill(inL);
   373→    rng.fill(inR);
   374→    {
   375→        octavox::MultichannelShifter ref;
   376→        check(c, ref.prepare(octavox::vst3wrap::defaultConfig(), 2), "reference prepare failed");
   377→        float* inP[2] = {inL.data(), inR.data()};
   378→        float* outP[2] = {refL.data(), refR.data()};
   379→        ref.process(inP, outP, 2, N);
   380→    }
   381→    check(c, setupLoadedInstance(Setup{}, 2), "activation failed");
   382→    for (int off = 0; off < N; off += 3000) {  // odd block sizes
   383→        const int n = std::min(3000, N - off);
   384→        float* inP[2] = {inL.data() + off, inR.data() + off};
   385→        float* outP[2] = {outL.data() + off, outR.data() + off};
   386→        driveLoaded(inP, outP, 2, n, nullptr);
   387→    }
   388→    teardownLoadedInstance();
   389→    const bool bitex = bitIdentical(outL.data(), refL.data(), N) &&
   390→                       bitIdentical(outR.data(), refR.data(), N);
   391→    addValue(c, "stereo_L_max_abs_diff", maxAbsDiff(outL, refL));
   392→    addValue(c, "stereo_R_max_abs_diff", maxAbsDiff(outR, refR));
   393→    addValue(c, "stereo_bit_exact", bitex ? 1.0 : 0.0);
   394→    check(c, bitex, "stereo stream is NOT bit-identical to the hand-driven core");
   395→    c.pass = c.note.empty();
   396→}

   398→// parameter queues: a full non-default surface applied through real
   399→// IParameterChanges queues at stream position 0 == the hand-driven reference
   400→// with the canonical application order (the wrapper test's proven pattern,
   401→// now through the loaded module); plus a mid-block event drop-control.
   402→void caseParamQueues(CaseResult& c) {
   403→    if (!g_audio) { c.note = "no processor"; return; }
   404→    using octavox::vst3wrap::ParamId;
   405→    using octavox::vst3wrap::voiceOffsetId;
   406→    using octavox::vst3wrap::voiceGainId;
   407→    using octavox::vst3wrap::voiceMuteId;
   408→    using octavox::vst3wrap::voiceDelayId;
   409→    const int N = 65536;
   410→    struct Ev { uint32_t id; double plain; };
   411→    const std::vector<Ev> surf = {
   412→        {ParamId::kPitch, 2.0},
   413→        {voiceOffsetId(3), 7.0},
   414→        {voiceGainId(5), 0.5},
   415→        {voiceMuteId(6), 1.0},
   416→        {voiceDelayId(2), 20.0},
   417→        {ParamId::kVoices, 8.0},
   418→        {ParamId::kWetDry, 0.7},
   419→    };
   420→    Lcg rng(555777);
   421→    std::vector<float> in(N);
   422→    rng.fill(in);
   423→    // ---- hand-driven reference (canonical order) ----
   424→    std::vector<float> refOut(N);
   425→    {
   426→        octavox::MultichannelShifter ref;
   427→        // R18-WDB: the reference engine uses THIS build variant's kernel.
   428→        octavox::Config cfg = octavox::vst3wrap::defaultConfig();
   429→        check(c, ref.prepare(cfg, 1), "reference prepare failed");
   430→        const double sr = cfg.sample_rate, grain = cfg.grain;
   431→        const double pitch = toEnginePlain(ParamId::kPitch, 2.0);
   432→        const double off3 = toEnginePlain(voiceOffsetId(3), 7.0);
   433→        const double g5 = toEnginePlain(voiceGainId(5), 0.5);
   434→        const bool m6 = toEnginePlain(voiceMuteId(6), 1.0) >= 0.5;
   435→        const double d2ms = toEnginePlain(voiceDelayId(2), 20.0);
   436→        const int voices = (int)std::lround(toEnginePlain(ParamId::kVoices, 8.0));
   437→        const double wet = toEnginePlain(ParamId::kWetDry, 0.7);
   438→        ref.setSpread(true);
   439→        ref.setRatio(std::pow(2.0, pitch / 12.0));
   440→        ref.setVoiceRatio(3, std::pow(2.0, pitch / 12.0) * std::pow(2.0, off3 / 12.0));
   441→        ref.setVoiceGain(5, g5);
   442→        ref.setVoiceMute(6, m6);
   443→        ref.setVoiceBaseDelay(2, octavox::vst3wrap::msToBaseDelaySamples(d2ms, sr, grain));
   444→        ref.setVoices(voices);
   445→        ref.setFeedback(0.0);
   446→        ref.setWetDry(wet);
   447→        ref.setBypass(false);
   448→        float* inP[1] = {in.data()};
   449→        float* outP[1] = {refOut.data()};
   450→        ref.process(inP, outP, 1, N);
   451→    }
   452→    // ---- the loaded module with a position-0 queue in canonical order ----
   453→    std::vector<float> aOut(N);
   454→    check(c, setupLoadedInstance(Setup{}, 1), "activation failed");
   455→    {
   456→        ParameterChanges ch;
   457→        for (const auto& e : surf)
   458→            addEvent(ch, e.id, 0,
   459→                     octavox::vst3wrap::normalizedFromPlain(e.id, e.plain));
   460→        float* inP[1] = {in.data()};
   461→        float* outP[1] = {aOut.data()};
   462→        driveLoaded(inP, outP, 1, 4096, &ch);
   463→        float* in2[1] = {in.data() + 4096};
   464→        float* out2[1] = {aOut.data() + 4096};
   465→        driveLoaded(in2, out2, 1, N - 4096, nullptr);
   466→    }
   467→    teardownLoadedInstance();
   468→    const double dLive = maxAbsDiff(aOut, refOut);
   469→    addValue(c, "queue_surface_vs_reference_maxdiff", dLive);
   470→    check(c, dLive == 0.0,
   471→          "queued surface application is not bit-identical to the reference");
   472→    // ---- drop-control: a mid-block event must change the stream vs none ----
   473→    std::vector<float> bOut(N), cOut(N);
   474→    check(c, setupLoadedInstance(Setup{}, 1), "re-activation failed");
   475→    {
   476→        ParameterChanges ch;
   477→        addEvent(ch, ParamId::kWetDry, 2000, 0.25);  // mid-block event
   478→        for (int off = 0; off < N; off += 4096) {
   479→            const int n = std::min(4096, N - off);
   480→            float* inP[1] = {in.data() + off};
   481→            float* outP[1] = {bOut.data() + off};
   482→            driveLoaded(inP, outP, 1, n, off == 0 ? &ch : nullptr);
   483→        }
   484→        for (int off = 0; off < N; off += 4096) {
   485→            const int n = std::min(4096, N - off);
   486→            float* inP[1] = {in.data() + off};
   487→            float* outP[1] = {cOut.data() + off};
   488→            driveLoaded(inP, outP, 1, n, nullptr);
   489→        }
   490→    }
   491→    teardownLoadedInstance();
   492→    const double dDrop = maxAbsDiff(bOut, cOut);
   493→    addValue(c, "midblock_event_effect_maxdiff", dDrop);
   494→    check(c, dDrop > 0.0,
   495→          "the mid-block parameter event had no effect (queue not consumed)");
   496→    c.pass = c.note.empty();
   497→}

   499→// state save through the module's IComponent::getState, restore into a
   500→// SECOND factory-created instance, both continue bit-identically
   501→void caseStateSaveRestore(CaseResult& c) {
   502→    if (!g_module || !g_audio) { c.note = "no module"; return; }
   503→    using octavox::vst3wrap::ParamId;
   504→    const int N = 32768;
   505→    Lcg rng(13579);
   506→    std::vector<float> in(N);
   507→    rng.fill(in);
   508→    MemoryStream state;
   509→    std::vector<float> run1(N), run2(N);
   510→    check(c, setupLoadedInstance(Setup{}, 1), "activation failed");
   511→    {
   512→        // a non-default surface via the queue, then run, then save
   513→        ParameterChanges ch;
   514→        addEvent(ch, ParamId::kPitch, 0,
   515→                 octavox::vst3wrap::normalizedFromPlain(ParamId::kPitch, -5.0));
   516→        addEvent(ch, ParamId::kFeedback, 0,
   517→                 octavox::vst3wrap::normalizedFromPlain(ParamId::kFeedback, 0.3));
   518→        for (int off = 0; off < N; off += 4096) {
   519→            const int n = std::min(4096, N - off);
   520→            float* inP[1] = {in.data() + off};
   521→            float* outP[1] = {run1.data() + off};
   522→            driveLoaded(inP, outP, 1, n, off == 0 ? &ch : nullptr);
   523→        }
   524→        check(c, g_component->getState(&state) == kResultOk, "getState failed");
   525→    }
   526→    teardownLoadedInstance();
   527→    addValue(c, "state_blob_bytes", (double)((int64_t)state.getSize()));
   528→    check(c, state.getSize() > 4, "state blob suspiciously small");
   529→    // ---- a SECOND instance from the same loaded factory ----
   530→    IPtr<IComponent> comp2 = g_module->getFactory().createInstance<IComponent>(
   531→        VST3::UID(octavox::vst3wrap::OctavoxNativeProcessorUID));
   532→    check(c, comp2 != nullptr, "second instance creation failed");
   533→    if (!comp2) { c.pass = false; return; }
   534→    IPtr<IAudioProcessor> audio2;
   535→    {
   536→        IAudioProcessor* ap2 = nullptr;
   537→        if (comp2->queryInterface(IAudioProcessor::iid,
   538→                                  reinterpret_cast<void**>(&ap2)) == kResultTrue && ap2)
   539→            audio2 = IPtr<IAudioProcessor>(ap2);
   540→    }
   541→    check(c, audio2 != nullptr, "second instance lacks IAudioProcessor");
   542→    check(c, comp2->initialize(g_hostContext) == kResultOk,
   543→          "second instance initialize failed");
   544→    SpeakerArrangement mono = SpeakerArr::kMono;
   545→    audio2->setBusArrangements(&mono, 1, &mono, 1);
   546→    ProcessSetup ps;
   547→    std::memset(&ps, 0, sizeof(ps));
   548→    ps.processMode = kRealtime;
   549→    ps.symbolicSampleSize = kSample32;
   550→    ps.sampleRate = 48000.0;
   551→    ps.maxSamplesPerBlock = 4096;
   552→    audio2->setupProcessing(ps);
   553→    state.seek(0, IBStream::kIBSeekSet, nullptr);
   554→    check(c, comp2->setState(&state) == kResultOk, "setState into instance 2 failed");
   555→    comp2->setActive(true);
   556→    audio2->setProcessing(true);
   557→    for (int off = 0; off < N; off += 4096) {
   558→        const int n = std::min(4096, N - off);
   559→        float* inP[1] = {in.data() + off};
   560→        float* outP[1] = {run2.data() + off};
   561→        AudioBusBuffers ib{}, ob{};
   562→        ib.numChannels = 1;
   563→        ib.channelBuffers32 = inP;
   564→        ob.numChannels = 1;
   565→        ob.channelBuffers32 = outP;
   566→        AudioBusBuffers inB[1] = {ib}, outB[1] = {ob};
   567→        ProcessData pd{};
   568→        pd.processMode = kRealtime;
   569→        pd.symbolicSampleSize = kSample32;
   570→        pd.numSamples = n;
   571→        pd.inputs = inB;
   572→        pd.numInputs = 1;
   573→        pd.outputs = outB;
   574→        pd.numOutputs = 1;
   575→        audio2->process(pd);
   576→    }
   577→    audio2->setProcessing(false);
   578→    comp2->setActive(false);
   579→    const bool bitex = bitIdentical(run1.data(), run2.data(), N);
   580→    addValue(c, "restored_replay_bit_exact", bitex ? 1.0 : 0.0);
   581→    addValue(c, "restored_replay_maxdiff", maxAbsDiff(run1, run2));
   582→    check(c, bitex,
   583→          "the restored second instance does not replay bit-identically");
   584→    check(c, comp2->terminate() == kResultOk, "second instance terminate failed");
   585→    c.pass = c.note.empty();
   586→}

   588→void caseLatencyReporting(CaseResult& c) {
   589→    if (!g_audio) { c.note = "no processor"; return; }
   590→    // the wrapper reports the core's latencySamples(); compare against the
   591→    // core library's own value for the variant's config (the exact contract).
   592→    octavox::MultichannelShifter ref;
   593→    check(c, ref.prepare(octavox::vst3wrap::defaultConfig(), 2), "reference prepare failed");
   594→    const uint32 expect = ref.latencySamples();
   595→    const uint32 got = g_audio->getLatencySamples();
   596→    addValue(c, "expected_latency_samples", (double)expect);
   597→    addValue(c, "reported_latency_samples", (double)got);
   598→    check(c, got == expect,
   599→          "the module's getLatencySamples() != the core's latencySamples()");
   600→    const uint32 tail = g_audio->getTailSamples();
   601→    addValue(c, "reported_tail_samples", (double)tail);
   602→    check(c, tail > 0, "tail samples not reported");
   603→    c.pass = c.note.empty();
   604→}
   605→
   606→void caseTerminateUnload(CaseResult& c) {
   607→    if (!g_component) { c.note = "no component"; return; }
   608→    check(c, g_component->terminate() == kResultOk, "terminate failed");
   609→    // release every interface, then destroy the Module (dlclose)
   610→    g_edit = nullptr;
   611→    g_audio = nullptr;
   612→    g_component = nullptr;
   613→    const std::string modName = g_module ? g_module->getName() : "";
   614→    g_module = nullptr;  // ~Module -> library unload
   615→    addValue(c, "module_name_len", (double)modName.size());
   616→    check(c, true, "");
   617→    c.pass = c.note.empty();
   618→}

   622→//==========================================================================
   623→// main
   624→//==========================================================================
   625→int main(int argc, char** argv) {
   626→    std::string modulePath, jsonPath;
   627→    for (int i = 1; i < argc; ++i) {
   628→        const std::string a = argv[i];
   629→        if (a == "--module" && i + 1 < argc) modulePath = argv[++i];
   630→        else if (a == "--json" && i + 1 < argc) jsonPath = argv[++i];
   631→        else {
   632→            std::fprintf(stderr, "usage: %s [--module <path/to/%s>] "
   633→                                 "[--json <path>]\n", argv[0],
   634→                         octavox::vst3wrap::build::kBundleName);
   635→            return 2;
   636→        }
   637→    }
   638→    // R18-WDB: THIS build variant's bundle (OctavoxRE_<Variant>.vst3) from
   639→    // the central build configuration helper.
   640→    const std::string bundleName = octavox::vst3wrap::build::bundleFileName();
   641→    // default: <exe_dir>/../../VST3/Release/<bundle> (both separators --
   642→    // the harness also runs on Windows builds where argv[0] uses '\')
   643→    if (modulePath.empty()) {
   644→        std::string exe = argv[0];
   645→        const size_t slash = exe.find_last_of("/\\");
   646→        const std::string dir = slash == std::string::npos ? "." : exe.substr(0, slash);
   647→        const std::string cands[] = {
   648→            "/../../VST3/Release/" + bundleName,
   649→            "/../../../VST3/Release/" + bundleName,
   650→            "/" + bundleName,
   651→        };
   652→        for (const std::string& suff : cands) {
   653→            std::string p = dir + suff;
   654→            FILE* f = std::fopen(p.c_str(), "rb");
   655→            if (f) { std::fclose(f); modulePath = p; break; }
   656→        }
   657→        if (modulePath.empty()) {
   658→            // last resort: relative to the CWD
   659→            const std::string rel[] = {"build/VST3/Release/" + bundleName,
   660→                                       "VST3/Release/" + bundleName};
   661→            for (const std::string& p : rel) {
   662→                FILE* f = std::fopen(p.c_str(), "rb");
   663→                if (f) { std::fclose(f); modulePath = p; break; }
   664→            }
   665→        }
   666→    }
   667→    if (modulePath.empty()) {
   668→        std::fprintf(stderr, "error: %s not found "
   669→                             "(build it first, or pass --module)\n",
   670→                     bundleName.c_str());
   671→        return 2;
   672→    }
   673→    std::printf("module: %s\n", modulePath.c_str());
   674→    std::printf("build variant: %s (default interpolation: %s)\n",
   675→                octavox::vst3wrap::build::kBundleName,
   676→                octavox::vst3wrap::build::kInterpolationName);
   677→    std::printf("harness scope: SDK module loader + factory + full interface "
   678→                "contract vs the hand-driven core (NOT a DAW test)\n\n");
   679→
   680→    std::vector<CaseResult> R;
   681→    {
   682→        CaseResult c{"module_load_factory"};
   683→        caseModuleLoad(c, modulePath);
   684→        pushCase(R, std::move(c));
   685→    }
   686→    {
   687→        CaseResult c{"processor_creation"};
   688→        caseProcessorCreation(c);
   689→        pushCase(R, std::move(c));
   690→    }
   691→    {
   692→        CaseResult c{"component_initialize_buses"};
   693→        caseComponentInit(c);
   694→        pushCase(R, std::move(c));
   695→    }
   696→    {
   697→        CaseResult c{"bus_negotiation_mono"};
   698→        caseBusMono(c);
   699→        pushCase(R, std::move(c));
   700→    }
   701→    {
   702→        CaseResult c{"bus_negotiation_stereo"};
   703→        caseBusStereo(c);
   704→        pushCase(R, std::move(c));
   705→    }
   706→    {
   707→        CaseResult c{"activation_lifecycle"};
   708→        caseActivationLifecycle(c);
   709→        pushCase(R, std::move(c));
   710→    }
   711→    {
   712→        CaseResult c{"process_mono_bitexact"};
   713→        caseProcessMono(c);
   714→        pushCase(R, std::move(c));
   715→    }
   716→    {
   717→        CaseResult c{"process_stereo_bitexact"};
   718→        caseProcessStereo(c);
   719→        pushCase(R, std::move(c));
   720→    }
   721→    {
   722→        CaseResult c{"param_queue_events"};
   723→        caseParamQueues(c);
   724→        pushCase(R, std::move(c));
   725→    }
   726→    {
   727→        CaseResult c{"state_save_restore_second_instance"};
   728→        caseStateSaveRestore(c);
   729→        pushCase(R, std::move(c));
   730→    }
   731→    {
   732→        CaseResult c{"latency_reporting"};
   733→        caseLatencyReporting(c);
   734→        pushCase(R, std::move(c));
   735→    }
   736→    {
   737→        CaseResult c{"terminate_unload"};
   738→        caseTerminateUnload(c);
   739→        pushCase(R, std::move(c));
   740→    }
   741→
   742→    int passed = 0, failed = 0;
   743→    for (const auto& c : R) (c.pass ? passed : failed)++;
   744→    std::printf("\n%s: %d passed, %d failed, %d total\n",
   745→                "octavox-module-integration", passed, failed, (int)R.size());
   746→    std::printf("TOTAL_CASES %d\n", (int)R.size());
   747→
   748→    if (!jsonPath.empty()) {
   749→        FILE* f = std::fopen(jsonPath.c_str(), "w");
   750→        if (f) {
   751→            std::fprintf(f, "{\n  \"suite\": \"octavox-module-integration\",\n");
   752→            std::fprintf(f, "  \"build_variant\": \"%s\",\n",
   753→                         octavox::vst3wrap::build::kBundleName);
   754→            std::fprintf(f, "  \"default_interpolation\": \"%s\",\n",
   755→                         octavox::vst3wrap::build::kInterpolationName);
   756→            std::fprintf(f, "  \"module_path\": \"%s\",\n",
   757→                         jsonEscape(modulePath).c_str());
   758→            std::fprintf(f, "  \"cases\": [\n");
   759→            bool first = true;
   760→            for (const auto& c : R) {
   761→                if (!first) std::fprintf(f, ",\n");
   762→                first = false;
   763→                std::fprintf(f, "    {\"name\": \"%s\", \"pass\": %s",
   764→                             jsonEscape(c.name).c_str(), c.pass ? "true" : "false");
   765→                if (!c.values.empty()) {
   766→                    std::fprintf(f, ", \"values\": {");
   767→                    for (size_t i = 0; i < c.values.size(); ++i) {
   768→                        if (i) std::fprintf(f, ", ");
   769→                        std::fprintf(f, "\"%s\": %.10g",
   770→                                     jsonEscape(c.values[i].first).c_str(),
   771→                                     c.values[i].second);
   772→                    }
   773→                    std::fprintf(f, "}");
   774→                }
   775→                std::fprintf(f, ", \"note\": \"%s\"}",
   776→                             jsonEscape(c.note).c_str());
   777→            }
   778→            std::fprintf(f, "\n  ],\n");
   779→            std::fprintf(f, "  \"summary\": {\"passed\": %d, \"failed\": %d, "
   780→                            "\"total\": %d}\n}\n",
   781→                         passed, failed, (int)R.size());
   782→            std::fclose(f);
   783→        }
   784→    }
   785→    return failed == 0 ? 0 : 1;
   786→}
