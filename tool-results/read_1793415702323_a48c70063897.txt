   143→# refcap_gate <evidence-file> <reference-file> <label>: all four hashes must

   145→refcap_gate() { # $1 = captured output $2 = reference hashes $3 = label
   146→    local OK=1
   147→    while IFS= read -r cfg; do
   148→        local got want
   149→        got="$(rg "${cfg}" "$1" | rg -o '0x[0-9a-f]{16}' | head -1)"
   150→        want="$(rg "^ *${cfg}" "$2" | rg -o '0x[0-9a-f]{16}' | head -1)"
   151→        [ "${got}" = "${want}" ] || { fail "$3 hash ${cfg}: got ${got} want ${want}"; OK=0; }
   152→    done <<<"${CFGS}"
   153→    [ "${OK}" = "1" ] && say "  $3: 4/4 exact vs $(basename "$2")"
   154→    return 0
   155→}

   180→# A Windows module is "complete" only if it exists AND parses as a PE32+
   181→# x86-64 DLL (guards against a link interrupted mid-write by the sandbox's
   182→# process termination -- an invalid module is removed and re-linked).
   183→#
   184→# file_pe_x64 probes file(1) via command substitution + case globbing (no
   185→# pipe, no regex escaping) with ONE defensive retry. (History: the chain's
   186→# first authoring used `rg "PE32+ executable..."` -- an unescaped `+` that
   187→# never matches the literal "PE32+" -- which made the informational exe
   188→# probe print a false negative; the glob form cannot regress that way.)
   189→file_pe_x64() { # $1 = path, $2 = optional "dll" (also require the DLL flag)
   190→    local out
   191→    out="$(file "$1" 2>/dev/null || true)"
   192→    case "${out}" in
   193→        *"PE32+ executable for MS Windows"*x86-64*) ;;
   194→        *) sleep 1; out="$(file "$1" 2>/dev/null || true)" ;;
   195→    esac
   196→    case "${out}" in
   197→        *"PE32+ executable for MS Windows"*x86-64*)
   198→            if [ "${2:-}" = "dll" ]; then
   199→                case "${out}" in *"(DLL)"*) return 0 ;; *) return 1 ;; esac
   200→            fi
   201→            return 0 ;;
   202→    esac
   203→    return 1
   204→}
