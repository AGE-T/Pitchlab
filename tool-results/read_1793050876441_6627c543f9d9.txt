    41→/// 64-bit LCG, identical on the Python side. u in [0,1); sample = (2u-1)*amp.
    42→struct Lcg {
    43→    uint64_t s;
    44→    explicit Lcg(uint64_t seed) : s(seed * 6364136223846793005ULL +
    45→                                     1442695040888963407ULL) {}
    46→    double next() {
    47→        s = s * 6364136223846793005ULL + 1442695040888963407ULL;
    48→        const uint64_t m = (s >> 11) & ((1ULL << 53) - 1);
    49→        return static_cast<double>(m) / 9007199254740992.0;
    50→    }
    51→    double noise(double amp) { return (2.0 * next() - 1.0) * amp; }
    52→};

    54→std::vector<float> tone(double f, int sr, int n, double amp) {
    55→    std::vector<float> x(n);
    56→    for (int i = 0; i < n; ++i)
    57→        x[i] = static_cast<float>(amp * std::sin(2.0 * M_PI * f * i / sr));
    58→    return x;
    59→}

    61→std::vector<float> chord(const std::vector<double>& fs, int sr, int n,
    62→                         double amp) {
    63→    std::vector<float> x(n, 0.0f);
    64→    for (double f : fs)
    65→        for (int i = 0; i < n; ++i)
    66→            x[i] += static_cast<float>(amp / fs.size() *
    67→                                       std::sin(2.0 * M_PI * f * i / sr));
    68→    return x;

    84→/// Percussive burst train: exponential decays.
    85→std::vector<float> bursts(int sr, int n, double period_s, double amp) {
    86→    std::vector<float> x(n, 0.0f);
    87→    Lcg rng(99);
    88→    const int per = static_cast<int>(period_s * sr);
    89→    for (int i = 0; i < n; ++i) {
    90→        const int t = i % per;
    91→        if (t < 200)
    92→            x[i] = static_cast<float>(amp * std::exp(-t / 40.0) *
    93→                                      (t == 0 ? 1.0 : rng.noise(1.0)));
    94→    }
    95→    return x;
    96→}

    98→std::vector<float> chirp(double f0, double f1, int sr, int n, double amp) {
    99→    std::vector<float> x(n);
   100→    const double T = static_cast<double>(n) / sr;
   101→    for (int i = 0; i < n; ++i) {
   102→        const double t = static_cast<double>(i) / sr;
   103→        const double f = f0 * std::pow(f1 / f0, t / T);
   104→        x[i] = static_cast<float>(amp * std::sin(2.0 * M_PI * f * t));
   105→    }
   106→    return x;
   107→}

   116→/// Radix-2 complex FFT (in-place, n a power of two).
   117→void fft(double* re, double* im, int n) {
   118→    for (int i = 1, j = 0; i < n; ++i) {
   119→        int bit = n >> 1;
   120→        for (; j & bit; bit >>= 1) j ^= bit;
   121→        j ^= bit;
   122→        if (i < j) {
   123→            std::swap(re[i], re[j]);
   124→            std::swap(im[i], im[j]);
   125→        }
   126→    }
   127→    for (int len = 2; len <= n; len <<= 1) {
   128→        const double ang = -2.0 * M_PI / len;
   129→        const double wr = std::cos(ang), wi = std::sin(ang);
   130→        for (int i = 0; i < n; i += len) {
   131→            double cr = 1.0, ci = 0.0;
   132→            for (int k = 0; k < len / 2; ++k) {
   133→                const double ur = re[i + k], ui = im[i + k];
   134→                const double vr = re[i + k + len / 2] * cr -
   135→                                  im[i + k + len / 2] * ci;
   136→                const double vi = re[i + k + len / 2] * ci +
   137→                                  im[i + k + len / 2] * cr;
   138→                re[i + k] = ur + vr;
   139→                im[i + k] = ui + vi;
   140→                re[i + k + len / 2] = ur - vr;
   141→                im[i + k + len / 2] = ui - vi;
   142→                const double nr = cr * wr - ci * wi;
   143→                ci = cr * wi + ci * wr;
   144→                cr = nr;
   145→            }
   146→        }
   147→    }
   148→}

   150→/// FFT peak pitch with the CORRECTED Hann-exact 3-point estimator.
   151→/// Returns 0.0 when no peak above the threshold is found.
   152→///
   153→/// R15-REB1 fix (recovered from the record): the interpolation now runs on
   154→/// the three bins' MAGNITUDES with the Hann-exact formula
   155→///     delta = 2*(c - a) / (a + 2*b + c)
   156→/// (a, b, c = magnitudes of bins kpeak-1, kpeak, kpeak+1) instead of the
   157→/// old power-domain parabola 0.5*(a-c)/(a-2b+c) on squared magnitudes,
   158→/// which is biased for off-bin tones under a Hann window (the historical
   159→/// round-15 record: the correction made battery cases PASS, none failed).
   160→///
   161→/// FFT size N = the largest power of two <= min(n, 65536): the analysed
   162→/// NEWEST block of N samples always carries a complete Hann window, and
   163→/// the window offset off = n - N is >= 0 BY CONSTRUCTION (the estimator
   164→/// never reads before the supplied buffer start).
   165→double pitchFFT(const float* x, int n, int sr, double fmin, double fmax) {
   166→    int N = 1;
   167→    while (N < n && N < 65536) N <<= 1;  // largest pow2 <= min(n, 65536)
   168→    if (N < 16) return 0.0;
   169→    std::vector<double> re(N, 0.0), im(N, 0.0);
   170→    const int off = n - N;  // use the newest N samples (off >= 0 by construction)
   171→    for (int i = 0; i < N; ++i) {
   172→        const double w = 0.5 - 0.5 * std::cos(2.0 * M_PI * i / N);
   173→        re[i] = static_cast<double>(x[off + i]) * w;
   174→    }
   175→    fft(re.data(), im.data(), N);
   176→    const int kmin = std::max(1, static_cast<int>(fmin * N / sr));
   177→    const int kmax = std::min(N / 2 - 2, static_cast<int>(fmax * N / sr));
   178→    if (kmax <= kmin) return 0.0;
   179→    int kpeak = kmin;
   180→    double best = 0.0;
   181→    for (int k = kmin; k <= kmax; ++k) {
   182→        const double m = re[k] * re[k] + im[k] * im[k];  // power (same argmax)
   183→        if (m > best) {
   184→            best = m;
   185→            kpeak = k;
   186→        }
   187→    }
   188→    if (best <= 0.0) return 0.0;
   189→    // magnitudes of the three bins around the peak (Hann-exact estimator)
   190→    const double a = std::sqrt(re[kpeak - 1] * re[kpeak - 1] +
   191→                               im[kpeak - 1] * im[kpeak - 1]);
   192→    const double b = std::sqrt(best);
   193→    const double c = std::sqrt(re[kpeak + 1] * re[kpeak + 1] +
   194→                               im[kpeak + 1] * im[kpeak + 1]);
   195→    double d = 0.0;
   196→    const double den = a + 2.0 * b + c;
   197→    if (den > 1e-300) d = 2.0 * (c - a) / den;
   198→    if (d > 0.5) d = 0.5;
   199→    if (d < -0.5) d = -0.5;
   200→    return static_cast<double>(sr) * (kpeak + d) / N;
   201→}

   203→double rms(const float* x, int n) {
   204→    double s = 0.0;
   205→    for (int i = 0; i < n; ++i) s += static_cast<double>(x[i]) * x[i];
   206→    return n ? std::sqrt(s / n) : 0.0;
   207→}
   208→
   209→double peak(const float* x, int n) {
   210→    double p = 0.0;
   211→    for (int i = 0; i < n; ++i) p = std::max(p, std::abs(static_cast<double>(x[i])));
   212→    return p;
   213→}

   215→bool allFinite(const float* x, int n) {
   216→    for (int i = 0; i < n; ++i)
   217→        if (!std::isfinite(x[i])) return false;
   218→    return true;
   219→}

   221→/// Envelope modulation in dB: max/min of RMS windows of `w` samples over the
   222→/// steady region (skip the first skip samples). w = sr/1000 gives the 1 ms
   223→/// metric; w = period gives the period-synchronous metric (the r11 lesson:
   224→/// a 1 ms window phase-breathes several dB on a plain tone).
   225→double envModulationDb(const float* x, int n, int sr, int skip, int w) {
   226→    std::vector<double> env;
   227→    for (int i0 = skip; i0 + w <= n; i0 += w) {
   228→        env.push_back(rms(x + i0, w));
   229→    }
   230→    if (env.empty()) return 0.0;
   231→    double lo = 1e300, hi = 0.0;
   232→    for (double v : env) {
   233→        if (v < lo) lo = v;
   234→        if (v > hi) hi = v;
   235→    }
   236→    if (lo <= 0.0) return hi > 0.0 ? 999.0 : 0.0;
   237→    return 20.0 * std::log10(hi / lo);
   238→}
   239→
   240→double centsBetween(double fMeasured, double fNominal) {
   241→    if (fMeasured <= 0.0 || fNominal <= 0.0) return 999.0;
   242→    return 1200.0 * std::log2(fMeasured / fNominal);
   243→}
   244→
   245→/// Max absolute sample-to-sample difference (transition click meter).
   246→double maxStep(const float* x, int n, int i0 = 0) {
   247→    double m = 0.0;
   248→    for (int i = std::max(1, i0 + 1); i < n; ++i)
   249→        m = std::max(m, std::abs(static_cast<double>(x[i]) - x[i - 1]));
   250→    return m;
   251→}

   278→// Runs the engine over a full signal in one go (64-aligned internally is
   279→// irrelevant -- process takes the whole block).
   280→std::vector<float> run(const Config& cfg, const std::vector<float>& in,
   281→                       int block = 1 << 20) {
   282→    AlignedDelayLine e;
   283→    if (!e.prepare(cfg)) {
   284→        return {};
   285→    }
   286→    std::vector<float> out(in.size(), 0.0f);
   287→    int done = 0;
   288→    while (done < static_cast<int>(in.size())) {
   289→        const int n = std::min(block, static_cast<int>(in.size()) - done);
   290→        e.process(in.data() + done, out.data() + done, n);
   291→        done += n;
   292→    }
   293→    return out;
   294→}

   299→void casesPrepareReset(std::vector<CaseResult>& R) {
   300→    {  // 1. default prepare
   301→        CaseResult c{"prepare_default"};
   302→        Config cfg;
   303→        AlignedDelayLine e;
   304→        const bool ok = e.prepare(cfg);
   305→        check(c, ok, "prepare failed");
   306→        check(c, e.latencySamples() == 2304, "latency != 2304");
   307→        check(c, e.ready(), "not ready");
   308→        c.pass = c.note.empty();
   309→        R.push_back(std::move(c));
   310→    }
   311→    {  // 2. invalid configs rejected
   312→        CaseResult c{"prepare_invalid_rejected"};
   313→        int rejected = 0, total = 0;
   314→        Config bad;
   315→        auto tryCfg = [&](const Config& cfg) {
   316→            ++total;
   317→            AlignedDelayLine e;
   318→            if (!e.prepare(cfg)) ++rejected;
   319→        };
   320→        bad = Config();
   321→        bad.sample_rate = 0.0;
   322→        tryCfg(bad);
   323→        bad = Config();
   324→        bad.voices = 9;
   325→        tryCfg(bad);
   326→        bad = Config();
   327→        bad.grain = 0.0;
   328→        tryCfg(bad);
   329→        bad = Config();
   330→        bad.feedback = 1.0;
   331→        tryCfg(bad);
   332→        bad = Config();
   333→        bad.base_delay = 0.0;
   334→        tryCfg(bad);
   335→        addValue(c, "rejected", rejected);
   336→        addValue(c, "total", total);
   337→        check(c, rejected == total, "some invalid configs accepted");
   338→        c.pass = c.note.empty();
   339→        R.push_back(std::move(c));
   340→    }
   341→    {  // 3. reset determinism
   342→        CaseResult c{"reset_determinism"};
   343→        Config cfg;
   344→        cfg.ratio = semitoneRatio(400.0);
   345→        const auto in = gen::tone(220.0, 48000, 48000, 0.5);
   346→        AlignedDelayLine a, b;
   347→        a.prepare(cfg);
   348→        b.prepare(cfg);
   349→        std::vector<float> oa(in.size(), 0.f), ob(in.size(), 0.f),
   350→            ob2(in.size(), 0.f);
   351→        a.process(in.data(), oa.data(), in.size());
   352→        b.process(in.data(), ob.data(), in.size());
   353→        b.reset();
   354→        b.process(in.data(), ob2.data(), in.size());
   355→        double d1 = 0.0, d2 = 0.0;
   356→        for (size_t i = 0; i < in.size(); ++i) {
   357→            d1 = std::max(d1, std::abs((double)oa[i] - ob[i]));
   358→            d2 = std::max(d2, std::abs((double)ob[i] - ob2[i]));
   359→        }
   360→        addValue(c, "fresh_vs_fresh", d1);
   361→        addValue(c, "reset_vs_fresh", d2);
   362→        check(c, d1 == 0.0, "two fresh engines differ");
   363→        check(c, d2 == 0.0, "reset engine differs from fresh");
   364→        c.pass = c.note.empty();
   365→        R.push_back(std::move(c));
   366→    }
   367→    {  // 4. repeated prepare
   368→        CaseResult c{"repeated_prepare"};
   369→        Config cfg;
   370→        cfg.ratio = semitoneRatio(500.0);
   371→        const auto in = gen::tone(330.0, 48000, 24000, 0.4);
   372→        AlignedDelayLine a, b;
   373→        a.prepare(cfg);
   374→        std::vector<float> oa(in.size(), 0.f), ob(in.size(), 0.f);
   375→        a.process(in.data(), oa.data(), in.size());
   376→        a.prepare(cfg);  // re-prepare
   377→        a.process(in.data(), ob.data(), in.size());
   378→        b.prepare(cfg);
   379→        std::vector<float> oc(in.size(), 0.f);
   380→        b.process(in.data(), oc.data(), in.size());
   381→        double d = 0.0;
   382→        for (size_t i = 0; i < in.size(); ++i)
   383→            d = std::max(d, std::abs((double)ob[i] - oc[i]));
   384→        addValue(c, "reprepared_vs_fresh", d);
   385→        check(c, d == 0.0, "re-prepared engine differs from fresh");
   386→        c.pass = c.note.empty();
   387→        R.push_back(std::move(c));
   388→    }
   389→    {  // 5. block-size invariance
   390→        CaseResult c{"block_size_invariance"};
   391→        Config cfg;
   392→        cfg.ratio = semitoneRatio(700.0);
   393→        const auto in = gen::tone(196.0, 48000, 48000, 0.4);
   394→        const auto ref = run(cfg, in, 1 << 20);
   395→        const int splits[] = {1, 7, 16, 64, 128, 512, 1024, 4096};
   396→        double worst = 0.0;
   397→        for (int s : splits) {
   398→            const auto got = run(cfg, in, s);
   399→            double d = 0.0;
   400→            for (size_t i = 0; i < in.size(); ++i)
   401→                d = std::max(d, std::abs((double)ref[i] - got[i]));
   402→            worst = std::max(worst, d);
   403→        }

   406→            AlignedDelayLine e;
   407→            e.prepare(cfg);
   408→            std::vector<float> got(in.size(), 0.f);
   409→            size_t i = 0;
   410→            int pat[] = {1, 64, 7, 512, 16, 4096, 3};
   411→            int pi = 0;
   412→            while (i < in.size()) {
   413→                int n = std::min(pat[pi % 7], (int)(in.size() - i));
   414→                e.process(in.data() + i, got.data() + i, n);
   415→                i += n;
   416→                ++pi;
   417→            }
   418→            double d = 0.0;
   419→            for (size_t k = 0; k < in.size(); ++k)
   420→                d = std::max(d, std::abs((double)ref[k] - got[k]));
   421→            worst = std::max(worst, d);
   422→        }
   423→        addValue(c, "worst_split_diff", worst);
   424→        check(c, worst == 0.0, "block split changed the output");
   425→        c.pass = c.note.empty();
   426→        R.push_back(std::move(c));
   427→    }
   428→    {  // 6. steady-state allocation-free
   429→        CaseResult c{"steady_state_no_allocation"};
   430→        Config cfg;
   431→        cfg.ratio = semitoneRatio(400.0);
   432→        cfg.voices = 8;
   433→        cfg.feedback = 0.35;
   434→        const auto in = gen::noise(1234, 48000, 0.5);
   435→        AlignedDelayLine e;
   436→        e.prepare(cfg);
   437→        std::vector<float> out(in.size(), 0.f);
   438→        e.process(in.data(), out.data(), 24000);  // warm-up (tables, paths)
   439→        allocs::reset();
   440→        allocs::arm(true);
   441→        e.process(in.data() + 24000, out.data() + 24000,
   442→                  static_cast<int>(in.size()) - 24000);
   443→        allocs::arm(false);
   444→        const long n = allocs::count();
   445→        addValue(c, "allocations_during_process", static_cast<double>(n));
   446→        check(c, n == 0, "allocations during steady-state process()");
   447→        c.pass = c.note.empty();
   448→        R.push_back(std::move(c));
   449→    }

   452→void casesUnity(std::vector<CaseResult>& R) {
   453→    const int sr = 48000;
   454→
   455→    // bit-exact delayed-copy gate used by the unity battery
   456→    auto unityExact = [&](const std::string& name, const std::vector<float>& in,
   457→                          bool arrive_from_shift = false) {
   458→        CaseResult c{name};
   459→        Config cfg;  // ratio exactly 1.0, 1 voice, neutral
   460→        AlignedDelayLine e;
   461→        e.prepare(cfg);
   462→        if (arrive_from_shift) {
   463→            // arrive from +4 st first: 1 s shifted, then unity, then settle
   464→            e.setRatio(semitoneRatio(400.0));
   465→            std::vector<float> pre(sr, 0.f);
   466→            e.process(in.data(), pre.data(), sr);
   467→            e.setRatio(1.0);
   468→            std::vector<float> mid(static_cast<int>(0.3 * sr), 0.f);
   469→            if ((int)in.size() > sr)
   470→                e.process(in.data() + sr, mid.data(),
   471→                           std::min((int)in.size() - sr, (int)mid.size()));
   472→        }
   473→        const int d_park = e.voiceStats(0).parked_delay;
   474→        addValue(c, "d_park", d_park);
   475→        check(c, d_park >= 256 && d_park < 256 + 4096, "d_park out of range");
   476→        check(c, e.voiceStats(0).parked == 1, "voice not parked");
   477→        const int skip = arrive_from_shift ? static_cast<int>(0.05 * sr) : 0;
   478→        std::vector<float> out(in.size(), 0.f);
   479→        e.process(in.data(), out.data(), in.size());
   480→        // Reference ring semantics: a read at delay d returns the sample
   481→        // pushed d-1 samples earlier, so the parked copy is in[k-d_park+1].
   482→        double worst = 0.0;
   483→        int worst_k = -1;
   484→        // an engine that arrived from a shift has legitimate ring history:
   485→        // bit-exactness is only decidable once the read reaches the current
   486→        // run's own input (k >= d_park-1)
   487→        const int k0 = std::max(skip, d_park - 1);
   488→        for (int k = k0; k < (int)in.size(); ++k) {
   489→            const int src = k - d_park + 1;
   490→            const double expect = (src >= 0) ? static_cast<double>(in[src]) : 0.0;
   491→            const double d = std::abs(static_cast<double>(out[k]) - expect);
   492→            if (d > worst) {
   493→                worst = d;
   494→                worst_k = k;
   495→            }
   496→        }

   500→        // the pre-delay region is exactly zero (fresh engines only -- an
   501→        // engine that arrived from a shift legitimately has ring history)
   502→        if (!arrive_from_shift) {
   503→            double pre_max = 0.0;
   504→            for (int k = 0; k < d_park - 1 && k < (int)in.size(); ++k)

   512→
   513→    unityExact("unity_tone_220_bitexact", gen::tone(220.0, sr, 48000, 0.5));

   570→    // near-unity battery: fresh-start vs via-unity-exit
   571→    auto nearUnity = [&](const std::string& tag, double cents, bool viaUnity) {
   572→        CaseResult c{"near_unity_" + tag + "_" +
   573→                     (viaUnity ? "via_unity" : "fresh")};

   575→        Config cfg;
   576→        cfg.ratio = viaUnity ? 1.0 : ratio;

   585→            e.process(in.data() + sr, out.data() + sr, 2 * sr);

   653→        c.pass = c.note.empty();
   654→        R.push_back(std::move(c));
   655→    }

   699→        addValue(c, "max_abs_diff_vs_exact_mean", worst);
   700→        check(c, worst == 0.0, "8-voice unison mix is not the exact mean");

   702→        R.push_back(std::move(c));
   703→    }

   906→        const uint32_t fsz = 16;

   908→        const uint16_t af = 1, ch = 1, bits = 24;
   909→        const uint32_t rate = sr, br = sr * 3;

   913→        std::memcpy(bytes.data() + 24, &rate, 4);
   914→        std::memcpy(bytes.data() + 28, &br, 4);
   915→        std::memcpy(bytes.data() + 32, &block, 2);
   916→        std::memcpy(bytes.data() + 34, &bits, 2);
   917→        std::memcpy(bytes.data() + 36, "data", 4);
   918→        const uint32_t dsz = 3 * n;
   919→        std::memcpy(bytes.data() + 40, &dsz, 4);
   920→        for (int i = 0; i < n; ++i) {
   921→            bytes[44 + 3 * i] = vals[i] & 0xFF;
   922→            bytes[45 + 3 * i] = (vals[i] >> 8) & 0xFF;
   923→            bytes[46 + 3 * i] = (vals[i] >> 16) & 0xFF;
   924→        }
   925→        FILE* f = std::fopen("/tmp/octavox_24bit.wav", "wb");
   926→        std::fwrite(bytes.data(), 1, bytes.size(), f);
   927→        std::fclose(f);
   928→        int sr2 = 0, ch2 = 0;
   929→        auto x = wav::readWav("/tmp/octavox_24bit.wav", sr2, ch2);
   930→        check(c, sr2 == sr && ch2 == 1 && (int)x.size() == n, "bad header");
   931→        double worst = 0.0;
   932→        for (int i = 0; i < n && i < (int)x.size(); ++i)
   933→            worst = std::max(worst,
   934→                             std::abs((double)x[i] - vals[i] / 8388608.0));
   935→        addValue(c, "max_read_error", worst);
   936→        check(c, worst == 0.0, "24-bit values wrong");
   937→        c.pass = c.note.empty();
   938→        R.push_back(std::move(c));
   939→    }
   940→}

   942→void casesStability(std::vector<CaseResult>& R) {
   943→    const int sr = 48000;
   944→    {  // long run: 60 s noise, 8 voices, feedback
   945→        CaseResult c{"longrun_60s_8voice_feedback"};
   946→        Config cfg;
   947→        cfg.ratio = semitoneRatio(1200.0);
   948→        cfg.voices = 8;
   949→        cfg.feedback = 0.35;
   950→        const auto in = gen::noise(2025, 60 * sr, 0.5);
   951→        const auto out = run(cfg, in);
   952→        check(c, meas::allFinite(out.data(), out.size()), "non-finite");
   953→        const double pk = meas::peak(out.data(), out.size());
   954→        addValue(c, "peak", pk);
   955→        check(c, pk < 4.0, "unstable growth");
   956→        // tail RMS must not exceed head RMS by much (no divergence)
   957→        const double rh = meas::rms(out.data(), 5 * sr);
   958→        const double rt = meas::rms(out.data() + 55 * sr, 5 * sr);
   959→        addValue(c, "head_rms", rh);
   960→        addValue(c, "tail_rms", rt);
   961→        check(c, rt < 4.0 * rh + 1e-9, "diverging tail");
   962→        c.pass = c.note.empty();
   963→        R.push_back(std::move(c));
   964→    }
   965→    {  // 30 s chirp +700c
   966→        CaseResult c{"chirp_30s_+700c"};
   967→        Config cfg;
   968→        cfg.ratio = semitoneRatio(700.0);
   969→        const auto in = gen::chirp(100.0, 1000.0, sr, 30 * sr, 0.5);
   970→        const auto out = run(cfg, in);
   971→        check(c, meas::allFinite(out.data(), out.size()), "non-finite");
   972→        const double pk = meas::peak(out.data(), out.size());
   973→        addValue(c, "peak", pk);
   974→        check(c, pk < 3.0, "peak too high on chirp");
   975→        c.pass = c.note.empty();
   976→        R.push_back(std::move(c));
   977→    }
   978→    {  // impulse latency
   979→        CaseResult c{"impulse_latency"};
   980→        Config cfg;
   981→        cfg.ratio = semitoneRatio(400.0);
   982→        const auto in = gen::impulse(sr);
   983→        const auto out = run(cfg, in);
   984→        int first = -1;
   985→        for (int k = 0; k < (int)out.size(); ++k)
   986→            if (out[k] != 0.0f) {
   987→                first = k;
   988→                break;
   989→            }
   990→        addValue(c, "first_nonzero_sample", first);
   991→        addValue(c, "nominal_latency", cfg.base_delay + cfg.grain * 0.5);
   992→        check(c, first >= 250, "responds before base_delay");
   993→        check(c, first <= 256 + 32 + 4400, "responds too late");
   994→        c.pass = c.note.empty();
   995→        R.push_back(std::move(c));
   996→    }
   997→    {  // unity impulse: first nonzero exactly at d_park
   998→        CaseResult c{"impulse_latency_unity"};
   999→        Config cfg;
  1000→        const auto in = gen::impulse(sr);
  1001→        const auto out = run(cfg, in);
  1002→        AlignedDelayLine e;
  1003→        e.prepare(cfg);
  1004→        const int d = e.voiceStats(0).parked_delay;
  1005→        int first = -1;
  1006→        for (int k = 0; k < (int)out.size(); ++k)
  1007→            if (out[k] != 0.0f) {
  1008→                first = k;
  1009→                break;
  1010→            }
  1011→        addValue(c, "first_nonzero_sample", first);
  1012→        addValue(c, "d_park", d);
  1013→        check(c, first == d - 1, "unity impulse not at d_park-1");
  1014→        c.pass = c.note.empty();
  1015→        R.push_back(std::move(c));
  1016→    }
  1017→    {  // DC input: finite (guards div-by-zero in spacing/median paths)
  1018→        CaseResult c{"dc_input_finite"};
  1019→        Config cfg;
  1020→        cfg.ratio = semitoneRatio(400.0);
  1021→        std::vector<float> in(10 * sr, 1.0f);
  1022→        const auto out = run(cfg, in);
  1023→        check(c, meas::allFinite(out.data(), out.size()), "non-finite on DC");
  1024→        c.pass = c.note.empty();
  1025→        R.push_back(std::move(c));
  1026→    }
  1027→    {  // clipping behaviour: hot input stays finite; WAV writer clips
  1028→        CaseResult c{"clipping_behaviour"};
  1029→        Config cfg;
  1030→        cfg.ratio = semitoneRatio(1200.0);
  1031→        cfg.voices = 8;
  1032→        std::vector<float> in(sr, 0.0f);
  1033→        for (int i = 0; i < sr; ++i)
  1034→            in[i] = (i % 480 < 240) ? 1.0f : -1.0f;  // full-scale square
  1035→        const auto out = run(cfg, in);
  1036→        const double pk = meas::peak(out.data(), out.size());
  1037→        addValue(c, "float_domain_peak", pk);
  1038→        check(c, meas::allFinite(out.data(), out.size()), "non-finite");
  1039→        check(c, pk < 8.0, "float peak beyond sane bound");
  1040→        wav::writeWav("/tmp/octavox_clip.wav", out.data(), out.size(), 1, sr);
  1041→        int sr2 = 0, ch2 = 0;
  1042→        auto back = wav::readWav("/tmp/octavox_clip.wav", sr2, ch2);
  1043→        const double pkq = meas::peak(back.data(), back.size());
  1044→        addValue(c, "wav_peak_after_clip", pkq);
  1045→        check(c, pkq <= 1.0, "WAV writer did not clip-guard");
  1046→        c.pass = c.note.empty();
  1047→        R.push_back(std::move(c));
  1048→    }
  1049→}

  1054→std::string jsonEscape(const std::string& s) {
  1055→    std::string o;
  1056→    for (char ch : s) {
  1057→        if (ch == '"' || ch == '\\') {
  1058→            o += '\\';
  1059→            o += ch;
  1060→        } else if (ch == '\n') {
  1061→            o += "\\n";
  1062→        } else {
  1063→            o += ch;
  1064→        }
  1065→    }
  1066→    return o;
  1067→}

  1069→void writeJson(const std::string& path) {
  1070→    std::string j = "{\n  \"suite\": \"octavox_native_tests\",\n  \"version\": \"";
  1071→    j += versionString();
  1072→    j += "\",\n  \"cases\": [\n";
  1073→    int idx = 0;
  1074→    for (const auto& c : g_results) {
  1075→        j += "    {\"name\": \"" + jsonEscape(c.name) + "\", \"pass\": " +
  1076→             (c.pass ? "true" : "false") + ", \"note\": \"" +
  1077→             jsonEscape(c.note) + "\", \"values\": {";
  1078→        for (size_t i = 0; i < c.values.size(); ++i) {
  1079→            char buf[64];
  1080→            std::snprintf(buf, sizeof(buf), "%.10g", c.values[i].second);
  1081→            j += "\"" + jsonEscape(c.values[i].first) + "\": " + buf;
  1082→            if (i + 1 < c.values.size()) j += ", ";
  1083→        }
  1084→        j += "}}";
  1085→        if (++idx < (int)g_results.size()) j += ",";
  1086→        j += "\n";
  1087→    }
  1088→    int passed = 0;
  1089→    for (const auto& c : g_results) passed += c.pass ? 1 : 0;
  1090→    j += "  ],\n  \"summary\": {\"total\": " + std::to_string(g_results.size()) +
  1091→         ", \"passed\": " + std::to_string(passed) + ", \"failed\": " +
  1092→         std::to_string(g_results.size() - passed) + "}\n}\n";
  1093→    FILE* f = std::fopen(path.c_str(), "wb");
  1094→    if (f) {
  1095→        std::fwrite(j.data(), 1, j.size(), f);
  1096→        std::fclose(f);
  1097→    }
  1098→}
