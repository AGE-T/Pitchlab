    42→/// 64-bit LCG, identical on the Python side. u in [0,1); sample = (2u-1)*amp.
    43→struct Lcg {
    44→    uint64_t s;
    45→    explicit Lcg(uint64_t seed) : s(seed * 6364136223846793005ULL +
    46→                                     1442695040888963407ULL) {}
    47→    double next() {
    48→        s = s * 6364136223846793005ULL + 1442695040888963407ULL;
    49→        const uint64_t m = (s >> 11) & ((1ULL << 53) - 1);
    50→        return static_cast<double>(m) / 9007199254740992.0;
    51→    }
    52→    double noise(double amp) { return (2.0 * next() - 1.0) * amp; }
    53→};

    55→std::vector<float> tone(double f, int sr, int n, double amp) {
    56→    std::vector<float> x(n);
    57→    for (int i = 0; i < n; ++i)
    58→        x[i] = static_cast<float>(amp * std::sin(2.0 * M_PI * f * i / sr));
    59→    return x;
    60→}

    62→std::vector<float> chord(const std::vector<double>& fs, int sr, int n,
    63→                         double amp) {
    64→    std::vector<float> x(n, 0.0f);
    65→    for (double f : fs)
    66→        for (int i = 0; i < n; ++i)
    67→            x[i] += static_cast<float>(amp / fs.size() *
    68→                                       std::sin(2.0 * M_PI * f * i / sr));
    69→    return x;
    70→}

    75→    for (int i = 0; i < n; ++i) x[i] = static_cast<float>(rng.noise(amp));

    85→/// Percussive burst train: exponential decays.
    86→std::vector<float> bursts(int sr, int n, double period_s, double amp) {
    87→    std::vector<float> x(n, 0.0f);
    88→    Lcg rng(99);
    89→    const int per = static_cast<int>(period_s * sr);
    90→    for (int i = 0; i < n; ++i) {
    91→        const int t = i % per;
    92→        if (t < 200)
    93→            x[i] = static_cast<float>(amp * std::exp(-t / 40.0) *
    94→                                      (t == 0 ? 1.0 : rng.noise(1.0)));
    95→    }
    96→    return x;
    97→}

    99→std::vector<float> chirp(double f0, double f1, int sr, int n, double amp) {
   100→    std::vector<float> x(n);
   101→    const double T = static_cast<double>(n) / sr;
   102→    for (int i = 0; i < n; ++i) {
   103→        const double t = static_cast<double>(i) / sr;
   104→        const double f = f0 * std::pow(f1 / f0, t / T);
   105→        x[i] = static_cast<float>(amp * std::sin(2.0 * M_PI * f * t));
   106→    }
   107→    return x;
   108→}

   117→/// Radix-2 complex FFT (in-place, n a power of two).
   118→void fft(double* re, double* im, int n) {
   119→    for (int i = 1, j = 0; i < n; ++i) {
   120→        int bit = n >> 1;
   121→        for (; j & bit; bit >>= 1) j ^= bit;
   122→        j ^= bit;
   123→        if (i < j) {
   124→            std::swap(re[i], re[j]);
   125→            std::swap(im[i], im[j]);
   126→        }
   127→    }
   128→    for (int len = 2; len <= n; len <<= 1) {
   129→        const double ang = -2.0 * M_PI / len;
   130→        const double wr = std::cos(ang), wi = std::sin(ang);
   131→        for (int i = 0; i < n; i += len) {
   132→            double cr = 1.0, ci = 0.0;
   133→            for (int k = 0; k < len / 2; ++k) {
   134→                const double ur = re[i + k], ui = im[i + k];
   135→                const double vr = re[i + k + len / 2] * cr -
   136→                                  im[i + k + len / 2] * ci;
   137→                const double vi = re[i + k + len / 2] * ci +
   138→                                  im[i + k + len / 2] * cr;
   139→                re[i + k] = ur + vr;
   140→                im[i + k] = ui + vi;
   141→                re[i + k + len / 2] = ur - vr;
   142→                im[i + k + len / 2] = ui - vi;
   143→                const double nr = cr * wr - ci * wi;
   144→                ci = cr * wi + ci * wr;
   145→                cr = nr;
   146→            }
   147→        }
   148→    }
   149→}

   151→/// FFT peak pitch with the CORRECTED Hann-exact 3-point estimator.
   152→/// Returns 0.0 when no peak above the threshold is found.
   153→///
   154→/// R15-REB1 fix (recovered from the record): the interpolation now runs on
   155→/// the three bins' MAGNITUDES with the Hann-exact formula
   156→///     delta = 2*(c - a) / (a + 2*b + c)
   157→/// (a, b, c = magnitudes of bins kpeak-1, kpeak, kpeak+1) instead of the
   158→/// old power-domain parabola 0.5*(a-c)/(a-2b+c) on squared magnitudes,
   159→/// which is biased for off-bin tones under a Hann window (the historical
   160→/// round-15 record: the correction made battery cases PASS, none failed).
   161→///
   162→/// FFT size N = the largest power of two <= min(n, 65536): the analysed
   163→/// NEWEST block of N samples always carries a complete Hann window, and
   164→/// the window offset off = n - N is >= 0 BY CONSTRUCTION (the estimator
   165→/// never reads before the supplied buffer start).
   166→double pitchFFT(const float* x, int n, int sr, double fmin, double fmax) {
   167→    int N = 1;
   168→    while (N < n && N < 65536) N <<= 1;  // largest pow2 <= min(n, 65536)
   169→    if (N < 16) return 0.0;
   170→    std::vector<double> re(N, 0.0), im(N, 0.0);
   171→    const int off = n - N;  // use the newest N samples (off >= 0 by construction)
   172→    for (int i = 0; i < N; ++i) {
   173→        const double w = 0.5 - 0.5 * std::cos(2.0 * M_PI * i / N);
   174→        re[i] = static_cast<double>(x[off + i]) * w;
   175→    }
   176→    fft(re.data(), im.data(), N);
   177→    const int kmin = std::max(1, static_cast<int>(fmin * N / sr));
   178→    const int kmax = std::min(N / 2 - 2, static_cast<int>(fmax * N / sr));
   179→    if (kmax <= kmin) return 0.0;
   180→    int kpeak = kmin;
   181→    double best = 0.0;
   182→    for (int k = kmin; k <= kmax; ++k) {
   183→        const double m = re[k] * re[k] + im[k] * im[k];  // power (same argmax)
   184→        if (m > best) {
   185→            best = m;
   186→            kpeak = k;
   187→        }
   188→    }
   189→    if (best <= 0.0) return 0.0;
   190→    // magnitudes of the three bins around the peak (Hann-exact estimator)
   191→    const double a = std::sqrt(re[kpeak - 1] * re[kpeak - 1] +
   192→                               im[kpeak - 1] * im[kpeak - 1]);
   193→    const double b = std::sqrt(best);
   194→    const double c = std::sqrt(re[kpeak + 1] * re[kpeak + 1] +
   195→                               im[kpeak + 1] * im[kpeak + 1]);
   196→    double d = 0.0;
   197→    const double den = a + 2.0 * b + c;
   198→    if (den > 1e-300) d = 2.0 * (c - a) / den;
   199→    if (d > 0.5) d = 0.5;
   200→    if (d < -0.5) d = -0.5;
   201→    return static_cast<double>(sr) * (kpeak + d) / N;
   202→}

   204→double rms(const float* x, int n) {
   205→    double s = 0.0;
   206→    for (int i = 0; i < n; ++i) s += static_cast<double>(x[i]) * x[i];
   207→    return n ? std::sqrt(s / n) : 0.0;
   208→}
   209→
   210→double peak(const float* x, int n) {
   211→    double p = 0.0;
   212→    for (int i = 0; i < n; ++i) p = std::max(p, std::abs(static_cast<double>(x[i])));
   213→    return p;
   214→}

   216→bool allFinite(const float* x, int n) {
   217→    for (int i = 0; i < n; ++i)
   218→        if (!std::isfinite(x[i])) return false;
   219→    return true;
   220→}

   222→/// Envelope modulation in dB: max/min of RMS windows of `w` samples over the
   223→/// steady region (skip the first skip samples). w = sr/1000 gives the 1 ms
   224→/// metric; w = period gives the period-synchronous metric (the r11 lesson:
   225→/// a 1 ms window phase-breathes several dB on a plain tone).
   226→double envModulationDb(const float* x, int n, int sr, int skip, int w) {
   227→    std::vector<double> env;
   228→    for (int i0 = skip; i0 + w <= n; i0 += w) {
   229→        env.push_back(rms(x + i0, w));
   230→    }
   231→    if (env.empty()) return 0.0;
   232→    double lo = 1e300, hi = 0.0;
   233→    for (double v : env) {
   234→        if (v < lo) lo = v;
   235→        if (v > hi) hi = v;
   236→    }
   237→    if (lo <= 0.0) return hi > 0.0 ? 999.0 : 0.0;
   238→    return 20.0 * std::log10(hi / lo);
   239→}
   240→
   241→double centsBetween(double fMeasured, double fNominal) {
   242→    if (fMeasured <= 0.0 || fNominal <= 0.0) return 999.0;
   243→    return 1200.0 * std::log2(fMeasured / fNominal);
   244→}
   245→
   246→/// Max absolute sample-to-sample difference (transition click meter).
   247→double maxStep(const float* x, int n, int i0 = 0) {
   248→    double m = 0.0;
   249→    for (int i = std::max(1, i0 + 1); i < n; ++i)
   250→        m = std::max(m, std::abs(static_cast<double>(x[i]) - x[i - 1]));
   251→    return m;
   252→}

   279→// Runs the engine over a full signal in one go (64-aligned internally is
   280→// irrelevant -- process takes the whole block).
   281→std::vector<float> run(const Config& cfg, const std::vector<float>& in,
   282→                       int block = 1 << 20) {
   283→    AlignedDelayLine e;
   284→    if (!e.prepare(cfg)) {
   285→        return {};
   286→    }
   287→    std::vector<float> out(in.size(), 0.0f);
   288→    int done = 0;
   289→    while (done < static_cast<int>(in.size())) {
   290→        const int n = std::min(block, static_cast<int>(in.size()) - done);
   291→        e.process(in.data() + done, out.data() + done, n);
   292→        done += n;
   293→    }
   294→    return out;
   295→}

   300→void casesPrepareReset(std::vector<CaseResult>& R) {
   301→    {  // 1. default prepare
   302→        CaseResult c{"prepare_default"};
   303→        Config cfg;
   304→        AlignedDelayLine e;
   305→        const bool ok = e.prepare(cfg);
   306→        check(c, ok, "prepare failed");
   307→        check(c, e.latencySamples() == 2304, "latency != 2304");
   308→        check(c, e.ready(), "not ready");
   309→        c.pass = c.note.empty();
   310→        R.push_back(std::move(c));
   311→    }
   312→    {  // 2. invalid configs rejected
   313→        CaseResult c{"prepare_invalid_rejected"};
   314→        int rejected = 0, total = 0;
   315→        Config bad;
   316→        auto tryCfg = [&](const Config& cfg) {
   317→            ++total;
   318→            AlignedDelayLine e;
   319→            if (!e.prepare(cfg)) ++rejected;
   320→        };
   321→        bad = Config();
   322→        bad.sample_rate = 0.0;
   323→        tryCfg(bad);
   324→        bad = Config();
   325→        bad.voices = 9;
   326→        tryCfg(bad);
   327→        bad = Config();
   328→        bad.grain = 0.0;
   329→        tryCfg(bad);
   330→        bad = Config();
   331→        bad.feedback = 1.0;
   332→        tryCfg(bad);
   333→        bad = Config();
   334→        bad.base_delay = 0.0;
   335→        tryCfg(bad);
   336→        addValue(c, "rejected", rejected);
   337→        addValue(c, "total", total);
   338→        check(c, rejected == total, "some invalid configs accepted");
   339→        c.pass = c.note.empty();
   340→        R.push_back(std::move(c));
   341→    }
   342→    {  // 3. reset determinism
   343→        CaseResult c{"reset_determinism"};
   344→        Config cfg;
   345→        cfg.ratio = semitoneRatio(400.0);
   346→        const auto in = gen::tone(220.0, 48000, 48000, 0.5);
   347→        AlignedDelayLine a, b;
   348→        a.prepare(cfg);
   349→        b.prepare(cfg);
   350→        std::vector<float> oa(in.size(), 0.f), ob(in.size(), 0.f),
   351→            ob2(in.size(), 0.f);
   352→        a.process(in.data(), oa.data(), in.size());
   353→        b.process(in.data(), ob.data(), in.size());
   354→        b.reset();
   355→        b.process(in.data(), ob2.data(), in.size());
   356→        double d1 = 0.0, d2 = 0.0;
   357→        for (size_t i = 0; i < in.size(); ++i) {
   358→            d1 = std::max(d1, std::abs((double)oa[i] - ob[i]));
   359→            d2 = std::max(d2, std::abs((double)ob[i] - ob2[i]));
   360→        }
   361→        addValue(c, "fresh_vs_fresh", d1);
   362→        addValue(c, "reset_vs_fresh", d2);
   363→        check(c, d1 == 0.0, "two fresh engines differ");
   364→        check(c, d2 == 0.0, "reset engine differs from fresh");
   365→        c.pass = c.note.empty();
   366→        R.push_back(std::move(c));
   367→    }
   368→    {  // 4. repeated prepare
   369→        CaseResult c{"repeated_prepare"};
   370→        Config cfg;
   371→        cfg.ratio = semitoneRatio(500.0);
   372→        const auto in = gen::tone(330.0, 48000, 24000, 0.4);

   381→        b.process(in.data(), oc.data(), in.size());

   383→        for (size_t i = 0; i < in.size(); ++i)

   389→    }

   391→        CaseResult c{"block_size_invariance"};
   392→        Config cfg;
   393→        cfg.ratio = semitoneRatio(700.0);
   394→        const auto in = gen::tone(196.0, 48000, 48000, 0.4);

   397→        double worst = 0.0;
   398→        for (int s : splits) {
   399→            const auto got = run(cfg, in, s);
   400→            double d = 0.0;
   401→            for (size_t i = 0; i < in.size(); ++i)
   402→                d = std::max(d, std::abs((double)ref[i] - got[i]));

   410→            size_t i = 0;
   411→            int pat[] = {1, 64, 7, 512, 16, 4096, 3};
   412→            int pi = 0;
   413→            while (i < in.size()) {
   414→                int n = std::min(pat[pi % 7], (int)(in.size() - i));
   415→                e.process(in.data() + i, got.data() + i, n);
   416→                i += n;
   417→                ++pi;
   418→            }
   419→            double d = 0.0;
   420→            for (size_t k = 0; k < in.size(); ++k)
   421→                d = std::max(d, std::abs((double)ref[k] - got[k]));
   422→            worst = std::max(worst, d);
   423→        }
   424→        addValue(c, "worst_split_diff", worst);
   425→        check(c, worst == 0.0, "block split changed the output");

   427→        R.push_back(std::move(c));
   428→    }
   429→    {  // 6. steady-state allocation-free
   430→        CaseResult c{"steady_state_no_allocation"};
   431→        Config cfg;
   432→        cfg.ratio = semitoneRatio(400.0);
   433→        cfg.voices = 8;
   434→        cfg.feedback = 0.35;
   435→        const auto in = gen::noise(1234, 48000, 0.5);
   436→        AlignedDelayLine e;

   438→        std::vector<float> out(in.size(), 0.f);
   439→        e.process(in.data(), out.data(), 24000);  // warm-up (tables, paths)
   440→        allocs::reset();

   442→        e.process(in.data() + 24000, out.data() + 24000,
   443→                  static_cast<int>(in.size()) - 24000);
   444→        allocs::arm(false);
   445→        const long n = allocs::count();
   446→        addValue(c, "allocations_during_process", static_cast<double>(n));
   447→        check(c, n == 0, "allocations during steady-state process()");
   448→        c.pass = c.note.empty();
   449→        R.push_back(std::move(c));
   450→    }

   453→void casesUnity(std::vector<CaseResult>& R) {
   454→    const int sr = 48000;
   455→
   456→    // bit-exact delayed-copy gate used by the unity battery
   457→    auto unityExact = [&](const std::string& name, const std::vector<float>& in,
   458→                          bool arrive_from_shift = false) {
   459→        CaseResult c{name};
   460→        Config cfg;  // ratio exactly 1.0, 1 voice, neutral
   461→        AlignedDelayLine e;
   462→        e.prepare(cfg);
   463→        if (arrive_from_shift) {
   464→            // arrive from +4 st first: 1 s shifted, then unity, then settle
   465→            e.setRatio(semitoneRatio(400.0));
   466→            std::vector<float> pre(sr, 0.f);
   467→            e.process(in.data(), pre.data(), sr);
   468→            e.setRatio(1.0);
   469→            std::vector<float> mid(static_cast<int>(0.3 * sr), 0.f);
   470→            if ((int)in.size() > sr)
   471→                e.process(in.data() + sr, mid.data(),
   472→                           std::min((int)in.size() - sr, (int)mid.size()));
   473→        }
   474→        const int d_park = e.voiceStats(0).parked_delay;
   475→        addValue(c, "d_park", d_park);
   476→        check(c, d_park >= 256 && d_park < 256 + 4096, "d_park out of range");
   477→        check(c, e.voiceStats(0).parked == 1, "voice not parked");
   478→        const int skip = arrive_from_shift ? static_cast<int>(0.05 * sr) : 0;
   479→        std::vector<float> out(in.size(), 0.f);
   480→        e.process(in.data(), out.data(), in.size());
   481→        // Reference ring semantics: a read at delay d returns the sample
   482→        // pushed d-1 samples earlier, so the parked copy is in[k-d_park+1].
   483→        double worst = 0.0;
   484→        int worst_k = -1;
   485→        // an engine that arrived from a shift has legitimate ring history:
   486→        // bit-exactness is only decidable once the read reaches the current
   487→        // run's own input (k >= d_park-1)
   488→        const int k0 = std::max(skip, d_park - 1);
   489→        for (int k = k0; k < (int)in.size(); ++k) {
   490→            const int src = k - d_park + 1;
   491→            const double expect = (src >= 0) ? static_cast<double>(in[src]) : 0.0;
   492→            const double d = std::abs(static_cast<double>(out[k]) - expect);
   493→            if (d > worst) {
   494→                worst = d;
   495→                worst_k = k;
   496→            }
