    23→// Musical spread table (reference parity: poc/octavox_offline.py)
    24→// ---------------------------------------------------------------------------
    25→constexpr double kSpreadSemitones[8] = {0.0, 7.0, 4.0, 12.0, -5.0, -12.0, 3.0, -3.0};

    27→// ---------------------------------------------------------------------------
    28→// Interpolator tables (built once, shared, immutable)
    29→// ---------------------------------------------------------------------------
    30→constexpr int kLag6Taps = 6;
    31→
    32→// 6-point Lagrange weights for phase mu, nodes -2,-1,0,1,2,3 (our own design;
    33→// same formula as the Python reference).
    34→void lagrange6Weights(double mu, double* w) {
    35→    static const double xs[6] = {-2.0, -1.0, 0.0, 1.0, 2.0, 3.0};
    36→    for (int i = 0; i < 6; ++i) {
    37→        double v = 1.0;
    38→        for (int j = 0; j < 6; ++j) {
    39→            if (i != j) v *= (mu - xs[j]) / (xs[i] - xs[j]);
    40→        }
    41→        w[i] = v;
    42→    }
    43→}
    44→
    45→struct InterpTables {
    46→    double lag2048[2048][6];
    47→    double lag256[256][6];
    48→
    49→    InterpTables() {
    50→        double w[6];
    51→        for (int r = 0; r < 2048; ++r) {
    52→            lagrange6Weights((r + 0.5) / 2048.0, w);  // bin centres (Python parity)
    53→            for (int k = 0; k < 6; ++k) lag2048[r][k] = w[k];
    54→        }
    55→        for (int r = 0; r < 256; ++r) {
    56→            lagrange6Weights((r + 0.5) / 256.0, w);
    57→            for (int k = 0; k < 6; ++k) lag256[r][k] = w[k];
    58→        }
    59→    }
    60→};
    61→
    62→const InterpTables& interpTables() {
    63→    static const InterpTables t;  // thread-safe magic static
    64→    return t;
    65→}

    67→// ---------------------------------------------------------------------------
    68→// YIN-style pitch detector (port of poc/octavox_poc.py PitchDetector)
    69→// Window 2048, fmin 60, fmax 1200 (the v2 engine's detector defaults); the
    70→// spacing gate in the engine uses the Config fmin/fmax (55/1200) exactly as
    71→// the Python does.
    72→// ---------------------------------------------------------------------------
    73→class PitchDetector {
    74→public:
    75→    static constexpr int kWin = 2048;
    76→
    77→    void configure(int sr) {
    78→        sr_ = sr;
    79→        tau_min_ = std::max(2, static_cast<int>(sr_ / 1200.0));
    80→        tau_max_ = std::min(kWin - 2, static_cast<int>(sr_ / 60.0));
    81→        buf_.assign(kWin, 0.0);
    82→        d_.assign(std::max(0, tau_max_ - tau_min_), 0.0);
    83→        cmnd_.assign(d_.size(), 0.0);
    84→        last_f0_ = 0.0;
    85→    }
    86→
    87→    void reset() {
    88→        std::fill(buf_.begin(), buf_.end(), 0.0);
    89→        last_f0_ = 0.0;
    90→    }
    91→
    92→    // Push a block into the sliding window (Python push_block semantics).
    93→    void pushBlock(const float* x, int n) {
    94→        if (n >= kWin) {
    95→            std::copy(x + n - kWin, x + n, buf_.begin());
    96→            return;
    97→        }
    98→        // roll left by n, append
    99→        std::copy(buf_.begin() + n, buf_.end(), buf_.begin());
   100→        for (int i = 0; i < n; ++i) buf_[kWin - n + i] = static_cast<double>(x[i]);
   101→    }
   102→
   103→    double estimate() {
   104→        const std::vector<double>& x = buf_;
   105→        double peak = 0.0;
   106→        for (int i = 0; i < kWin; ++i) peak = std::max(peak, std::abs(x[i]));
   107→        if (peak < 1e-6) {
   108→            last_f0_ = 0.0;
   109→            return 0.0;
   110→        }
   111→        if (tau_max_ <= tau_min_) {
   112→            last_f0_ = 0.0;
   113→            return 0.0;
   114→        }
   115→        // seg = second half of the window; m = len(seg) - tau_max
   116→        const int seg0 = kWin / 2;
   117→        const int m = (kWin - seg0) - tau_max_;
   118→        if (m < 32) {
   119→            last_f0_ = 0.0;
   120→            return 0.0;
   121→        }
   122→        const int ntau = tau_max_ - tau_min_;
   123→        for (int i = 0; i < ntau; ++i) {
   124→            const int tau = tau_min_ + i;
   125→            double s = 0.0;
   126→            for (int j = 0; j < m; ++j) {
   127→                const double a = x[seg0 + j] - x[seg0 + tau + j];
   128→                s += a * a;
   129→            }
   130→            d_[i] = s;
   131→        }
   132→        // cumulative mean normalisation
   133→        double run = 0.0;
   134→        for (int i = 0; i < ntau; ++i) {
   135→            run += d_[i];
   136→            cmnd_[i] = (run > 0.0) ? d_[i] * static_cast<double>(i + 1) / run : 1.0;
   137→        }
   138→        // first minimum below threshold
   139→        int idx = -1;
   140→        const double thr = 0.20;
   141→        for (int i = 1; i < ntau - 1; ++i) {
   142→            if (cmnd_[i] < thr && cmnd_[i] <= cmnd_[i - 1] && cmnd_[i] <= cmnd_[i + 1]) {
   143→                idx = i;
   144→                break;
   145→            }
   146→        }
   147→        if (idx < 0) {
   148→            // argmin
   149→            idx = 0;
   150→            for (int i = 1; i < ntau; ++i)
   151→                if (cmnd_[i] < cmnd_[idx]) idx = i;
   152→        }
   153→        double tau = static_cast<double>(tau_min_ + idx);
   154→        // parabolic refinement
   155→        if (idx > 0 && idx < ntau - 1) {
   156→            const double y0 = cmnd_[idx - 1], y1 = cmnd_[idx], y2 = cmnd_[idx + 1];
   157→            const double den = y0 - 2.0 * y1 + y2;
   158→            if (std::abs(den) > 1e-15) tau += 0.5 * (y0 - y2) / den;
   159→        }
   160→        double f0 = static_cast<double>(sr_) / tau;
   161→        if (!(60.0 <= f0 && f0 <= 1200.0)) f0 = 0.0;
   162→        last_f0_ = f0;
   163→        return f0;
   164→    }
   165→
   166→    double lastF0() const { return last_f0_; }
   167→
   168→private:
   169→    int sr_ = 48000;
   170→    int tau_min_ = 2, tau_max_ = 800;
   171→    std::vector<double> buf_, d_, cmnd_;
   172→    double last_f0_ = 0.0;
   173→};

   179→    double ratio_t = 1.0;   // target ratio (2^(st/12))
   180→    double ratio = 1.0;     // gliding ratio
   181→    double base_delay = 256.0;
   182→    double frac = 0.5;      // grain phase [0,1)
   183→    double S = 4096.0;      // current tap spacing
   184→    // unity engagement
   185→    double e = 1.0;         // 1 = dual-tap, 0 = single-tap A
   186→    double a = 0.0;         // parking blend: 1 = parked integer read
   187→    int d_park = -1;        // parked integer delay
   188→    // diagnostics
   189→    long wraps = 0;
   190→    long commits = 0;

   195→// ---------------------------------------------------------------------------
   196→// AlignedDelayLine::Impl
   197→// ---------------------------------------------------------------------------
   198→struct AlignedDelayLine::Impl {
   199→    static constexpr int kRingBits = 18;
   200→    static constexpr int kRingSize = 1 << kRingBits;  // 262144
   201→    static constexpr int kRingMask = kRingSize - 1;
   202→
   203→    std::vector<double> ring;        // age-indexed delay line
   204→    int w = 0;                       // newest sample index
   205→    PitchDetector det;
   206→    std::vector<VoiceState> vs;      // [max_voices]
   207→    std::vector<double> f0_hist;     // circular, size f0_history
   208→    std::vector<double> med_scratch; // preallocated median scratch (no alloc)
   209→    std::vector<float> det_pending;  // control-rate accumulator (raw input)
   210→    int det_n = 0;
   211→    int f0_hist_n = 0, f0_hist_i = 0;
   212→    bool have_estimate = false;
   213→    double S_target = 4096.0;
   214→    double fb_state = 0.0;
   215→    double norm = 1.0;               // voice-count normalisation (ramped)
   216→    double norm_target = 1.0;
   217→    double last_f0 = 0.0;
   218→    double e_step = 0.0;             // engagement slew per sample
   219→    Config cfg{};
   220→
   221→    // -- ring helpers -------------------------------------------------------
   222→    inline void push(double x) {
   223→        ring[w] = x;
   224→        w = (w - 1) & kRingMask;
   225→    }
   226→
   227→    inline double readRaw(int delay) const {
   228→        return ring[(w + delay) & kRingMask];
   229→    }
   230→
   231→    // Interpolated read at `delay` (>= 0). Exact-integer delays take the
   232→    // direct sample path (bit-exact; used by the parked unity state).
   233→    double readInterp(double delay) const {
   234→        if (delay < 0.0) delay = 0.0;
   235→        const double d0 = std::floor(delay);
   236→        const double mu = delay - d0;
   237→        const int i0 = static_cast<int>(d0);
   238→        if (mu == 0.0) {
   239→            return ring[(w + i0) & kRingMask];  // exact integer delay
   240→        }
   241→        switch (cfg.interpolation) {
   242→            case Interpolation::Linear: {
   243→                const double a = ring[(w + i0) & kRingMask];
   244→                const double b = ring[(w + i0 + 1) & kRingMask];
   245→                return a + (b - a) * mu;
   246→            }
   247→            case Interpolation::Lagrange6: {
   248→                int ph = static_cast<int>(mu * 2048.0);  // floor, mu < 1
   249→                if (ph > 2047) ph = 2047;
   250→                const double* wv = interpTables().lag2048[ph];
   251→                double s = 0.0;
   252→                for (int k = 0; k < 6; ++k)
   253→                    s += wv[k] * ring[(w + i0 + k - 2) & kRingMask];
   254→                return s;
   255→            }
   256→            case Interpolation::Lagrange6Q: {
   257→                int ph = static_cast<int>(mu * 256.0);
   258→                if (ph > 255) ph = 255;
   259→                const double* wv = interpTables().lag256[ph];
   260→                double s = 0.0;
   261→                for (int k = 0; k < 6; ++k)
   262→                    s += wv[k] * ring[(w + i0 + k - 2) & kRingMask];
   263→                return s;
   264→            }
   265→            case Interpolation::ObsKernel6: {
   266→                // OBS data mode: floor-indexed 256 phases (target phase law).
   267→                int ph = static_cast<int>(mu * 256.0);
   268→                if (ph > 255) ph = 255;
   269→                const float* wv = obs_kernel::kTable[ph];
   270→                double s = 0.0;
   271→                for (int k = 0; k < 6; ++k)
   272→                    s += static_cast<double>(wv[k]) *
   273→                         ring[(w + i0 + k - 2) & kRingMask];
   274→                return s;
   275→            }
   276→        }
   277→        return 0.0;
   278→    }
   279→
   280→    // -- spacing management (v2 parity) --------------------------------------
   281→    void updateSpacingTarget(double f0) {
   282→        if (f0 != 0.0 && cfg.fmin <= f0 && f0 <= cfg.fmax) {
   283→            f0_hist[f0_hist_i] = static_cast<double>(cfg.sample_rate) / f0;
   284→            f0_hist_i = (f0_hist_i + 1) % static_cast<int>(f0_hist.size());
   285→            if (f0_hist_n < static_cast<int>(f0_hist.size())) ++f0_hist_n;
   286→            have_estimate = true;
   287→        }
   288→        if (f0_hist_n == 0) {
   289→            S_target = cfg.grain;
   290→            return;
   291→        }
   292→        // median of the history (np.median parity: even count averages the
   293→        // middle two). Uses the preallocated scratch -- NO allocation here
   294→        // (steady-state guarantee).
   295→        med_scratch.assign(f0_hist.begin(), f0_hist.begin() + f0_hist_n);
   296→        std::sort(med_scratch.begin(), med_scratch.end());
   297→        const int n = static_cast<int>(med_scratch.size());
   298→        double t_hat = (n % 2 == 1)
   299→                           ? med_scratch[n / 2]
   300→                           : 0.5 * (med_scratch[n / 2 - 1] + med_scratch[n / 2]);
   301→        // Python round() = round-half-to-even = nearbyint with default env
   302→        const double k = std::nearbyint(cfg.grain / t_hat);
   303→        const double kk = std::max(1.0, k);
   304→        const double S = kk * t_hat;
   305→        S_target = std::min(std::max(S, 0.5 * cfg.grain), 1.5 * cfg.grain);
   306→    }
   307→
   308→    // Wrap-time commit (v2 parity, hysteresis gated).
   309→    inline void commitSpacing(VoiceState& v) {
   310→        if (!have_estimate) return;
   311→        if (std::abs(S_target / v.S - 1.0) > cfg.s_hysteresis) {
   312→            v.S = S_target;
   313→            ++v.commits;
   314→        }
   315→    }
   316→
   317→    // Unity-exit commit: refresh S while B's weight is ~0, preserving the
   318→    // read point exactly (frac_new = frac*S_old/S_new in [0,1)).
   319→    void exitCommitSpacing(VoiceState& v) {
   320→        if (!have_estimate) return;
   321→        if (std::abs(S_target / v.S - 1.0) <= cfg.s_hysteresis) return;
   322→        const double frac_new = v.frac * v.S / S_target;
   323→        if (frac_new >= 0.0 && frac_new < 1.0) {
   324→            v.S = S_target;
   325→            v.frac = frac_new;
   326→            ++v.commits;
   327→        }
   328→        // else: keep S; the next natural wrap will commit it (v2 rule)
   329→    }
   330→
   331→    // Park on the nearest integer delay inside [base, base+S).
   332→    void parkVoice(VoiceState& v) {
   333→        const double d = v.base_delay + v.frac * v.S;
   334→        int d_int = static_cast<int>(std::floor(d + 0.5));
   335→        const double lo = std::ceil(v.base_delay);
   336→        const double hi = std::floor(v.base_delay + v.S) - 1.0;
   337→        if (static_cast<double>(d_int) > hi) d_int = static_cast<int>(hi);
   338→        if (static_cast<double>(d_int) < lo) d_int = static_cast<int>(lo);
   339→        v.d_park = d_int;
   340→    }
   341→
   342→    // -- per-sample voice processing -----------------------------------------
   343→    // Returns this voice's output sample.
   344→    inline double processVoiceSample(VoiceState& v) {
   345→        // ratio glide (geometric, reference parity)
   346→        v.ratio = v.ratio_t + (v.ratio - v.ratio_t) * Config::kGlideMult;
   347→        const double r = v.ratio;
   348→
   349→        // grain phase ramp + wrap/commit (v2 parity)
   350→        double frac = v.frac + (1.0 - r) / v.S;
   351→        if (r > 1.0) {
   352→            if (frac < 0.0) {
   353→                const double S_old = v.S;
   354→                commitSpacing(v);
   355→                frac = (frac + 1.0) * S_old / v.S;
   356→                while (frac >= 1.0) frac -= 1.0;
   357→                ++v.wraps;
   358→            }
   359→        } else {
   360→            if (frac >= 1.0) {
   361→                const double S_old = v.S;
   362→                commitSpacing(v);
   363→                frac = (frac - 1.0) * S_old / v.S;
   364→                while (frac < 0.0) frac += 1.0;
   365→                ++v.wraps;
   366→            }
   367→        }
   368→        v.frac = frac;
   369→
   370→        const double e = v.e;
   371→        const double a = v.a;
   372→
   373→        if (e <= 0.0 && a >= 1.0) {
   374→            // Fully parked: bit-exact single integer-tap read.
   375→            return readRaw(v.d_park);
   376→        }
   377→
   378→        const double d = v.base_delay + frac * v.S;
   379→        const double A = readInterp(d);
   380→
   381→        double dual;
   382→        if (e >= 1.0) {
   383→            const double B = readInterp(d + v.S);
   384→            const double w = 0.5 - 0.5 * std::cos(M_PI * frac);
   385→            dual = w * A + (1.0 - w) * B;
   386→        } else if (e <= 0.0) {
   387→            dual = A;  // single-tap A (wA_eff = 1, wB_eff = 0)
   388→        } else {
   389→            const double B = readInterp(d + v.S);
   390→            const double w = 0.5 - 0.5 * std::cos(M_PI * frac);
   391→            const double wA = e * w + (1.0 - e);
   392→            const double wB = e * (1.0 - w);
   393→            dual = wA * A + wB * B;
   394→        }
   395→
   396→        if (a > 0.0) {
   397→            const double park = readInterp(static_cast<double>(v.d_park));
   398→            return (1.0 - a) * dual + a * park;
   399→        }
   400→        return dual;
   401→    }
   402→
   403→    // -- engagement slew ------------------------------------------------------
   404→    inline void slewEngagement(VoiceState& v) {
   405→        const bool unity = (v.ratio_t == 1.0);  // bit-exact double compare
   406→        const double e_target = unity ? 0.0 : 1.0;
   407→        // 'a' parks only once e has fully reached 0
   408→        const double a_target = (unity && v.e <= 0.0) ? 1.0 : 0.0;
   409→        if (v.e < e_target) v.e = std::min(e_target, v.e + e_step);
   410→        else if (v.e > e_target) v.e = std::max(e_target, v.e - e_step);
   411→        if (v.a < a_target) {
   412→            if (v.a <= 0.0) parkVoice(v);  // (re)compute the parked delay
   413→            v.a = std::min(a_target, v.a + e_step);
   414→        } else if (v.a > a_target) {
   415→            v.a = std::max(a_target, v.a - e_step);
   416→        }
   417→    }
   418→};

   426→bool AlignedDelayLine::prepare(const Config& cfg) {
   427→    delete impl_;
   428→    impl_ = nullptr;
   429→    ready_ = false;
   430→    if (!(cfg.sample_rate > 0.0)) return false;
   431→    if (cfg.max_voices < 1 || cfg.max_voices > 64) return false;
   432→    if (cfg.voices < 1 || cfg.voices > cfg.max_voices) return false;
   433→    if (!(cfg.grain > 0.0)) return false;
   434→    if (cfg.base_delay < 8.0) return false;
   435→    // Maximum read delay: B tap at d + S with d < base + 32*(mv-1) + S and
   436→    // S <= 1.5*grain, plus 3 extra taps beyond the anchor -> require the whole
   437→    // range to stay inside the 2^18 ring.
   438→    if (cfg.base_delay + 32.0 * (cfg.max_voices - 1) + 3.0 * cfg.grain + 8.0 >=
   439→        static_cast<double>(1 << 18)) {
   440→        return false;
   441→    }
   442→    if (cfg.fmin <= 0.0 || cfg.fmax <= cfg.fmin) return false;
   443→    if (cfg.s_hysteresis < 0.0) return false;
   444→    if (cfg.f0_history < 1 || cfg.f0_history > 64) return false;
   445→    if (cfg.detection_interval < 1) return false;
   446→    if (cfg.engagement_seconds < 0.0) return false;
   447→    if (!(cfg.feedback > -1.0 && cfg.feedback < 1.0)) return false;
   448→
   449→    impl_ = new Impl();
   450→    impl_->cfg = cfg;
   451→    impl_->ring.assign(Impl::kRingSize, 0.0);
   452→    impl_->vs.resize(cfg.max_voices);
   453→    impl_->f0_hist.assign(cfg.f0_history, 0.0);
   454→    impl_->med_scratch.assign(cfg.f0_history, 0.0);
   455→    impl_->det_pending.assign(cfg.detection_interval, 0.0f);
   456→    impl_->det_n = 0;
   457→    impl_->det.configure(static_cast<int>(cfg.sample_rate));
   458→    impl_->e_step =
   459→        cfg.engagement_seconds > 0.0
   460→            ? 1.0 / (cfg.engagement_seconds * cfg.sample_rate)
   461→            : 1.0;
   462→    cfg_ = cfg;
   463→    active_voices_ = cfg.voices;
   464→    ready_ = true;
   465→    reset();
   466→    return true;
   467→}

   469→void AlignedDelayLine::reset() {
   470→    if (!impl_) return;
   471→    Impl& m = *impl_;
   472→    std::fill(m.ring.begin(), m.ring.end(), 0.0);
   473→    m.w = 0;
   474→    m.det.reset();
   475→    const double st = (m.cfg.ratio > 0.0) ? 12.0 * std::log2(m.cfg.ratio) : 0.0;
   476→    for (int i = 0; i < m.cfg.max_voices; ++i) {
   477→        VoiceState& v = m.vs[i];
   478→        const double off = (m.cfg.spread && i < 8) ? kSpreadSemitones[i] : 0.0;
   479→        v.ratio_t = std::pow(2.0, (st + off) / 12.0);
   480→        v.ratio = v.ratio_t;
   481→        v.base_delay = m.cfg.base_delay + 32.0 * i;
   482→        v.frac = 0.5;
   483→        v.S = m.cfg.grain;
   484→        v.wraps = 0;
   485→        v.commits = 0;
   486→        // Unity engagement: start parked if this voice is at exact unity.
   487→        if (v.ratio_t == 1.0) {
   488→            v.e = 0.0;
   489→            v.a = 1.0;
   490→            m.parkVoice(v);
   491→        } else {
   492→            v.e = 1.0;
   493→            v.a = 0.0;
   494→            v.d_park = -1;
   495→        }
   496→    }
   497→    std::fill(m.f0_hist.begin(), m.f0_hist.end(), 0.0);
   498→    m.f0_hist_n = 0;
   499→    m.f0_hist_i = 0;
   500→    m.have_estimate = false;
   501→    m.S_target = m.cfg.grain;
   502→    m.fb_state = 0.0;
   503→    m.norm = 1.0 / active_voices_;
   504→    m.norm_target = m.norm;
   505→    m.det_n = 0;
   506→    m.last_f0 = 0.0;
   507→}
   508→
   509→void AlignedDelayLine::setRatio(double ratio) {
   510→    if (!impl_ || !(ratio > 0.0)) return;
   511→    Impl& m = *impl_;
   512→    const double st = 12.0 * std::log2(ratio);
   513→    for (int i = 0; i < m.cfg.max_voices; ++i) {
   514→        VoiceState& v = m.vs[i];
   515→        const double off = (m.cfg.spread && i < 8) ? kSpreadSemitones[i] : 0.0;
   516→        const double new_t = std::pow(2.0, (st + off) / 12.0);
   517→        const bool was_unity = (v.ratio_t == 1.0);
   518→        v.ratio_t = new_t;
   519→        if (was_unity && new_t != 1.0) {
   520→            // Unity exit: refresh the spacing while tap B carries ~0 weight.
   521→            m.exitCommitSpacing(v);
   522→        }
   523→    }
   524→    m.cfg.ratio = ratio;
   525→    cfg_.ratio = ratio;
   526→}
   527→
   528→void AlignedDelayLine::setVoices(int n) {
   529→    if (!impl_) return;
   530→    if (n < 1 || n > impl_->cfg.max_voices) return;
   531→    active_voices_ = n;
   532→    impl_->norm_target = 1.0 / n;
   533→}
   534→
   535→void AlignedDelayLine::setSpread(bool on) {
   536→    if (!impl_) return;
   537→    if (impl_->cfg.spread == on) return;
   538→    impl_->cfg.spread = on;
   539→    cfg_.spread = on;
   540→    // Re-derive voice targets from the current ratio (audio-thread update).
   541→    const double st = 12.0 * std::log2(impl_->cfg.ratio > 0.0 ? impl_->cfg.ratio : 1.0);
   542→    for (int i = 0; i < impl_->cfg.max_voices; ++i) {
   543→        VoiceState& v = impl_->vs[i];
   544→        const double off = (on && i < 8) ? kSpreadSemitones[i] : 0.0;
   545→        const double new_t = std::pow(2.0, (st + off) / 12.0);
   546→        const bool was_unity = (v.ratio_t == 1.0);
   547→        v.ratio_t = new_t;
   548→        if (was_unity && new_t != 1.0) impl_->exitCommitSpacing(v);
   549→    }
   550→}
   551→
   552→void AlignedDelayLine::setFeedback(double fb) {
   553→    if (!impl_) return;
   554→    impl_->cfg.feedback = std::min(0.999999, std::max(-0.999999, fb));
   555→    cfg_.feedback = impl_->cfg.feedback;
   556→}

   558→int AlignedDelayLine::latencySamples() const {
   559→    if (!impl_) return 0;
   560→    return static_cast<int>(impl_->cfg.base_delay + impl_->cfg.grain * 0.5);
   561→}
   562→
   563→void AlignedDelayLine::process(const float* in, float* out, int n) {
   564→    if (!impl_ || n <= 0) return;
   565→    Impl& m = *impl_;
   566→    const bool fb_on = (m.cfg.feedback != 0.0);
   567→    const double fb = m.cfg.feedback;
   568→    const int nv = active_voices_;
   569→    const int interval = m.cfg.detection_interval;
   570→
   571→    for (int k = 0; k < n; ++k) {
   572→        // Control rate, stream-aligned: accumulate the RAW input (pre-feedback,
   573→        // as the Python pushes xb = x) and fire push+estimate when the interval
   574→        // boundary is crossed. The estimate applies from the boundary sample
   575→        // onward. NOTE (documented delta): the Python reference estimates with
   576→        // the full 64-sample block in the window BEFORE processing that block,
   577→        // which is impossible in a streaming API that writes every output on
   578→        // arrival; with 64-sample-aligned calls the native control timing
   579→        // lags the Python's by at most one interval.
   580→        m.det_pending[m.det_n++] = in[k];
   581→        if (m.det_n >= interval) {
   582→            m.det.pushBlock(m.det_pending.data(), m.det_n);
   583→            m.det_n = 0;
   584→            m.last_f0 = m.det.estimate();
   585→            m.updateSpacingTarget(m.last_f0);
   586→        }
   587→
   588→        const double xin = static_cast<double>(in[k]) + (fb_on ? fb * m.fb_state : 0.0);
   589→        m.push(xin);
   590→
   591→        double acc = 0.0;
   592→        for (int i = 0; i < nv; ++i) {
   593→            VoiceState& v = m.vs[i];
   594→            m.slewEngagement(v);  // engagement/parking slews (per sample)
   595→            acc += m.processVoiceSample(v);
   596→        }
   597→        // voice-count normalisation (ramped)
   598→        if (m.norm != m.norm_target) {
   599→            const double step = m.e_step;
   600→            if (m.norm < m.norm_target)
   601→                m.norm = std::min(m.norm_target, m.norm + step);
   602→            else
   603→                m.norm = std::max(m.norm_target, m.norm - step);
   604→        }
   605→        const double y = acc * m.norm;
   606→        out[k] = static_cast<float>(y);
   607→        m.fb_state = fb_on ? y : 0.0;
   608→    }
   609→}
   610→
   611→const VoiceStats& AlignedDelayLine::voiceStats(int v) const {
   612→    static VoiceStats empty;
   613→    if (!impl_ || v < 0 || v >= impl_->cfg.max_voices) return empty;
   614→    const VoiceState& s = impl_->vs[v];
   615→    static thread_local VoiceStats out;  // diagnostics scratch (copy if kept)
   616→    out.wraps = s.wraps;
   617→    out.commits = s.commits;
   618→    out.engaged = (s.e >= 1.0) ? 1 : 0;
   619→    out.parked = (s.e <= 0.0 && s.a >= 1.0) ? 1 : 0;
   620→    out.spacing = s.S;
   621→    out.phase = s.frac;
   622→    out.parked_delay = out.parked ? s.d_park : -1;
   623→    return out;
   624→}

   643→bool MultichannelShifter::prepare(const Config& cfg, int channels) {
   644→    if (channels < 1 || channels > 64) return false;
   645→    if (mimpl_) {
   646→        for (auto* e : mimpl_->engines) delete e;
   647→        delete mimpl_;
   648→        mimpl_ = nullptr;
   649→    }
   650→    mimpl_ = new MImpl();
   651→    mimpl_->cfg = cfg;
   652→    mimpl_->channels = channels;
   653→    for (int c = 0; c < channels; ++c) {
   654→        auto* e = new AlignedDelayLine();
   655→        if (!e->prepare(cfg)) {
   656→            delete e;
   657→            for (auto* x : mimpl_->engines) delete x;
   658→            delete mimpl_;
   659→            mimpl_ = nullptr;
   660→            return false;
   661→        }
   662→        mimpl_->engines.push_back(e);
   663→    }
   664→    return true;
   665→}
