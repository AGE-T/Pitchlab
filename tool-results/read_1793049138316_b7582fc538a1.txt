     9→// in main(), and add `#include <limits>` to the include block. It reuses

    14→// The 13-category battery it completes (the 7 P1 cases cover the rest):
    15→//   r15acc_ratio_change_isolation, r15acc_level_accuracy,
    16→//   r15acc_block_boundary_semantics, r15acc_reset_reprepare_determinism,
    17→//   r15acc_stereo_surface, r15acc_robustness_battery, r15acc_invalid_args.
    18→// Every gate was fixed before running (no widening-to-pass); where a bound
    19→// is implementation-independent-by-necessity the literal value is recorded
    20→// (the P1 pattern). Budget: ~6 s of extra suite runtime at 48 kHz.
    21→
    22→void casesR15Acc(std::vector<CaseResult>& R) {
    23→    const int sr = 48000;
    24→
    25→    // FNV-1a 64 over raw output float bits (same law as default_neutrality).
    26→    auto fnv1a64 = [](const std::vector<float>& x) {
    27→        uint64_t h = 1469598103934665603ULL;
    28→        for (float s : x) {
    29→            uint32_t b;
    30→            std::memcpy(&b, &s, 4);
    31→            for (int i = 0; i < 4; ++i) {
    32→                h ^= static_cast<uint64_t>((b >> (8 * i)) & 0xffu);
    33→                h *= 1099511628211ULL;
    34→            }
    35→        }
    36→        return h;
    37→    };
    38→    const double kNaN = std::numeric_limits<double>::quiet_NaN();
    39→    const double kInf = std::numeric_limits<double>::infinity();
    40→
    41→    {  // 1. r15acc_ratio_change_isolation: a mid-stream per-voice ratio
    42→        //    change touches ONLY its own voice. Measurement: voice 3 is
    43→        //    muted throughout two parallel reference runs (identical
    44→        //    input/config/setter cadence), one WITH the voice-3 ratio
    45→        //    change and one without -- the 7 other voices' sum must stay
    46→        //    BIT-IDENTICAL for the whole stream (feedback 0: the global
    47→        //    feedback is wet-path-only and would legitimately couple the
    48→        //    voices, so isolation is measured feedback-free). Voice 3
    49→        //    itself must converge to its new pitch within 1.0 cent.
    50→        CaseResult c{"r15acc_ratio_change_isolation"};
    51→        const double semis[8] = {-12.0, -7.0, -4.0, 3.0,
    52→                                 -5.0,  5.0,  -7.0, 12.0};
    53→        const double f0in = 220.0;
    54→        const int P = 118784;  // 2.47 s = 1024*116 (block boundary)
    55→        const int n = P + 3 * sr;
    56→        const auto in = gen::tone(f0in, sr, n, 0.5);
    57→        Config cfg;  // voices=8, spread off, ratio 1 (targets come from the
    58→        cfg.voices = 8;  // per-voice overrides), feedback 0 (see above)
    59→        cfg.spread = false;
    60→        const double r3a = std::pow(2.0, 3.0 / 12.0);   // voice 3 initial
    61→        const double r3b = std::pow(2.0, -2.0 / 12.0);  // voice 3 new target
    62→
    63→        auto runMuted = [&](bool change) {
    64→            AlignedDelayLine e;
    65→            check(c, e.prepare(cfg), "prepare failed");
    66→            for (int j = 0; j < 8; ++j)
    67→                e.setVoiceRatio(j, std::pow(2.0, semis[j] / 12.0));
    68→            e.setVoiceMute(3, true);  // voice 3 muted throughout
    69→            std::vector<float> out(n, 0.0f);
    70→            int done = 0;
    71→            while (done < n) {
    72→                const int k = std::min(1024, n - done);
    73→                e.process(in.data() + done, out.data() + done, k);
    74→                done += k;
    75→                if (done == P)
    76→                    e.setVoiceRatio(3, change ? r3b : r3a);
    77→            }
    78→            return out;
    79→        };
    80→        const auto outY = runMuted(true);
    81→        const auto outY0 = runMuted(false);
    82→        double otherMax = 0.0;
    83→        for (int k = 0; k < n; ++k)
    84→            otherMax =
    85→                std::max(otherMax, std::abs((double)outY[k] - outY0[k]));
    86→        addValue(c, "other_voices_fullstream_max_diff", otherMax);
    87→        check(c, otherMax == 0.0,
    88→              "voice 3's mid-stream ratio change perturbed the other voices");
    89→
    90→        // voice 3 alone (the other 7 muted): pitch before and after P.
    91→        {
    92→            AlignedDelayLine e;
    93→            check(c, e.prepare(cfg), "prepare failed");
    94→            for (int j = 0; j < 8; ++j) {
    95→                e.setVoiceRatio(j, std::pow(2.0, semis[j] / 12.0));
    96→                if (j != 3) e.setVoiceMute(j, true);
    97→            }
    98→            std::vector<float> out(n, 0.0f);
    99→            int done = 0;
   100→            while (done < n) {
   101→                const int k = std::min(1024, n - done);
   102→                e.process(in.data() + done, out.data() + done, k);
   103→                done += k;
   104→                if (done == P) e.setVoiceRatio(3, r3b);
   105→            }
   106→            // R13-d windows: 65536 samples (1.365 s) >= 0.5 s and >= 200
   107→            // periods of 261.6 Hz (0.765 s); post window 2 s vs 196 Hz.
   108→            const double fPre =
   109→                meas::pitchFFT(out.data() + (P - 65536), 65536, sr,
   110→                               50.0, 4000.0);
   111→            const double cPre = meas::centsBetween(fPre, f0in * r3a);
   112→            addValue(c, "voice3_pre_change_cents", cPre);
   113→            check(c, std::abs(cPre) <= 1.0,
   114→                  "voice 3 pre-change pitch off by > 1.0 cent");
   115→            const int post0 = P + sr;  // glide (worst ~0.27 s) settled
   116→            const double fPost = meas::pitchFFT(out.data() + post0,
   117→                                                n - post0, sr, 50.0, 4000.0);
   118→            const double cPost = meas::centsBetween(fPost, f0in * r3b);
   119→            addValue(c, "voice3_post_change_cents", cPost);
   120→            check(c, std::abs(cPost) <= 1.0,
   121→                  "voice 3 did not converge to its new ratio within 1.0 cent");
   122→        }
   123→        c.pass = c.note.empty();
   124→        R.push_back(std::move(c));
   125→    }
   126→
   127→    {  // 2. r15acc_level_accuracy: per-voice gains {0.1, 0.25, 0.5, 1.0} on
   128→        //    a steady tone, ONE active voice per measurement (7 muted),
   129→        //    steady RMS ratio vs the expected level within +/- 0.1 dB.
   130→        //    Expected is computed in-test from the header law: final voice
   131→        //    gain = normalisation * user_gain with normalisation = 1/voices.
   132→        CaseResult c{"r15acc_level_accuracy"};
   133→        Config cfg;
   134→        cfg.ratio = semitoneRatio(400.0);
   135→        cfg.voices = 8;
   136→        cfg.spread = false;
   137→        const auto in = gen::tone(220.0, sr, 3 * sr, 0.5);
   138→        const int settle = sr;  // gain/mute ramps (720) + detector lock
   139→        const double gains[4] = {0.1, 0.25, 0.5, 1.0};
   140→        double worst = 0.0;
   141→        for (double g : gains) {
   142→            AlignedDelayLine e;
   143→            check(c, e.prepare(cfg), "prepare failed");
   144→            e.setVoiceGain(0, g);
   145→            for (int j = 1; j < 8; ++j) e.setVoiceMute(j, true);
   146→            std::vector<float> out(in.size(), 0.0f);
   147→            e.process(in.data(), out.data(), static_cast<int>(in.size()));
   148→            const int w = static_cast<int>(in.size()) - settle;  // 2 s
   149→            const double ro = meas::rms(out.data() + settle, w);
   150→            const double ri = meas::rms(in.data() + settle, w);
   151→            const double expected = g / cfg.voices;  // header law
   152→            const double db = 20.0 * std::log10(ro / ri) -
   153→                              20.0 * std::log10(expected);
   154→            char vk[40];
   155→            std::snprintf(vk, sizeof(vk), "err_db_gain_%.2f", g);
   156→            addValue(c, vk, db);
   157→            worst = std::max(worst, std::abs(db));
   158→        }
   159→        addValue(c, "worst_err_db", worst);
   160→        check(c, worst <= 0.1,
   161→              "steady level outside +/-0.1 dB of the expected gain");
   162→        c.pass = c.note.empty();
   163→        R.push_back(std::move(c));
   164→    }
   165→
   166→    {  // 3. r15acc_block_boundary_semantics: a parameter change issued
   167→        //    between process() calls applies from the NEXT block as a
   168→        //    STREAM-POSITION event. Run A: 64-sample blocks; run B: the
   169→        //    identical stream in 150-sample blocks; the same simultaneous
   170→        //    setVoiceBaseDelay + setVoiceGain + setWetDry fire at stream
   171→        //    position P (a boundary in BOTH splittings) -> outputs
   172→        //    identical (max|delta| == 0), proving the semantics are
   173→        //    position-, not call-pattern-, driven. Plus: 20 small gain
   174→        //    steps vs one jump to the same final target -> identical final
   175→        //    steady state (the ramp terminates exactly at its target).
   176→        CaseResult c{"r15acc_block_boundary_semantics"};
   177→        Config cfg;
   178→        cfg.ratio = semitoneRatio(400.0);
   179→        cfg.voices = 8;
   180→        cfg.spread = false;
   181→        cfg.feedback = 0.25;
   182→        const auto in = gen::tone(220.0, sr, 24000, 0.5);
   183→        const int n = static_cast<int>(in.size());
   184→        const int P = 9600;  // 64*150: a block boundary in both splittings
   185→
   186→        auto runSplit = [&](int block) {
   187→            AlignedDelayLine e;
   188→            check(c, e.prepare(cfg), "prepare failed");
   189→            std::vector<float> out(n, 0.0f);
   190→            int done = 0;
   191→            while (done < n) {
   192→                const int k = std::min(block, n - done);
   193→                e.process(in.data() + done, out.data() + done, k);
   194→                done += k;
   195→                if (done == P) {  // simultaneous multi-parameter update
   196→                    e.setVoiceBaseDelay(5, 1200.0);
   197→                    e.setVoiceGain(2, 0.5);
   198→                    e.setWetDry(0.5);
   199→                }
   200→            }
   201→            return out;
   202→        };
   203→        const auto outA = runSplit(64);
   204→        const auto outB = runSplit(150);
   205→        double pre = 0.0, post = 0.0;
   206→        for (int k = 0; k < n; ++k) {
   207→            const double d = std::abs((double)outA[k] - outB[k]);
   208→            if (k < P) pre = std::max(pre, d);
   209→            else post = std::max(post, d);
   210→        }
   211→        addValue(c, "pre_change_max_diff", pre);
   212→        addValue(c, "post_change_max_diff", post);
   213→        check(c, pre == 0.0 && post == 0.0,
   214→              "application is not stream-position (block-boundary) semantics");
   215→
   216→        // sanity: the boundary change really changed the stream (the gate
   217→        // above would otherwise be vacuous)
   218→        {
   219→            AlignedDelayLine e;
   220→            e.prepare(cfg);
   221→            std::vector<float> out(n, 0.0f);
   222→            e.process(in.data(), out.data(), n);
   223→            double d = 0.0;
   224→            for (int k = P; k < n; ++k)
   225→                d = std::max(d, std::abs((double)outA[k] - out[k]));
   226→            addValue(c, "change_effect_from_P", d);
   227→            check(c, d > 0.0, "the boundary parameter change had no effect");
   228→        }
   229→
   230→        // gradual (20 x 0.025 steps) vs one jump, both ending at gain 0.5
   231→        {
   232→            const int n2 = 24000;
   233→            const auto in2 = gen::tone(220.0, sr, n2, 0.5);
   234→            auto runGain = [&](bool gradual) {
   235→                AlignedDelayLine e;
   236→                check(c, e.prepare(cfg), "prepare failed");
   237→                std::vector<float> out(n2, 0.0f);
   238→                int done = 0, step = 0;
   239→                while (done < n2) {
   240→                    const int k = std::min(480, n2 - done);
   241→                    e.process(in2.data() + done, out.data() + done, k);
   242→                    done += k;
   243→                    if (gradual) {
   244→                        if (done >= 2400 && step < 20) {
   245→                            e.setVoiceGain(3, 1.0 - 0.025 * (step + 1));
   246→                            ++step;
   247→                        }
   248→                    } else if (done == 2400) {
   249→                        e.setVoiceGain(3, 0.5);
   250→                    }
   251→                }
   252→                return out;
   253→            };
   254→            const auto outG = runGain(true);
   255→            const auto outJ = runGain(false);
   256→            const int settled = 2400 + 19 * 480 + 720 + 64;  // last ramp done
   257→            double steady = 0.0, trans = 0.0;
   258→            for (int k = 0; k < n2; ++k) {
   259→                const double d = std::abs((double)outG[k] - outJ[k]);
   260→                if (k < settled) trans = std::max(trans, d);
   261→                else steady = std::max(steady, d);
   262→            }
   263→            addValue(c, "gradual_vs_jump_transition_max_diff", trans);
   264→            addValue(c, "gradual_vs_jump_steady_max_diff", steady);
   265→            check(c, steady == 0.0,
   266→                  "gradual and jumped gain differ in the final steady state");
   267→            check(c, trans > 0.0,
   268→                  "gradual and jumped gain were identical throughout "
   269→                  "(vacuous comparison)");
   270→        }
   271→        c.pass = c.note.empty();
   272→        R.push_back(std::move(c));
   273→    }
   274→
   275→    {  // 4. r15acc_reset_reprepare_determinism: a scripted event sequence
   276→        //    (per-voice ratios + gains + mute/unmute + wet/dry moves +
   277→        //    bypass enter/exit + base-delay change) -> full-run output hash;
   278→        //    then reset() -> replay, prepare(same cfg) -> replay, destroy +
   279→        //    fresh engine -> replay: all four must match BIT-EXACTLY.
   280→        //    NOTE (semantics): reset() KEEPS the parameter values while
   281→        //    prepare() restores the fresh-engine defaults, so the script is
   282→        //    CLOSED -- it restores the neutral surface at t7 (setRatio()
   283→        //    re-derives every voice target through the engine's own
   284→        //    derivation path and clears the overrides). The closure is
   285→        //    itself part of the gate: post-reset must be behaviourally
   286→        //    identical to the fresh post-prepare state.
   287→        CaseResult c{"r15acc_reset_reprepare_determinism"};
   288→        Config cfg;
   289→        cfg.ratio = semitoneRatio(400.0);
   290→        cfg.voices = 8;
   291→        cfg.spread = false;
   292→        cfg.feedback = 0.25;
   293→        const int n = 130048;  // 1024 * 127
   294→        const auto in = gen::noise(4242, n, 0.5);
   295→        const double semis[8] = {-12.0, -5.0, 0.0, 7.0,
   296→                                 -3.0,  4.0, -7.0, 12.0};
   297→
   298→        auto script = [&](AlignedDelayLine& e) {
   299→            std::vector<float> out(n, 0.0f);
   300→            int done = 0;
   301→            while (done < n) {
   302→                const int k = std::min(1024, n - done);
   303→                e.process(in.data() + done, out.data() + done, k);
   304→                done += k;
   305→                switch (done) {
   306→                    case 16384:  // t1: per-voice ratios + gains
   307→                        for (int j = 0; j < 8; ++j)
   308→                            e.setVoiceRatio(j,
   309→                                            std::pow(2.0, semis[j] / 12.0));
   310→                        e.setVoiceGain(0, 0.4);
   311→                        e.setVoiceGain(3, 0.7);
   312→                        break;
   313→                    case 32768: e.setVoiceMute(2, true); break;      // t2
   314→                    case 49152: e.setWetDry(0.5); break;             // t3
   315→                    case 65536: e.setBypass(true); break;            // t4
   316→                    case 81920: e.setBypass(false); break;           // t5
   317→                    case 98304: e.setVoiceBaseDelay(4, 1200.0); break;  // t6
   318→                    case 114688:  // t7: restore the neutral surface
   319→                        e.setRatio(cfg.ratio);  // clears every override
   320→                        e.setVoiceMute(2, false);
   321→                        e.setVoiceGain(0, 1.0);
   322→                        e.setVoiceGain(3, 1.0);
   323→                        e.setWetDry(1.0);
   324→                        e.setVoiceBaseDelay(4, 256.0 + 32.0 * 4);
   325→                        break;
   326→                    default: break;
   327→                }
   328→            }
   329→            return out;
   330→        };
   331→
   332→        AlignedDelayLine e;
   333→        check(c, e.prepare(cfg), "prepare failed");
   334→        const auto out1 = script(e);
   335→        const uint64_t h1 = fnv1a64(out1);
   336→        e.reset();
   337→        const auto out2 = script(e);
   338→        const uint64_t h2 = fnv1a64(out2);
   339→        check(c, e.prepare(cfg), "re-prepare failed");  // R13-c in-place path
   340→        const auto out3 = script(e);
   341→        const uint64_t h3 = fnv1a64(out3);
   342→        uint64_t h4 = 0;
   343→        std::vector<float> out4;
   344→        {
   345→            AlignedDelayLine f;
   346→            check(c, f.prepare(cfg), "fresh prepare failed");
   347→            out4 = script(f);
   348→            h4 = fnv1a64(out4);
   349→        }
   350→        std::printf("r15acc_reset_reprepare_determinism hashes: "
   351→                    "run=0x%016llx reset=0x%016llx prepare=0x%016llx "
   352→                    "fresh=0x%016llx\n",
   353→                    static_cast<unsigned long long>(h1),
   354→                    static_cast<unsigned long long>(h2),
   355→                    static_cast<unsigned long long>(h3),
   356→                    static_cast<unsigned long long>(h4));
   357→        auto maxdiff = [](const std::vector<float>& a,
   358→                          const std::vector<float>& b) {
   359→            double d = 0.0;
   360→            for (size_t i = 0; i < a.size(); ++i)
   361→                d = std::max(d, std::abs((double)a[i] - b[i]));
   362→            return d;
   363→        };
   364→        const double dR = maxdiff(out1, out2);
   365→        const double dP = maxdiff(out1, out3);
   366→        const double dF = maxdiff(out1, out4);
   367→        addValue(c, "reset_replay_max_diff", dR);
   368→        addValue(c, "prepare_replay_max_diff", dP);
   369→        addValue(c, "fresh_replay_max_diff", dF);
   370→        addValue(c, "hashes_equal",
   371→                 (h1 == h2 && h2 == h3 && h3 == h4) ? 1.0 : 0.0);
   372→        check(c, dR == 0.0, "reset()+replay differs from the first run");
   373→        check(c, dP == 0.0, "prepare()+replay differs from the first run");
   374→        check(c, dF == 0.0, "fresh-engine replay differs from the first run");
   375→        check(c, h1 == h2 && h2 == h3 && h3 == h4,
   376→              "full-run output hashes differ");
   377→        c.pass = c.note.empty();
   378→        R.push_back(std::move(c));
   379→    }
   380→
   381→    {  // 5. r15acc_stereo_surface: MultichannelShifter 2ch with the full
   382→        //    per-voice surface active (per-voice ratios + one voice muted +
   383→        //    wet 0.5 + a bypass enter/exit mid-run + feedback). Broadcast
   384→        //    parity: the same input on both channels -> identical outputs
   385→        //    and identical per-voice voiceStats. Channel isolation: L_out
   386→        //    with only-L input == L_out with L+R input, exactly.
   387→        CaseResult c{"r15acc_stereo_surface"};
   388→        Config cfg;
   389→        cfg.ratio = semitoneRatio(400.0);
   390→        cfg.voices = 8;
   391→        cfg.spread = false;
   392→        cfg.feedback = 0.25;
   393→        const double semis[8] = {-12.0, -5.0, 0.0, 7.0,
   394→                                 -3.0,  4.0, -7.0, 12.0};
   395→        const int n = 147456;             // 1024 * 144 (3.072 s)
   396→        const int t1 = 49152, t2 = 98304;  // bypass enter / exit
   397→        const auto tin = gen::tone(220.0, sr, n, 0.5);
   398→        const auto tR = gen::tone(330.0, sr, n, 0.3);
   399→
   400→        auto runOn = [&](MultichannelShifter& m,
   401→                         const std::vector<float>& inL,
   402→                         const std::vector<float>& inR,
   403→                         std::vector<float>& outL,
   404→                         std::vector<float>& outR) {
   405→            for (int j = 0; j < 8; ++j)
   406→                m.setVoiceRatio(j, std::pow(2.0, semis[j] / 12.0));
   407→            m.setVoiceMute(5, true);
   408→            m.setWetDry(0.5);
   409→            const float* in[2] = {inL.data(), inR.data()};
   410→            float* out[2] = {outL.data(), outR.data()};
   411→            int done = 0;
   412→            while (done < n) {
   413→                const int k = std::min(1024, n - done);
   414→                m.process(in, out, 2, k);
   415→                done += k;
   416→                if (done == t1) m.setBypass(true);
   417→                if (done == t2) m.setBypass(false);
   418→            }
   419→        };
   420→
   421→        // broadcast parity
   422→        {
   423→            MultichannelShifter m;
   424→            check(c, m.prepare(cfg, 2), "prepare failed");
   425→            std::vector<float> Lo(n, 0.0f), Ro(n, 0.0f);
   426→            runOn(m, tin, tin, Lo, Ro);
   427→            double od = 0.0;
   428→            for (int k = 0; k < n; ++k)
   429→                od = std::max(od, std::abs((double)Lo[k] - Ro[k]));
   430→            addValue(c, "broadcast_L_vs_R_max_diff", od);
   431→            check(c, od == 0.0, "broadcast outputs differ between channels");
   432→            int statsMismatch = 0;
   433→            for (int v = 0; v < 8; ++v) {
   434→                const VoiceStats a = m.voiceStats(0, v);
   435→                const VoiceStats b = m.voiceStats(1, v);
   436→                if (a.wraps != b.wraps || a.commits != b.commits ||
   437→                    a.engaged != b.engaged || a.parked != b.parked ||
   438→                    a.spacing != b.spacing || a.phase != b.phase ||
   439→                    a.parked_delay != b.parked_delay || a.ratio != b.ratio)
   440→                    ++statsMismatch;
   441→            }
   442→            addValue(c, "voicestats_mismatched_voices", statsMismatch);
   443→            check(c, statsMismatch == 0, "voiceStats differ across channels");
   444→        }

   454→                runOn(m, tin, sil, La, Ra);

   457→                MultichannelShifter m;
   458→                m.prepare(cfg, 2);
   459→                runOn(m, tin, tR, Lb, Rb);
   460→            }
   461→            double dL = 0.0;
   462→            for (int k = 0; k < n; ++k)
   463→                dL = std::max(dL, std::abs((double)La[k] - Lb[k]));
   464→            addValue(c, "L_out_onlyL_vs_LplusR_max_diff", dL);
   465→            check(c, dL == 0.0, "L output depends on the R input (leak)");
   466→            const double rSil = meas::peak(Ra.data(), n);
   467→            addValue(c, "R_out_peak_with_silent_R", rSil);
   468→            check(c, rSil == 0.0,
   469→                  "silent R input produced non-zero R output");
   470→        }
   471→        c.pass = c.note.empty();
   472→        R.push_back(std::move(c));

   475→    {  // 6. r15acc_robustness_battery: 8 voices + the surface active
   476→        //    (wet 0.5, voice 5 muted, one bypass enter/exit mid-run) across
   477→        //    silence / -60 dBFS sine / 440 Hz tone / 3-tone chord / white
   478→        //    noise / 100 Hz impulse train. Gates: all finite and bounded,
   479→        //    silence in -> EXACTLY zero out, -60 dBFS spectral spurs <=
   480→        //    -40 dB rel. output RMS outside the expected fundamentals, and
   481→        //    bit-exact determinism across a replay.
   482→        CaseResult c{"r15acc_robustness_battery"};
   483→        Config cfg;
   484→        cfg.ratio = semitoneRatio(400.0);  // spread table ON: 8 intervals
   485→        cfg.voices = 8;                    // around +400c
   486→        const int n = 192000;              // 4 s
   487→        const int t1 = 49152, t2 = 98304;  // bypass enter / exit
   488→
   489→        auto runBatt = [&](const std::vector<float>& in,
   490→                           std::vector<float>& out) {
   491→            AlignedDelayLine e;
   492→            check(c, e.prepare(cfg), "prepare failed");
   493→            e.setVoiceMute(5, true);
   494→            e.setWetDry(0.5);
   495→            int done = 0;
   496→            while (done < n) {
   497→                const int k = std::min(1024, n - done);
   498→                e.process(in.data() + done, out.data() + done, k);
   499→                done += k;
   500→                if (done == t1) e.setBypass(true);
   501→                if (done == t2) e.setBypass(false);
   502→            }
   503→        };
   504→
   505→        // (a) digital silence: out must be EXACTLY 0 (wet 0.5 + bypass
   506→        //     regions included: every path is exactly zero on zero input)
   507→        {
   508→            const std::vector<float> sil(n, 0.0f);
   509→            std::vector<float> out(n, 0.0f);
   510→            runBatt(sil, out);
   511→            const double mx = meas::peak(out.data(), n);
   512→            addValue(c, "silence_max_abs_out", mx);
   513→            check(c, meas::allFinite(out.data(), n), "silence: non-finite");
   514→            check(c, mx == 0.0, "silence in produced non-zero output");
   515→        }
   516→
   517→        // (b) -60 dBFS sine: finite + spur floor. The analysed stretch is
   518→        //     the steady post-bypass-exit tail (65536 samples, Hann,
   519→        //     amplitude = 4|X_k|/N). Expected components: the dry-path
   520→        //     fundamental (the wet/dry mix legitimately carries it) and
   521→        //     the 8 spread-table shifted fundamentals; guard +-10 bins
   522→        //     each (covers the Hann leakage skirt); everything else must
   523→        //     stay <= -40 dB relative to the output RMS. NOTE: single-bin
   524→        //     amplitudes underestimate an OFF-BIN spur by up to 1.42 dB
   525→        //     (Hann scalloping), so the gate is conservative in that
   526→        //     direction; the measured margin is recorded.
   527→        {
   528→            const auto in = gen::tone(220.0, sr, n, 0.001);
   529→            std::vector<float> out(n, 0.0f);
   530→            runBatt(in, out);
   531→            check(c, meas::allFinite(out.data(), n), "-60dB: non-finite");
   532→            const int N = 65536;
   533→            const int off = n - N;  // 126464 > t2+480: bypass fully off
   534→            std::vector<double> re(N, 0.0), im(N, 0.0);
   535→            for (int i = 0; i < N; ++i) {
   536→                const double w = 0.5 - 0.5 * std::cos(2.0 * M_PI * i / N);
   537→                re[i] = static_cast<double>(out[off + i]) * w;
   538→            }
   539→            meas::fft(re.data(), im.data(), N);
   540→            const double stSemis[8] = {0.0,  7.0,  4.0,  12.0,
   541→                                       -5.0, -12.0, 3.0, -3.0};
   542→            double guard[9];
   543→            guard[0] = 220.0;  // the dry-path fundamental (wet 0.5)
   544→            for (int j = 0; j < 8; ++j)
   545→                guard[j + 1] =
   546→                    220.0 * std::pow(2.0, (4.0 + stSemis[j]) / 12.0);
   547→            const double rmsOut = meas::rms(out.data() + off, N);
   548→            double spur = 0.0;
   549→            int spurBin = -1;
   550→            for (int k = 1; k < N / 2; ++k) {
   551→                bool g = false;
   552→                for (int q = 0; q < 9 && !g; ++q) {
   553→                    const double bk = guard[q] * N / static_cast<double>(sr);
   554→                    if (std::abs(k - bk) <= 10.0) g = true;
   555→                }
   556→                if (g) continue;
   557→                const double amp = 4.0 * std::sqrt(re[k] * re[k] +
   558→                                                   im[k] * im[k]) / N;
   559→                if (amp > spur) {
   560→                    spur = amp;
   561→                    spurBin = k;
   562→                }
   563→            }
   564→            const double spurDb = 20.0 * std::log10(spur / rmsOut);
   565→            addValue(c, "minus60_spur_db_rel_rms", spurDb);
   566→            addValue(c, "minus60_spur_bin", spurBin);
   567→            addValue(c, "minus60_spur_hz",
   568→                     spurBin * static_cast<double>(sr) / N);
   569→            check(c, spurDb <= -40.0,
   570→                  "-60 dBFS spur floor above -40 dB rel. output RMS");
   571→        }
   572→
   573→        // (c) single 440 Hz tone, (d) 3-tone chord, (e) white noise,
   574→        //     (f) 100 Hz impulse train: finite + bounded (+ energy in a
   575→        //     documented window for the impulse train).
   576→        {
   577→            const auto in = gen::tone(440.0, sr, n, 0.5);
   578→            std::vector<float> out(n, 0.0f);
   579→            runBatt(in, out);
   580→            const double pk = meas::peak(out.data(), n);
   581→            addValue(c, "tone440_peak", pk);
   582→            check(c, meas::allFinite(out.data(), n), "440 Hz: non-finite");
   583→            check(c, pk <= 1.0, "440 Hz: peak above 2x input peak");
   584→        }
   585→        {
   586→            const std::vector<double> fs = {220.0, 330.0, 440.0};
   587→            const auto in = gen::chord(fs, sr, n, 0.5);
   588→            std::vector<float> out(n, 0.0f);
   589→            runBatt(in, out);
   590→            const double pk = meas::peak(out.data(), n);
   591→            const double inPk = meas::peak(in.data(), n);
   592→            addValue(c, "chord_peak", pk);
   593→            addValue(c, "chord_two_x_input_peak", 2.0 * inPk);
   594→            check(c, meas::allFinite(out.data(), n), "chord: non-finite");
   595→            check(c, pk <= 2.0 * inPk, "chord: peak above 2x input peak");
   596→        }
   597→        {
   598→            const auto in = gen::noise(777, n, 0.5);
   599→            std::vector<float> out(n, 0.0f);
   600→            runBatt(in, out);
   601→            const double pk = meas::peak(out.data(), n);
   602→            addValue(c, "noise_peak", pk);
   603→            check(c, meas::allFinite(out.data(), n), "noise: non-finite");
   604→            check(c, pk <= 1.0, "noise: peak above 2x input peak");
   605→        }
   606→        {
   607→            std::vector<float> in(n, 0.0f);  // 100 Hz impulse train
   608→            for (int k = 0; k < n; k += 480) in[k] = 1.0f;
   609→            std::vector<float> out(n, 0.0f);
   610→            runBatt(in, out);
   611→            const double pk = meas::peak(out.data(), n);
   612→            double ein = 0.0, eout = 0.0;
   613→            for (int k = 0; k < n; ++k) {
   614→                ein += (double)in[k] * in[k];
   615→                eout += (double)out[k] * out[k];
   616→            }
   617→            // GATE ANALYSIS: with 8 normalised voices the incoherent wet
   618→            // mix legitimately carries ~1/8 of the input energy (each
   619→            // voice at 1/8 amplitude) and the dry half adds its own 1/4
   620→            // power share, so the honest robustness bound is a WINDOW,
   621→            // not unity: no runaway growth (fb = 0) and no total collapse.
   622→            // The literal ratio is recorded (expected ~ -5..-6 dB).
   623→            const double ratioDb = 10.0 * std::log10(eout / ein);
   624→            addValue(c, "impulsetrain_peak", pk);
   625→            addValue(c, "impulsetrain_energy_ratio_db", ratioDb);
   626→            check(c, meas::allFinite(out.data(), n),
   627→                  "impulse train: non-finite");
   628→            check(c, pk <= 2.0, "impulse train: peak above 2x input peak");
   629→            check(c, ratioDb >= -12.0 && ratioDb <= 6.0,
   630→                  "impulse-train energy ratio outside [-12, +6] dB");
   631→        }
   632→
   633→        // (g) determinism across a replay (fresh engine, same surface)
   634→        {
   635→            const auto in = gen::noise(777, n, 0.5);
   636→            std::vector<float> o1(n, 0.0f), o2(n, 0.0f);
   637→            runBatt(in, o1);
   638→            runBatt(in, o2);
   639→            double d = 0.0;
   640→            for (int k = 0; k < n; ++k)
   641→                d = std::max(d, std::abs((double)o1[k] - o2[k]));
   642→            const bool heq = fnv1a64(o1) == fnv1a64(o2);
   643→            addValue(c, "replay_max_diff", d);
   644→            addValue(c, "replay_hash_equal", heq ? 1.0 : 0.0);
   645→            check(c, d == 0.0 && heq, "replay is not bit-identical");
   646→        }
   647→        c.pass = c.note.empty();
   648→        R.push_back(std::move(c));
   649→    }

   651→    {  // 7. r15acc_invalid_args: the invalid-argument matrix. Rejected
   652→        //    calls (out-of-range voice index, non-positive ratio/gain,
   653→        //    non-finite anything, base delay <= 0 or beyond the ring
   654→        //    range) must be NO-OPS: a run with the invalid calls sprinkled
   655→        //    between blocks is BIT-IDENTICAL to a run without them, and
   656→        //    the engine stays usable afterwards. setWetDry CLAMPS to
   657→        //    [0,1] per the header contract -- tested as identity with the
   658→        //    clamped endpoint's own run.
   659→        CaseResult c{"r15acc_invalid_args"};
   660→
   661→        // (a) every entry point on an UNPREPARED engine: no crash, no
   662→        //     effect, not ready.
   663→        {
   664→            AlignedDelayLine e;
   665→            float buf[64] = {0.0f};

   667→            e.setVoices(4);

   693→
   694→        // (b) the rejection matrix on a prepared engine
   695→        Config cfg;
   696→        cfg.ratio = semitoneRatio(400.0);
   697→        cfg.voices = 8;
   698→        cfg.spread = false;
   699→        const int n = 48000;
   700→        const auto in = gen::tone(220.0, sr, n, 0.5);
   701→        auto runMatrix = [&](bool withInvalid) {
   702→            AlignedDelayLine e;
   703→            check(c, e.prepare(cfg), "prepare failed");
   704→            std::vector<float> out(n, 0.0f);
   705→            int done = 0;
   706→            while (done < n) {
   707→                const int k = std::min(512, n - done);
   708→                e.process(in.data() + done, out.data() + done, k);
   709→                done += k;
   710→                if (done == 5120) {  // valid surface change in BOTH runs
   711→                    e.setVoiceRatio(3, std::pow(2.0, 5.0 / 12.0));
   712→                    e.setVoiceGain(2, 0.5);
   713→                    e.setVoiceBaseDelay(4, 700.0);
   714→                    e.setWetDry(0.6);
   715→                }
   716→                if (withInvalid && done == 10240) {
   717→                    // indices
   718→                    e.setVoiceRatio(-1, 2.0);
   719→                    e.setVoiceRatio(8, 2.0);
   720→                    e.setVoiceGain(-1, 0.5);
   721→                    e.setVoiceGain(8, 0.5);
   722→                    e.setVoiceMute(-1, true);
   723→                    e.setVoiceMute(8, true);
   724→                    e.setVoiceBaseDelay(-1, 100.0);
   725→                    e.setVoiceBaseDelay(8, 100.0);
   726→                    // values (must not clobber live overrides either)
   727→                    e.setVoiceRatio(3, 0.0);
   728→                    e.setVoiceRatio(3, -1.5);
   729→                    e.setVoiceRatio(3, kNaN);
   730→                    e.setVoiceRatio(3, kInf);
   731→                    e.setVoiceGain(2, -0.25);
   732→                    e.setVoiceGain(2, kNaN);
   733→                    e.setVoiceBaseDelay(4, 0.0);
   734→                    e.setVoiceBaseDelay(4, -3.0);
   735→                    e.setVoiceBaseDelay(4, kNaN);
   736→                    e.setVoiceBaseDelay(4, 249848.0);  // ring-range edge
   737→                    e.setVoiceBaseDelay(4, 300000.0);
   738→                    e.setWetDry(kNaN);

   740→                if (withInvalid && done == 15360) {
   741→                    e.setVoiceRatio(-2, -0.0);
   742→                    e.setVoiceGain(2, -1e-9);
   743→                    e.setVoiceBaseDelay(4, 262144.0);
   744→                }
   745→            }
   746→            return out;
   747→        };
   748→        const auto outRef = runMatrix(false);
   749→        const auto outInv = runMatrix(true);
   750→        double dInv = 0.0;
   751→        for (int k = 0; k < n; ++k)
   752→            dInv = std::max(dInv, std::abs((double)outRef[k] - outInv[k]));
   753→        addValue(c, "rejected_calls_output_max_diff", dInv);
   754→        check(c, dInv == 0.0,
   755→              "rejected calls changed the engine state or output");
   756→
   757→        // engine still usable after the matrix: finite, non-silent output
   758→        {
   759→            AlignedDelayLine e;
   760→            check(c, e.prepare(cfg), "prepare failed");
   761→            std::vector<float> out(n, 0.0f);
   762→            int done = 0;
   763→            while (done < n) {
   764→                const int k = std::min(512, n - done);
   765→                e.process(in.data() + done, out.data() + done, k);
   766→                done += k;
   767→                if (done == 5120) {
   768→                    e.setVoiceRatio(3, std::pow(2.0, 5.0 / 12.0));
   769→                    e.setVoiceGain(2, 0.5);
   770→                    e.setVoiceBaseDelay(4, 700.0);
   771→                    e.setWetDry(0.6);
   772→                }
   773→                if (done == 10240 || done == 15360) {
   774→                    e.setVoiceRatio(8, 2.0);
   775→                    e.setVoiceGain(-1, -1.0);
   776→                    e.setVoiceBaseDelay(4, 1e9);
   777→                    e.setWetDry(kNaN);

   781→            check(c, meas::allFinite(out.data(), n),
   782→                  "non-finite output after rejected calls");
   783→            check(c, meas::peak(out.data(), n) > 0.0,
   784→                  "engine silent after rejected calls");
   785→        }
   786→
   787→        // (c) setWetDry CLAMPS to [0,1] (the header contract): 1.7 == 1.0
   788→        //     run, -0.3 == 0.0 run == the bit-exact delayed copy.
   789→        {
   790→            auto runWet = [&](double wet) {
   791→                AlignedDelayLine e;
   792→                e.prepare(cfg);
   793→                e.setWetDry(wet);

   796→                return out;
   797→            };
   798→            const auto oHi = runWet(1.7);
   799→            const auto oOne = runWet(1.0);
   800→            double dHi = 0.0;
   801→            for (int k = 0; k < n; ++k)
   802→                dHi = std::max(dHi, std::abs((double)oHi[k] - oOne[k]));

   804→            check(c, dHi == 0.0, "setWetDry(1.7) is not clamped to 1.0");
   805→

   807→            AlignedDelayLine probe;
   808→            probe.prepare(cfg);
   809→            const int L = probe.latencySamples();
   810→            double dLo = 0.0;
   811→            for (int k = 0; k < n; ++k) {
   812→                const int src = k - (L - 1);
   813→                const double expect = (src >= 0) ? (double)in[src] : 0.0;
   814→                dLo = std::max(dLo, std::abs((double)oLo[k] - expect));
   815→            }
   816→            addValue(c, "wet_m0p3_vs_delayed_copy_max_diff", dLo);
   817→            check(c, dLo == 0.0,
   818→                  "setWetDry(-0.3) is not clamped to the bit-exact dry "
   819→                  "copy");
   820→        }

   822→        // (d) boundary sanity: the largest IN-RANGE base delay IS accepted

   838→            };
   839→            const auto oA = runDelay(700.0);
   840→            const auto oB = runDelay(249847.0);
   841→            double d = 0.0;
   842→            for (int k = 5120; k < n; ++k)

   847→        }
   848→        c.pass = c.note.empty();
   849→        R.push_back(std::move(c));
   850→    }
